// Publish checked render assets through the existing editor; no scene changes.
var root=Sandbox.Project.Current.GetRootPath();
var folder=System.IO.Path.Combine(root,"output/railhead/extinguisher-v2");
var request=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder,"import-request.json")));
var stage=request["stage"].GetValue<string>();
if(!System.Text.RegularExpressions.Regex.IsMatch(stage,"^[a-z0-9-]+$")) throw new InvalidOperationException("Invalid stage.");
var manifest=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder,stage,"manifest.json")));
if(manifest["revision"].GetValue<string>()!="railhead-extinguisher-v2") throw new InvalidOperationException("Wrong revision.");
var selected=Editor.SceneEditorSession.Active;var playing=Game.IsPlaying;var game=Game.ActiveScene?.Id;var dirty=selected?.HasUnsavedChanges;
if(playing) throw new InvalidOperationException("A play session started; inspect it before replacing its models.");
string Hash(string path)=>Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(System.IO.File.ReadAllBytes(path))).ToLowerInvariant();
foreach(var group in new[]{"source_sha256","protected_sha256"})
    foreach(var pair in manifest[group].AsObject())
        if(Hash(System.IO.Path.Combine(root,pair.Key))!=pair.Value.GetValue<string>()) throw new InvalidOperationException("Shared input changed: "+pair.Key);
var originals=new System.Collections.Generic.HashSet<string>();
foreach(var side in new[]{"east","west"})
    foreach(var extension in new[]{".obj",".mtl",".vmdl"}) originals.Add("Assets/models/maps/railhead/railhead_dress_"+side+extension);
var plans=new System.Collections.Generic.List<(string Source,string Target,string Backup,string Before,string After,int Order)>();
foreach(var pair in manifest["files"].AsObject())
{
    var relative=pair.Key;
    if(relative.Contains("..") || !(originals.Contains(relative)
        || relative.StartsWith("Assets/models/maps/railhead/extinguisher-v2/",StringComparison.Ordinal)
        || relative.StartsWith("Assets/models/props/extinguisher-v2/",StringComparison.Ordinal)
        || relative.StartsWith("Assets/materials/props/extinguisher-v2/",StringComparison.Ordinal)))
        throw new InvalidOperationException("Unexpected target: "+relative);
    if(!new[]{".obj",".mtl",".vmdl",".vmat",".png"}.Any(extension=>relative.EndsWith(extension,StringComparison.Ordinal)))
        throw new InvalidOperationException("Unexpected file type.");
    var source=System.IO.Path.Combine(root,pair.Value["source"].GetValue<string>());
    var target=System.IO.Path.Combine(root,relative);
    var before=pair.Value["before"]?.GetValue<string>();var after=pair.Value["after"].GetValue<string>();
    var backup=before is null?null:System.IO.Path.Combine(root,pair.Value["backup"].GetValue<string>());
    if(Hash(source)!=after || (backup is not null&&Hash(backup)!=before)) throw new InvalidOperationException("Stage or backup changed.");
    if(System.IO.File.Exists(target)&&Hash(target)==after) continue;
    if(before is null?System.IO.File.Exists(target):!System.IO.File.Exists(target)||Hash(target)!=before)
        throw new InvalidOperationException("Shared target changed: "+relative);
    var order=relative.EndsWith(".png",StringComparison.Ordinal)?0:relative.EndsWith(".vmat",StringComparison.Ordinal)?1:
        originals.Contains(relative)?(relative.EndsWith(".vmdl",StringComparison.Ordinal)?4:3):2;
    plans.Add((source,target,backup,before,after,order));
}
var changed=new System.Collections.Generic.List<(string Target,string Backup)>();
try
{
    foreach(var item in plans.OrderBy(p=>p.Order))
    {
        cancellationToken.ThrowIfCancellationRequested();
        if(item.Before is null?System.IO.File.Exists(item.Target):!System.IO.File.Exists(item.Target)||Hash(item.Target)!=item.Before)
            throw new InvalidOperationException("Shared target changed during import: "+item.Target);
        System.IO.Directory.CreateDirectory(System.IO.Path.GetDirectoryName(item.Target));
        var incoming=item.Target+".extinguisher-v2-incoming";
        try
        {
            System.IO.File.Copy(item.Source,incoming,true);
            if(Hash(incoming)!=item.After) throw new InvalidOperationException("Incoming checksum differs.");
            System.IO.File.Move(incoming,item.Target,true);changed.Add((item.Target,item.Backup));
        }
        finally {if(System.IO.File.Exists(incoming)) System.IO.File.Delete(incoming);}
    }
}
catch
{
    foreach(var item in changed.AsEnumerable().Reverse())
        if(item.Backup is not null) System.IO.File.Copy(item.Backup,item.Target,true);
        else if(System.IO.File.Exists(item.Target)) System.IO.File.Delete(item.Target);
    throw;
}
foreach(var item in plans.OrderBy(p=>p.Order)) Editor.AssetSystem.RegisterFile(item.Target);
foreach(var pair in manifest["protected_sha256"].AsObject())
    if(Hash(System.IO.Path.Combine(root,pair.Key))!=pair.Value.GetValue<string>()) throw new InvalidOperationException("Protected file changed during import.");
var preserved=ReferenceEquals(selected,Editor.SceneEditorSession.Active)&&playing==Game.IsPlaying&&game==Game.ActiveScene?.Id&&dirty==selected?.HasUnsavedChanges;
var result=new {Imported=plans.Count,Stage=stage,EditorStatePreserved=preserved,SceneChanged=false,CollisionChanged=false,
    Replaced=manifest["placements_replaced"].GetValue<int>(),TrianglesPerInstance=manifest["triangles_per_instance"].GetValue<int>(),
    StandaloneModel=manifest["standalone_model"].GetValue<string>()};
System.IO.File.WriteAllText(System.IO.Path.Combine(folder,"native-import-"+stage+".json"),System.Text.Json.JsonSerializer.Serialize(result));
if(!preserved) throw new InvalidOperationException("Editor state changed during import; inspect the completed result.");
return result;
