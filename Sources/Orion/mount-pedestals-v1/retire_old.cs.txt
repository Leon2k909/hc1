var root=Sandbox.Project.Current.GetRootPath();
var folder=System.IO.Path.Combine(root,"Sources/Orion/mount-pedestals-v1/review");
var source=System.IO.Path.Combine(root,"Assets/models/maps/orion/mount-pedestals-v1");
var archive=System.IO.Path.Combine(folder,"retired-before");
var expected=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder,"preview-plan.json")))["files"].AsObject();
string Hash(string p)=>Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(System.IO.File.ReadAllBytes(p))).ToLowerInvariant();
var files=System.IO.Directory.GetFiles(source,"before.*");
if(files.Length==0)throw new InvalidOperationException("No old assets; inspect retirement receipt");
foreach(var file in files) {
 var rel=System.IO.Path.GetRelativePath(root,file).Replace('\\','/');
 if(expected.TryGetPropertyValue(rel,out var sha)&&Hash(file)!=sha.GetValue<string>())throw new InvalidOperationException("Old asset changed "+rel);
 if(System.IO.File.Exists(System.IO.Path.Combine(archive,System.IO.Path.GetFileName(file))))throw new InvalidOperationException("Archive exists");
}
System.IO.Directory.CreateDirectory(archive);
var report=new System.Collections.Generic.Dictionary<string,object>();
foreach(var file in files) {
 var rel=System.IO.Path.GetRelativePath(root,file).Replace('\\','/');var target=System.IO.Path.Combine(archive,System.IO.Path.GetFileName(file));
 var hash=Hash(file);System.IO.File.Move(file,target);report[rel]=new{Archive=System.IO.Path.GetRelativePath(root,target).Replace('\\','/'),Sha256=hash};
}
System.IO.File.WriteAllText(System.IO.Path.Combine(folder,"retirement.json"),System.Text.Json.JsonSerializer.Serialize(report));return new{OldFilesRemoved=files.Length,NewPedestalsKept=true};
