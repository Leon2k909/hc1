// Before and after of Orion's tarp pile, rendered by s&box: run with sbox-cli --call run_csharp '{"path":...}'.
// Loads orion.scene's visual roots into an isolated editor scene, so the open scene and play state are never touched and nothing is saved.
// "Before" is the merged model as it is. "After" hides Yard Props' oa_tarp slot, which only the pile uses, and stands the new model where the old one is.
// Reads engine/config.json from the work folder (exposure, width, height, stream_frames, settle_frames, at, yaw, views: [ { name, eye, target, fov } ], in engine coordinates). Writes <view>-before.png and <view>-after.png beside it.
var projectRoot = Sandbox.Project.Current.GetRootPath();
var destination = System.IO.Path.Combine(projectRoot, "output/asset-improvement-20261030/tarp-pile/engine");
var configuration = System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(destination, "config.json"))).AsObject();
var source = Editor.SceneEditorSession.Active?.Scene;
var before = new { Game.IsPlaying, Scene = source?.Name, SceneId = source?.Id, GameSceneId = Game.ActiveScene?.Id };
var visual = new System.Collections.Generic.HashSet<string>(StringComparer.OrdinalIgnoreCase) { "Sun", "Sky", "Fog", "Look", "Orion", "Lights", "Volumetric Fog", "Steam", "Reflection Probes" };
var root = System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(projectRoot, "Assets/scenes/maps/Orion/orion.scene"))).AsObject();
var objects = new System.Text.Json.Nodes.JsonArray();
foreach (var item in root["GameObjects"].AsArray())
    if (visual.Contains(item["Name"]?.GetValue<string>() ?? "")) objects.Add(item.DeepClone());
root["GameObjects"] = objects;
if (root["SceneProperties"]?["NavMesh"] is System.Text.Json.Nodes.JsonObject nav) nav["Enabled"] = false;
if (root["SceneProperties"] is System.Text.Json.Nodes.JsonObject sceneProperties) sceneProperties["WantsSystemScene"] = false;

var model = Model.Load("models/props/orion/tarp_pile_v2/tarp_pile.vmdl");
if (model is null || model.IsError) throw new InvalidOperationException("The new pile hasn't compiled.");
var hiddenMaterial = Material.Load("attachments/m4a1/materials/hidden.vmat");
if (hiddenMaterial is null || hiddenMaterial.IsError) throw new InvalidOperationException("No hidden material.");

var scene = Scene.CreateEditorScene();
var flags = System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.Public | System.Reflection.BindingFlags.NonPublic;
var preRender = typeof(Scene).GetMethod("PreRender", flags, null, Type.EmptyTypes, null) ?? throw new MissingMethodException("Scene.PreRender");
var preCameraRender = typeof(Scene).GetMethod("PreCameraRender", flags, null, Type.EmptyTypes, null) ?? throw new MissingMethodException("Scene.PreCameraRender");
Vector3 Vec(System.Text.Json.Nodes.JsonNode node) { var a = node.AsArray(); return new(a[0].GetValue<float>(), a[1].GetValue<float>(), a[2].GetValue<float>()); }
try
{
    CameraComponent camera;
    ModelRenderer props;
    GameObject pile;
    var slots = new System.Collections.Generic.List<int>();
    using (scene.Push())
    {
        scene.Deserialize(root);
        // Auto exposure wouldn't have settled by the first shots, which would make "before" darker. Held at one value, both match.
        foreach (var tonemap in scene.GetAllComponents<Tonemapping>())
        {
            tonemap.AutoExposureEnabled = true;
            tonemap.MinimumExposure = tonemap.MaximumExposure = configuration["exposure"].GetValue<float>();
            tonemap.ExposureCompensation = 0;
            tonemap.Rate = 10;
        }
        props = scene.GetAllObjects(false).Single(g => g.Name == "Yard Props").Components.Get<ModelRenderer>();
        for (var slot = 0; slot < props.Materials.Count; slot++)
            if ((props.Materials.GetOriginal(slot)?.ResourcePath ?? "").EndsWith("/oa_tarp.vmat")) slots.Add(slot);
        pile = new GameObject(false, "New tarp pile");
        pile.WorldPosition = Vec(configuration["at"]);
        pile.WorldRotation = Rotation.FromYaw(configuration["yaw"].GetValue<float>());
        pile.Components.Create<ModelRenderer>().Model = model;
        var holder = new GameObject(false, "Capture camera");
        camera = holder.Components.Create<CameraComponent>();
        camera.ZNear = 1;
        camera.ZFar = 60000;
        holder.Enabled = true;
        scene.EditorTick(.016f, .016f);
        preRender.Invoke(scene, null);
        preCameraRender.Invoke(scene, null);
    }
    if (slots.Count != 1) throw new InvalidOperationException($"Found {slots.Count} oa_tarp slots.");

    // Nothing above 1024 for the map or 2048 for the pile: the editor is shared, and a whole map at full size has run it out of video memory.
    var textureNames = new[] { "g_tColor", "g_tNormal", "g_tRoughness", "g_tAmbientOcclusion", "SkyTexture" };
    foreach (var material in scene.GetAllComponents<ModelRenderer>().Where(r => r.Model is not null && r.Model != model).SelectMany(r => r.Model.Materials).Distinct())
        foreach (var texture in textureNames.Select(material.GetTexture).Append(material.FirstTexture).Where(t => t is not null).Distinct())
            texture.MarkUsed(1024);
    foreach (var material in model.Materials)
        foreach (var texture in textureNames.Select(material.GetTexture).Append(material.FirstTexture).Where(t => t is not null).Distinct())
            texture.MarkUsed(2048);
    for (var wait = 0; wait < configuration["stream_frames"].GetValue<int>(); wait++)
    {
        using (scene.Push()) { scene.EditorTick(.016f, .016f); preRender.Invoke(scene, null); }
        await Task.Delay(33, cancellationToken);
    }

    var settle = configuration["settle_frames"].GetValue<int>();
    var written = new System.Collections.Generic.List<string>();
    // The first pass only warms up: the map's textures stream in as each view first sees them, which would leave "before" the blurrier.
    foreach (var variant in new[] { "warm", "before", "after" })
    {
        using (scene.Push())
        {
            pile.Enabled = variant == "after";
            foreach (var slot in slots) props.Materials.SetOverride(slot, variant == "after" ? hiddenMaterial : null);
        }
        foreach (var view in configuration["views"].AsArray())
        {
            cancellationToken.ThrowIfCancellationRequested();
            var name = view["name"].GetValue<string>();
            using var bitmap = new Bitmap(configuration["width"].GetValue<int>(), configuration["height"].GetValue<int>());
            var eye = Vec(view["eye"]);
            var target = Vec(view["target"]);
            for (var frame = 0; frame < settle; frame++)
            {
                using (scene.Push())
                {
                    camera.WorldPosition = eye;
                    camera.WorldRotation = Rotation.LookAt(target - eye);
                    camera.FieldOfView = view["fov"].GetValue<float>();
                    scene.EditorTick(.016f, .016f);
                    preRender.Invoke(scene, null);
                    preCameraRender.Invoke(scene, null);
                    camera.RenderToBitmap(bitmap);
                }
                await Task.Delay(16, cancellationToken);
            }
            if (variant == "warm") continue;
            System.IO.File.WriteAllBytes(System.IO.Path.Combine(destination, $"{name}-{variant}.png"), bitmap.ToPng());
            written.Add($"{name}-{variant}");
        }
    }
    return new
    {
        Before = before, Game.IsPlaying,
        SourceUnchanged = Editor.SceneEditorSession.Active?.Scene?.Id == before.SceneId && Game.ActiveScene?.Id == before.GameSceneId,
        Written = written, Slots = slots, Renderers = scene.GetAllComponents<ModelRenderer>().Count(),
        ErrorModels = scene.GetAllComponents<ModelRenderer>().Where(r => r.Model is null || r.Model.IsError).Select(r => r.GameObject.Name).ToArray(),
        ModelBounds = model.Bounds.Size.ToString(), ModelMaterials = model.Materials.Select(m => m.ResourcePath).ToArray()
    };
}
catch (Exception error)
{
    System.IO.File.WriteAllText(System.IO.Path.Combine(destination, "capture-error.txt"), error.ToString());
    throw;
}
finally { scene.Destroy(); }
