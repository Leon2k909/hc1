var game=Game.ActiveScene?.Id;var ids=session.Scene.GetAllObjects(true).Select(o=>o.Id).ToHashSet();
string Hash(string path)=>Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(System.IO.File.ReadAllBytes(path))).ToLowerInvariant();
foreach(var group in new[]{"source_sha256","protected_sha256"})
 foreach(var pair in manifest[group].AsObject())
  if(Hash(System.IO.Path.Combine(root,pair.Key))!=pair.Value.GetValue<string>())throw new InvalidOperationException("Shared input changed: "+pair.Key);

var plan=new System.Collections.Generic.List<(string Source,string Target,string Backup,string After)>();

{
 var item=pair.Value;var target=System.IO.Path.Combine(root,pair.Key);var source=System.IO.Path.Combine(root,item["source"].GetValue<string>());var backup=System.IO.Path.Combine(root,item["backup"].GetValue<string>());
 var before=item["before"].GetValue<string>();var after=item["after"].GetValue<string>();
 if(Hash(source)!=after||Hash(backup)!=before)throw new InvalidOperationException("Stage or backup changed.");
 var current=Hash(target);if(current==after)continue;
 if(current!=before)throw new InvalidOperationException("Shared target changed: "+pair.Key);
 plan.Add((source,target,backup,after));
}
var changed=new System.Collections.Generic.List<(string Target,string Backup)>();

{

 {
  var incoming=item.Target+".wall-removal-incoming";
  try
  {
   System.IO.File.Copy(item.Source,incoming,true);if(Hash(incoming)!=item.After)throw new InvalidOperationException("Incoming hash differs.");
   changed.Add((item.Target,item.Backup));System.IO.File.Move(incoming,item.Target,true);
  }
  finally{if(System.IO.File.Exists(incoming))System.IO.File.Delete(incoming);}
 }
}
catch
{
 foreach(var item in changed.AsEnumerable().Reverse())System.IO.File.Copy(item.Backup,item.Target,true);
 throw;
}
