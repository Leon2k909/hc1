// Capture an already-open Collider editor document with its persistent editor camera.
// Does not open a scene, start play, render reflection cubemaps or install frame callbacks.
// Copy to the current native MCP exchange directory before run_csharp.
var root = Sandbox.Project.Current.GetRootPath();
var config = System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(
    System.IO.Path.Combine(root,"Sources/Collider/editor/capture-config.json"))).AsObject();
var destination = System.IO.Path.Combine(root,"output/collider/captures");
System.IO.Directory.CreateDirectory(destination);
var previous = Editor.SceneEditorSession.Active;
var gameId = Game.ActiveScene?.Id;
var playing = Game.IsPlaying;
var session = Editor.SceneEditorSession.All.Single(s => s.Scene.IsEditor &&
    s.Scene.Source?.ResourcePath?.Equals("scenes/maps/collider/collider.scene",StringComparison.OrdinalIgnoreCase)==true);
var scene = session.Scene;
var flags = System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.Public | System.Reflection.BindingFlags.NonPublic;
var camera = scene.GetAllComponents<CameraComponent>().Single(c =>
    c.GameObject.Name=="editor_camera" && c.GameObject.Flags.HasFlag(GameObjectFlags.EditorOnly));
var oldPosition = camera.WorldPosition;
var oldRotation = camera.WorldRotation;
var oldFov = camera.FieldOfView;
var oldSize=camera.CustomSize;var oldAxis=camera.FovAxis;var oldOrtho=camera.Orthographic;
var unsaved = session.HasUnsavedChanges;
var models = scene.GetAllComponents<ModelRenderer>().Where(r =>
    r.Model?.ResourcePath?.StartsWith("models/maps/collider/",StringComparison.OrdinalIgnoreCase)==true).ToArray();
if(models.Length!=4 || models.Any(r => r.Model.IsError))
    throw new InvalidOperationException("Expected four successfully compiled Collider models");
// Request useful texture mip levels before changing the camera. Never await with a scene scope open.
using(scene.Push())
{
    foreach(var renderer in models)
    for(var slot=0;slot<renderer.Materials.Count;slot++)
    {
        var material=renderer.Materials.GetOverride(slot) ?? renderer.Materials.GetOriginal(slot);
        if(material is null || material.IsError) throw new InvalidOperationException("Collider has a missing material");
        foreach(var key in new[]{"g_tColor","g_tNormal","g_tRma"}) material.GetTexture(key)?.MarkUsed(2048);
    }
    foreach(var volume in scene.GetAllComponents<IndirectLightVolume>())
    foreach(var texture in new[]{volume.IrradianceTexture,volume.DistanceTexture,volume.RelocationTexture})
        if(texture is not null && !texture.IsError) texture.MarkUsed(Math.Max(texture.Width,texture.Height));
}
await Task.Delay(1000,cancellationToken);
Vector3 World(System.Text.Json.Nodes.JsonNode p) => new(p[1].GetValue<float>(),-p[0].GetValue<float>(),p[2].GetValue<float>());
var captures = new System.Collections.Generic.List<object>();
try
{
    using(scene.Push())
    using(var bitmap = new Bitmap(config["width"].GetValue<int>(),config["height"].GetValue<int>()))
    {
        foreach(var view in config["views"].AsArray())
        {
            cancellationToken.ThrowIfCancellationRequested();
            var name=view["name"].GetValue<string>();
            if(!System.Text.RegularExpressions.Regex.IsMatch(name,"^[a-z0-9-]+$"))
                throw new InvalidOperationException("Invalid capture name");
            camera.WorldPosition=World(view["eye"]);
            camera.WorldRotation=Rotation.LookAt(World(view["target"])-World(view["eye"]));
            camera.FieldOfView=view["fov"].GetValue<float>();
            typeof(Scene).GetMethod("PreRender",flags,null,Type.EmptyTypes,null).Invoke(scene,null);
            typeof(Scene).GetMethod("PreCameraRender",flags,null,Type.EmptyTypes,null).Invoke(scene,null);
            var post=scene.GetSystem<PostProcessSystem>();
            typeof(PostProcessSystem).GetMethod("UpdateCamera",flags).Invoke(post,new object[]{camera});
            // Hidden documents can miss the native DDGI buffer refresh after a bake/hotload.
            var ddgiType=typeof(Scene).Assembly.GetType("Sandbox.DDGIVolumeSystem");
            var getter=typeof(Scene).GetMethods().Single(m=>m.Name=="GetSystem" && m.IsGenericMethodDefinition && m.GetParameters().Length==0);
            var ddgi=getter.MakeGenericMethod(ddgiType).Invoke(scene,null);
            ddgiType.GetMethod("MarkDirty",flags).Invoke(ddgi,null);
            ddgiType.GetMethod("UpdateVolumes",flags).Invoke(ddgi,null);
            if(!camera.AutoExposure.Enabled || MathF.Abs(camera.AutoExposure.MinimumExposure-1f)>.001f ||
                MathF.Abs(camera.AutoExposure.MaximumExposure-2.3f)>.001f)
                throw new InvalidOperationException("Collider's tone mapping did not reach the editor camera");
            for(var frame=0;frame<3;frame++)
            {
                camera.WorldPosition=World(view["eye"]);
                camera.WorldRotation=Rotation.LookAt(World(view["target"])-World(view["eye"]));
                camera.FieldOfView=view["fov"].GetValue<float>();
                camera.CustomSize=new Vector2(config["width"].GetValue<int>(),config["height"].GetValue<int>());
                camera.FovAxis=CameraComponent.Axis.Horizontal;camera.Orthographic=false;
                camera.RenderToBitmap(bitmap,false);
            }
            System.IO.File.WriteAllBytes(System.IO.Path.Combine(destination,name+".png"),bitmap.ToPng());
            captures.Add(new { Name=name, Position=camera.WorldPosition.ToString(),
                Angles=camera.WorldRotation.Angles().ToString(), camera.FieldOfView,
                camera.EnablePostProcessing, camera.AutoExposure.Enabled,
                camera.AutoExposure.MinimumExposure, camera.AutoExposure.MaximumExposure });
        }
    }
}
finally
{
    if(camera.IsValid())
    {
        camera.WorldPosition=oldPosition;
        camera.WorldRotation=oldRotation;
        camera.FieldOfView=oldFov;
        camera.CustomSize=oldSize;camera.FovAxis=oldAxis;camera.Orthographic=oldOrtho;
    }
}
var result = new { Captures=captures, SourceScene=scene.Source?.ResourcePath,
    CheckedAt=DateTime.UtcNow, SavedSceneCurrent=!unsaved,
    SavedSceneSha256=Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(System.IO.File.ReadAllBytes(
        System.IO.Path.Combine(root,"Assets/scenes/maps/Collider/collider.scene")))).ToLowerInvariant(),
    NativeEngineCapture=true, SceneDirtyStatePreserved=session.HasUnsavedChanges==unsaved,
    GamePreserved=Game.ActiveScene?.Id==gameId && Game.IsPlaying==playing,
    SelectedDocumentPreserved=ReferenceEquals(previous,Editor.SceneEditorSession.Active) };
System.IO.File.WriteAllText(System.IO.Path.Combine(destination,"capture-report.json"),System.Text.Json.JsonSerializer.Serialize(result));
return result;
