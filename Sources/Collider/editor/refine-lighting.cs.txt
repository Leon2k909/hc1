if(Game.IsPlaying) throw new InvalidOperationException("Stop play before adjusting Collider lighting");
var session=Editor.SceneEditorSession.All.Single(s=>s.Scene.Source?.ResourcePath?.Equals("scenes/maps/collider/collider.scene",StringComparison.OrdinalIgnoreCase)==true);
var lights=session.Scene.GetAllComponents<PointLight>().ToArray();
foreach(var light in lights)
{
    var fog=typeof(PointLight).GetProperty("FogMode");
    fog.SetValue(light,Enum.Parse(fog.PropertyType,"Disabled"));
    light.FogStrength=0;
    if(light.GameObject.Name.StartsWith("blue chamber ")) light.LightColor=new Color(2.5f,4,7);
    if(light.GameObject.Name.StartsWith("machine overhead ") || light.GameObject.Name.StartsWith("high "))
        light.Contribution=Light.LightContribution.Diffuse | Light.LightContribution.Transmissive;
}
typeof(Editor.SceneEditorSession).GetProperty("HasUnsavedChanges").SetValue(session,true);
return new {Lights=lights.Length,ShadowLights=lights.Count(l=>l.Shadows)};
