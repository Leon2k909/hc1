// Recover a completed bake after its editor document was closed before saving.
if(Game.IsPlaying) throw new InvalidOperationException("Stop play before adopting Collider lighting");
var root=Sandbox.Project.Current.GetRootPath();
var config=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(root,"output/collider/bake-config.json")));
var revision=config["directory"].GetValue<string>();
if(!System.Text.RegularExpressions.Regex.IsMatch(revision,"^lighting-v[0-9]+$")) throw new InvalidOperationException("Invalid bake revision");
var directory=System.IO.Path.Combine(root,"output/collider",revision);
var result=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(directory,"bake-result.json")));
if(!result["GamePreserved"].GetValue<bool>() || !result["InputAssetsUnchanged"].GetValue<bool>() || !result["SavedFinalIrradiance"].GetValue<bool>())
    throw new InvalidOperationException("Bake did not complete in a stable context");
var inputs=System.Text.Json.JsonSerializer.Deserialize<System.Collections.Generic.Dictionary<string,string>>(
    System.IO.File.ReadAllText(System.IO.Path.Combine(directory,"input-assets.json")));
foreach(var input in inputs)
{
    var hash=Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(System.IO.File.ReadAllBytes(System.IO.Path.Combine(root,input.Key)))).ToLowerInvariant();
    if(hash!=input.Value) throw new InvalidOperationException("Bake input changed: "+input.Key);
}
var session=Editor.SceneEditorSession.All.Single(s=>s.Scene.IsEditor && s.Scene.Source?.ResourcePath?.Equals("scenes/maps/collider/collider.scene",StringComparison.OrdinalIgnoreCase)==true);
if(session.HasUnsavedChanges) throw new InvalidOperationException("Collider has unsaved changes");
var textures=new[]{"irradiance","distance","relocation"}.Select(name=>Texture.Load("lighting/collider/"+revision+"/"+name+".vtex",false)).ToArray();
if(textures.Any(t=>t is null || t.IsError)) throw new InvalidOperationException("Completed lighting texture is unavailable");
var indirect=session.Scene.GetAllComponents<IndirectLightVolume>().Single();
indirect.IrradianceTexture=textures[0];indirect.DistanceTexture=textures[1];indirect.RelocationTexture=textures[2];indirect.BakeVersion=1;
typeof(Editor.SceneEditorSession).GetProperty("HasUnsavedChanges").SetValue(session,true);
session.Save(false);
if(session.HasUnsavedChanges) throw new InvalidOperationException("Collider lighting was not saved");
var saved=new{Revision=revision,SavedScene=true,CheckedAt=DateTime.UtcNow,InputAssetsUnchanged=true,Textures=textures.Select(t=>t.ResourcePath)};
System.IO.File.WriteAllText(System.IO.Path.Combine(directory,"adoption-result.json"),System.Text.Json.JsonSerializer.Serialize(saved));
return saved;
