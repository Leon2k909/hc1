// Import only the checked services-v1 model changes. Run after editor_status.
// No scene replacement, no play-mode changes, no automatic undo; originals are backed up.
if(Game.IsPlaying) throw new InvalidOperationException("Stop play before importing Collider geometry");
var root=Sandbox.Project.Current.GetRootPath();
var output=System.IO.Path.Combine(root,"output/collider/services-v1");
var manifest=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(output,"manifest.json")));
if(manifest["revision"]?.GetValue<string>()!="services-v1" ||
    manifest["collision_identical"]?.GetValue<bool>()!=true ||
    manifest["physics_definitions_identical"]?.GetValue<bool>()!=true ||
    manifest["offline_positions_passed"]?.GetValue<int>()!=68)
    throw new InvalidOperationException("The services candidate has not passed its offline checks");
var selected=Editor.SceneEditorSession.Active;
var game=Game.ActiveScene?.Id;
var sceneFlags=Editor.SceneEditorSession.All.ToDictionary(s=>s.Scene.Id,s=>s.HasUnsavedChanges);
string Hash(string p)=>Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(System.IO.File.ReadAllBytes(p))).ToLowerInvariant();
var stage=System.IO.Path.GetFullPath(System.IO.Path.Combine(root,manifest["stage"].GetValue<string>()));
if(!stage.StartsWith(System.IO.Path.GetFullPath(output)+System.IO.Path.DirectorySeparatorChar,StringComparison.OrdinalIgnoreCase))
    throw new InvalidOperationException("Stage is outside the services candidate directory");
var jobs=new System.Collections.Generic.List<(string Path,string Source,string Target,string Backup,string Before,string After)>();
foreach(var item in manifest["files"].AsArray())
{
    var path=item["path"].GetValue<string>();
    if(!System.Text.RegularExpressions.Regex.IsMatch(path,@"^models/maps/collider/collider_[a-z_]+\.(obj|mtl|vmdl)$"))
        throw new InvalidOperationException("Unexpected model path: "+path);
    var source=System.IO.Path.Combine(stage,path);
    var target=System.IO.Path.Combine(root,"Assets",path);
    var before=item["before_sha256"].GetValue<string>();
    var after=item["sha256"].GetValue<string>();
    if(Hash(source)!=after) throw new InvalidOperationException("Staged asset changed: "+path);
    var actual=Hash(target);
    if(actual!=before && actual!=after) throw new InvalidOperationException("Imported asset changed independently: "+path);
    var backup=System.IO.Path.Combine(output,"before/Assets",path);
    if(System.IO.File.Exists(backup) && Hash(backup)!=before)
        throw new InvalidOperationException("Backup does not match the baseline: "+path);
    jobs.Add((path,source,target,backup,before,after));
}
var imported=new System.Collections.Generic.List<string>();
var alreadyImported=new System.Collections.Generic.List<string>();
foreach(var job in jobs.OrderBy(j=>j.Path.EndsWith(".vmdl")?1:0))
{
    cancellationToken.ThrowIfCancellationRequested();
    if(Hash(job.Target)==job.After) { alreadyImported.Add(job.Path); continue; }
    if(!System.IO.File.Exists(job.Backup))
    {
        System.IO.Directory.CreateDirectory(System.IO.Path.GetDirectoryName(job.Backup));
        System.IO.File.Copy(job.Target,job.Backup);
    }
    var temporary=job.Target+".collider-services-import";
    System.IO.File.Copy(job.Source,temporary,true);
    System.IO.File.Move(temporary,job.Target,true);
    Editor.AssetSystem.RegisterFile(job.Target);
    if(Hash(job.Target)!=job.After) throw new InvalidOperationException("Import did not persist: "+job.Path);
    imported.Add(job.Path);
}
var report=new {Imported=imported,AlreadyImported=alreadyImported,
    RequiredModelCompile=new[]{"models/maps/collider/collider_structure.vmdl","models/maps/collider/collider_detail.vmdl","models/maps/collider/collider_glass.vmdl"},
    SelectedDocumentPreserved=ReferenceEquals(selected,Editor.SceneEditorSession.Active),
    GamePreserved=!Game.IsPlaying && Game.ActiveScene?.Id==game,
    SceneDirtyStatesPreserved=Editor.SceneEditorSession.All.All(s=>sceneFlags.TryGetValue(s.Scene.Id,out var dirty) && dirty==s.HasUnsavedChanges),
    NativeVisualReviewPassed=false,LightingReviewPending=true};
System.IO.File.WriteAllText(System.IO.Path.Combine(output,"native-import.json"),System.Text.Json.JsonSerializer.Serialize(report));
return report;
