var scene=Game.ActiveScene;
if(!Game.IsPlaying || scene?.Name!="collider")throw new InvalidOperationException("Collider must be running");
var proposed=scene.GetAllObjects(true).Single(o=>o.Name=="REVIEW ONLY - proposed cabinet");
var original=scene.GetAllObjects(true).Single(o=>o.Name=="REVIEW ONLY - original case");
Vector3 World(GameObject go,Vector3 plan)=>go.WorldTransform.PointToWorld(new Vector3(plan.y,-plan.x,plan.z));
var checks=new System.Collections.Generic.List<object>();var failures=new System.Collections.Generic.List<string>();
void Probe(string name,GameObject go,string tag,Vector3 a,Vector3 b,bool expected)
{
    var trace=scene.Trace.Ray(World(go,a),World(go,b)).WithTag(tag).Run();
    var hit=trace.Hit||trace.StartedSolid;var passed=hit==expected;
    checks.Add(new {Name=name,ExpectedHit=expected,Hit=hit,Passed=passed,HitObject=trace.GameObject?.Name,End=trace.EndPosition.ToString()});
    if(!passed)failures.Add(name);
}
