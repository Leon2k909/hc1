// Import the four owned gallery-detail assets; never rewrite a scene or other models.
var root=Sandbox.Project.Current.GetRootPath();
if(root.TrimEnd('\\','/').Replace('\\','/').ToLowerInvariant()!="c:/sbox-projects/hc1/forked-shooter-main")
    throw new InvalidOperationException("Wrong project");
var output=System.IO.Path.Combine(root,"output/collider-gallery-drain-20261027");
var stage=System.IO.Path.Combine(output,"stage-v1");
var manifest=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(stage,"manifest.json")));
var selected=Editor.SceneEditorSession.Active;
var selectedDirty=selected?.HasUnsavedChanges;
var session=Editor.SceneEditorSession.All.SingleOrDefault(s=>s.Scene.Source?.ResourcePath?.Equals("scenes/maps/collider/collider.scene",StringComparison.OrdinalIgnoreCase)==true);
if(Game.IsPlaying || session is null)throw new InvalidOperationException("Expected a stopped, existing Collider document");
var dirty=session.HasUnsavedChanges;
var game=Game.ActiveScene?.Id;
var sceneId=session.Scene.Id;
string Hash(string p)=>System.IO.File.Exists(p)?Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(System.IO.File.ReadAllBytes(p))).ToLowerInvariant():null;
var sourceScene=System.IO.Path.Combine(root,"Assets/scenes/maps/Collider/collider.scene");
var sceneHash=Hash(sourceScene);
var allowed=new[]{"models/maps/collider/collider_detail.obj","models/maps/collider/collider_detail.mtl",
    "models/maps/collider/collider_detail.vmdl","materials/collider/co_drain_frame.vmat"};
var entries=manifest["files"].AsArray();
if(entries.Count!=allowed.Length)throw new InvalidOperationException("Unexpected file count");
foreach(var entry in entries)
{
    var relative=entry["relative"].GetValue<string>();
    if(!allowed.Contains(relative) || Hash(System.IO.Path.Combine(stage,"assets",relative))!=entry["after"].GetValue<string>() ||
       Hash(System.IO.Path.Combine(root,"Assets",relative))!=entry["before"]?.GetValue<string>())
        throw new InvalidOperationException("Changed or unexpected target: "+relative);
}
foreach(var pair in manifest["protected"].AsObject())
    if(Hash(System.IO.Path.Combine(root,"Assets",pair.Key))!=pair.Value.GetValue<string>())throw new InvalidOperationException("Protected source changed: "+pair.Key);
foreach(var pair in manifest["sources"].AsObject())
    if(Hash(System.IO.Path.Combine(root,pair.Key))!=pair.Value.GetValue<string>())throw new InvalidOperationException("Generator changed: "+pair.Key);
var doc="models/maps/collider/collider_detail.vmdl";
var beforeDoc=System.IO.File.ReadAllText(System.IO.Path.Combine(root,"Assets",doc));
var afterDoc=System.IO.File.ReadAllText(System.IO.Path.Combine(stage,"assets",doc));
var marker="\t\t\t{\n\t\t\t\t_class = \"PhysicsShapeList\"";
if(beforeDoc[beforeDoc.IndexOf(marker,StringComparison.Ordinal)..]!=afterDoc[afterDoc.IndexOf(marker,StringComparison.Ordinal)..])
    throw new InvalidOperationException("Physics source changed");
var completed=new System.Collections.Generic.List<string>();
try
{
    foreach(var entry in entries)
    {
        var relative=entry["relative"].GetValue<string>();
        var target=System.IO.Path.Combine(root,"Assets",relative);
        var backup=System.IO.Path.Combine(stage,"native-before",relative);
        if(System.IO.File.Exists(target))
        {
            System.IO.Directory.CreateDirectory(System.IO.Path.GetDirectoryName(backup));
            System.IO.File.Copy(target,backup,false);
        }
        System.IO.Directory.CreateDirectory(System.IO.Path.GetDirectoryName(target));
        System.IO.File.WriteAllBytes(target+".gallery-incoming",System.IO.File.ReadAllBytes(System.IO.Path.Combine(stage,"assets",relative)));
        System.IO.File.Move(target+".gallery-incoming",target,true);
        completed.Add(relative);
    }
    foreach(var relative in completed.OrderBy(p=>p.EndsWith(".vmdl")?2:p.EndsWith(".vmat")?1:0))
        Editor.AssetSystem.RegisterFile(System.IO.Path.Combine(root,"Assets",relative));
    foreach(var entry in entries)
        if(Hash(System.IO.Path.Combine(root,"Assets",entry["relative"].GetValue<string>()))!=entry["after"].GetValue<string>())
            throw new InvalidOperationException("Import did not match the reviewed stage");
}
catch
{
    foreach(var relative in completed)
    {
        var target=System.IO.Path.Combine(root,"Assets",relative);
        var backup=System.IO.Path.Combine(stage,"native-before",relative);
        if(System.IO.File.Exists(backup))System.IO.File.Copy(backup,target,true);
        else System.IO.File.Delete(target);
        Editor.AssetSystem.RegisterFile(target);
    }
    throw;
}
if(Hash(sourceScene)!=sceneHash || session.Scene.Id!=sceneId || session.HasUnsavedChanges!=dirty ||
   !ReferenceEquals(selected,Editor.SceneEditorSession.Active) || selected?.HasUnsavedChanges!=selectedDirty ||
   Game.ActiveScene?.Id!=game || Game.IsPlaying)
    throw new InvalidOperationException("Editor state changed during import");
var result=new{Revision="gallery-drain-v1",Imported=completed,PhysicsPreserved=true,SceneSourcePreserved=true,
    SourceSceneSHA256=sceneHash,SelectedDocumentPreserved=true,SceneDirtyStatePreserved=true,GamePreserved=true,CheckedAt=DateTime.UtcNow};
System.IO.File.WriteAllText(System.IO.Path.Combine(stage,"native-import.json"),Json.Serialize(result));
return result;
