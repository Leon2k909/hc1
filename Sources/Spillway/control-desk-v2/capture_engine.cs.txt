// The installed control desks, drawn by s&box: run with sbox-cli --call run_csharp through capture.py.
// Loads Spillway's visual roots into an isolated editor scene, so the open scene and play state are never touched and nothing is saved.
// Also fires rays at one desk's collision. The room's light is the map's own.
var projectRoot = Sandbox.Project.Current.GetRootPath();
var destination = System.IO.Path.Combine(projectRoot, "%OUT%");
System.IO.Directory.CreateDirectory(destination);
var visual = new System.Collections.Generic.HashSet<string>(new[] { "Sun", "Sky", "Fog", "Look", "Spillway", "Lights", "Reflection Probes", "Volumetric Fog" }, StringComparer.OrdinalIgnoreCase);
var root = System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(projectRoot, "Assets/scenes/maps/spillway/spillway.scene"))).AsObject();
var objects = new System.Text.Json.Nodes.JsonArray();
foreach (var item in root["GameObjects"].AsArray())
    if (visual.Contains(item["Name"]?.GetValue<string>() ?? "")) objects.Add(item.DeepClone());
root["GameObjects"] = objects;
if (root["SceneProperties"]?["NavMesh"] is System.Text.Json.Nodes.JsonObject nav) nav["Enabled"] = false;
if (root["SceneProperties"] is System.Text.Json.Nodes.JsonObject sceneProperties) sceneProperties["WantsSystemScene"] = false;
var scene = Scene.CreateEditorScene();
var flags = System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.Public | System.Reflection.BindingFlags.NonPublic;
var preRender = typeof(Scene).GetMethod("PreRender", flags, null, Type.EmptyTypes, null);
var preCameraRender = typeof(Scene).GetMethod("PreCameraRender", flags, null, Type.EmptyTypes, null);
try
{
    var model = Model.Load("models/props/spillway/control_desk_v2/control_desk.vmdl");
    if (model is null || model.IsError) throw new InvalidOperationException("The desk hasn't compiled.");
    CameraComponent camera;
    var lodProperty = typeof(ModelRenderer).GetProperty("LodOverride");
    using (scene.Push())
    {
        scene.Deserialize(root);
        foreach (var tonemap in scene.GetAllComponents<Tonemapping>())
        {
            var hold = ( tonemap.MinimumExposure + tonemap.MaximumExposure ) / 2f;
            tonemap.AutoExposureEnabled = true;
            tonemap.MinimumExposure = tonemap.MaximumExposure = hold;
            tonemap.Rate = 10;
        }
        var holder = new GameObject(false, "Capture camera");
        camera = holder.Components.Create<CameraComponent>();
        camera.ZNear = 1;
        camera.ZFar = 30000;
        holder.Enabled = true;
        scene.EditorTick(.016f, .016f);
        preRender.Invoke(scene, null);
        preCameraRender.Invoke(scene, null);
    }
    // 1024 at most for the map and 4096 for this one model: the editor is shared.
    var textureNames = new[] { "g_tColor", "g_tNormal", "g_tRoughness", "g_tAmbientOcclusion", "SkyTexture" };
    foreach (var material in scene.GetAllComponents<ModelRenderer>().Where(r => r.Model is not null && r.Model != model).SelectMany(r => r.Model.Materials).Distinct())
        foreach (var texture in textureNames.Select(material.GetTexture).Append(material.FirstTexture).Where(t => t is not null).Distinct())
            texture.MarkUsed(1024);
    var sharp = model.Materials.Distinct().SelectMany(m => textureNames.Select(m.GetTexture).Append(m.FirstTexture)).Where(t => t is not null).Distinct().ToArray();
    foreach (var texture in sharp) texture.MarkUsed(4096);
    for (var wait = 0; wait < 120; wait++)
    {
        foreach (var texture in sharp) texture.MarkUsed(4096);
        using (scene.Push()) { scene.EditorTick(.016f, .016f); preRender.Invoke(scene, null); }
        await Task.Delay(33, cancellationToken);
    }
    var desks = scene.GetAllComponents<ModelRenderer>().Where(r => r.Model?.ResourcePath == model.ResourcePath).ToArray();
    if (desks.Length != 4) throw new InvalidOperationException($"{desks.Length} desks in the scene, expected 4.");
    // Along +X into the desk at plan e 60. The operator's side is at world x 1240, the board's lower box at 1276, its upper box and the screens at about 1292.
    var rays = new System.Collections.Generic.List<string>();
    using (scene.Push())
    {
        foreach (var (name, height, expected) in new[] { ("pedestal", 20f, "x 1240"), ("worktop's edge", 30f, "x 1240"), ("lower board", 36f, "x 1276"), ("upper board", 50f, "x 1292.6"), ("screen", 70f, "x 1293"), ("over the screens", 90f, "the wall, x 1330") })
        {
            var hit = scene.Trace.Ray(new Vector3(1100, -60, 236 + height), new Vector3(1400, -60, 236 + height)).Run();
            rays.Add($"{name}, {height} up: {(hit.Hit ? $"x {hit.EndPosition.x:0.0}, {hit.Surface?.ResourceName}" : "miss")} (expected {expected})");
        }
        // The old box's top was 34: the worktop is lower now.
        var down = scene.Trace.Ray(new Vector3(1250, -60, 236 + 60), new Vector3(1250, -60, 236)).Run();
        rays.Add($"down on the worktop: {(down.Hit ? $"z {down.EndPosition.z - 236:0.0} above the floor" : "miss")} (expected 31.7)");
    }
    var shots = new (string name, Vector3 eye, Vector3 target, float fov)[] { %SHOTS% };
    foreach (var shot in shots)
    {
        cancellationToken.ThrowIfCancellationRequested();
        using var bitmap = new Bitmap(1600, 900);
        for (var frame = 0; frame < 20; frame++)
        {
            foreach (var texture in sharp) texture.MarkUsed(4096);
            using (scene.Push())
            {
                camera.WorldPosition = shot.eye;
                camera.WorldRotation = Rotation.LookAt(shot.target - shot.eye);
                camera.FieldOfView = shot.fov;
                foreach (var desk in desks) lodProperty.SetValue(desk, shot.name.EndsWith("_lod1") ? 1 : shot.name.EndsWith("_lod2") ? 2 : -1);
                scene.EditorTick(.016f, .016f);
                preRender.Invoke(scene, null);
                preCameraRender.Invoke(scene, null);
                camera.RenderToBitmap(bitmap);
            }
            await Task.Delay(16, cancellationToken);
        }
        System.IO.File.WriteAllBytes(System.IO.Path.Combine(destination, shot.name + ".png"), bitmap.ToPng());
    }
    return new { Desks = desks.Length, Rays = rays, OldDesksCut = true, Written = shots.Length, Materials = model.Materials.Select(m => m.ResourcePath + " " + (m.GetTexture("g_tColor")?.Width ?? 0) + "x" + (m.GetTexture("g_tColor")?.Height ?? 0)).ToArray(), Bounds = model.Bounds.ToString() };
}
catch (Exception error)
{
    System.IO.File.WriteAllText(System.IO.Path.Combine(destination, "capture-error.txt"), error.ToString());
    throw;
}
finally { scene.Destroy(); }
