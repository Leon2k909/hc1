var root=Sandbox.Project.Current.GetRootPath();var folder=System.IO.Path.Combine(root,"output/spillway/service-cabinet-v1");
var plan=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder,"install-plan.json")));
var session=Editor.SceneEditorSession.Active;
var before=new {Editor=session.Scene.Id,Runtime=Game.ActiveScene?.Id,Game.IsPlaying,Dirty=session.HasUnsavedChanges};
if(Game.IsPlaying&&Game.ActiveScene?.Source?.ResourcePath?.Contains("/spillway/",StringComparison.OrdinalIgnoreCase)==true)
 throw new InvalidOperationException("Spillway itself is running; inspect before updating its visual model");
string Hash(string p)=>Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(System.IO.File.ReadAllBytes(p))).ToLowerInvariant();
foreach(var pair in plan["guards"].AsObject())
 if(Hash(System.IO.Path.Combine(root,pair.Key))!=pair.Value.GetValue<string>())throw new InvalidOperationException("Public input changed: "+pair.Key);
var allowed=new[]{"Assets/models/maps/spillway/spillway_cargo.obj","Assets/models/maps/spillway/spillway_cargo.mtl","Assets/models/maps/spillway/spillway_cargo.vmdl","Tools/spillway_map.py","Tools/spillway_service_cabinet.py"};
if(System.IO.File.Exists(System.IO.Path.Combine(root,"Tools/spillway_service_cabinet.py")))throw new InvalidOperationException("Generator already exists; inspect before replacing");
foreach(var pair in plan["files"].AsObject()) {
 if(Hash(System.IO.Path.Combine(folder,"stage",pair.Key))!=pair.Value.GetValue<string>())throw new InvalidOperationException("Stage changed: "+pair.Key);
 if(!allowed.Contains(pair.Key)&&Hash(System.IO.Path.Combine(root,pair.Key))!=pair.Value.GetValue<string>())throw new InvalidOperationException("Compile the reviewed private assets before installing");
}
foreach(var pair in plan["files"].AsObject().Where(p=>allowed.Contains(p.Key))) {
 var target=System.IO.Path.Combine(root,pair.Key);System.IO.File.Copy(System.IO.Path.Combine(folder,"stage",pair.Key),target+".cabinet-writing",true);
 System.IO.File.Move(target+".cabinet-writing",target,true);
 if(pair.Key.StartsWith("Assets/"))Editor.AssetSystem.RegisterFile(target);
}
var result=new {Cabinets=2,SceneChanged=false,CollisionChanged=false,OtherGeometryPreserved=true,
 UserStatePreserved=session.Scene.Id==before.Editor&&Game.ActiveScene?.Id==before.Runtime&&Game.IsPlaying==before.IsPlaying&&session.HasUnsavedChanges==before.Dirty};
System.IO.File.WriteAllText(System.IO.Path.Combine(folder,"installed.json"),System.Text.Json.JsonSerializer.Serialize(result));return result;
