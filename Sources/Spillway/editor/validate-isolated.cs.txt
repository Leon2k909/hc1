var root=Sandbox.Project.Current.GetRootPath();
var before=Editor.SceneEditorSession.Active?.Scene?.Id;
var scene=Scene.CreateEditorScene();
try {
var json=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(root,"Assets/scenes/maps/spillway/spillway.scene"))).AsObject();
var keep=new[]{"Spillway","Player Clips","Catch Floor"};
var roots=new System.Text.Json.Nodes.JsonArray();
foreach(var item in json["GameObjects"].AsArray()) if(keep.Contains(item["Name"]?.GetValue<string>()))roots.Add(item.DeepClone());
json["GameObjects"]=roots;
json["SceneProperties"]["WantsSystemScene"]=false;
json["SceneProperties"]["NavMesh"]["EditorAutoUpdate"]=false;
using(scene.Push()){scene.Deserialize(json);scene.EditorTick(.016f,.016f);}
Task<bool> build;
using(scene.Push())build=scene.NavMesh.Generate(scene.PhysicsWorld);
if(!await build)throw new InvalidOperationException("Navigation generation failed");
var requested=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(root,"output/spillway/check-points.json"))).AsArray();
Vector3 V(System.Text.Json.Nodes.JsonNode a)=>new(a[0].GetValue<float>(),a[1].GetValue<float>(),a[2].GetValue<float>());
var anchor=scene.NavMesh.GetClosestPoint(V(requested[0]["position"]),48);
var checks=requested.Select(node=>{
    var p=V(node["position"]);
    var floor=scene.Trace.Ray(p+Vector3.Up*8,p+Vector3.Down*32).WithoutTags("zone").Run();
    var head=scene.Trace.Box(new BBox(new Vector3(-16,-16,2),new Vector3(16,16,64)),p,p).WithoutTags("zone").Run();
    var snap=scene.NavMesh.GetClosestPoint(p,48);
    var status="MissingNavMesh"; var count=0; var endError=float.MaxValue;
    if(anchor is not null && snap is not null)
    {
        var path=scene.NavMesh.CalculatePath(new Sandbox.Navigation.CalculatePathRequest{Start=anchor.Value,Target=snap.Value});
        status=path.Status.ToString(); count=path.Points?.Count()??0;
        if(count>0) endError=(path.Points.Last().Position-snap.Value).Length;
    }
    return new { Name=node["name"].GetValue<string>(), Position=p.ToString(), Floor=floor.Hit,
        Clear=!head.Hit&&!head.StartedSolid, BlockingObject=head.GameObject?.Name,
        Snapped=snap?.ToString(), SnapDistance=snap is null ? float.MaxValue : (snap.Value-p).Length,
        PathStatus=status, Waypoints=count, EndError=endError };
}).ToArray();
Vector3 World(Vector3 p)=>new(p.y,-p.x,p.z);
var stairPairs=new[] {
    (Name:"Control west stair",Start:new Vector3(-275,145,0),End:new Vector3(-275,720,236)),
    (Name:"Control east stair",Start:new Vector3(560,145,0),End:new Vector3(560,720,236)),
    (Name:"B2 west stair",Start:new Vector3(1130,-1320,0),End:new Vector3(1500,-1320,128)),
    (Name:"B2 north stair",Start:new Vector3(1515,-480,0),End:new Vector3(1515,-870,128))
};
var stairs=stairPairs.Select(pair=>{
    var start=scene.NavMesh.GetClosestPoint(World(pair.Start),48);
    var end=scene.NavMesh.GetClosestPoint(World(pair.End),48);
    if(start is null || end is null)return new {pair.Name,Status="MissingNavMesh",Length=0f,Direct=false};
    var route=scene.NavMesh.CalculatePath(new Sandbox.Navigation.CalculatePathRequest{Start=start.Value,Target=end.Value});
    var points=route.Points.Select(x=>x.Position).ToArray();
    var length=points.Zip(points.Skip(1),(a,b)=>a.Distance(b)).Sum();
    return new {pair.Name,Status=route.Status.ToString(),Length=length,Direct=route.Status.ToString()=="Complete" && length<(pair.End-pair.Start).Length*1.55f};
}).ToArray();
var models=scene.GetAllComponents<ModelRenderer>().Where(r=>r.Model?.ResourcePath?.StartsWith("models/maps/spillway/",StringComparison.OrdinalIgnoreCase)==true).ToArray();
var materials=models.SelectMany(r=>Enumerable.Range(0,r.Materials.Count).Select(i=>r.Materials.GetOverride(i)??r.Materials.GetOriginal(i))).Distinct().ToArray();
var missingModels=models.Where(r=>r.Model is null||r.Model.IsError).Select(r=>r.GameObject.Name).ToArray();
var missingMaterials=materials.Where(m=>m is null||m.IsError||m.ShaderName=="shaders/error.shader").Select(m=>m?.ResourcePath??"null").ToArray();
var result=new { Scene="scenes/maps/spillway/spillway.scene", Checked=checks.Length, CheckedAt=DateTime.UtcNow,
    ModelCount=models.Length,
    SceneSha256=Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(System.IO.File.ReadAllBytes(
        System.IO.Path.Combine(root,"Assets/scenes/maps/Spillway/spillway.scene")))).ToLowerInvariant(),
    MissingModels=missingModels, MissingMaterials=missingMaterials, Stairs=stairs,
    Failed=checks.Where(c=>!c.Floor||!c.Clear||c.PathStatus!="Complete"||c.EndError>4).ToArray(),
    Checks=checks, NavEnabled=scene.NavMesh.IsEnabled, scene.NavMesh.IsGenerating,
    scene.NavMesh.AgentRadius, scene.NavMesh.AgentHeight, scene.NavMesh.AgentStepSize,
    Modes=Facepunch.GameMode.GetAll(Editor.AssetSystem.FindByPath("scenes/maps/spillway/spillway.scene").LoadResource<SceneFile>()).Select(m=>m.Ident).ToArray() };
System.IO.File.WriteAllText(System.IO.Path.Combine(root,"output/spillway/native-validation.json"),System.Text.Json.JsonSerializer.Serialize(result,new System.Text.Json.JsonSerializerOptions{WriteIndented=true}));
return new { result.Scene, result.Checked, result.MissingModels, result.MissingMaterials, result.Failed, result.Stairs, result.NavEnabled, result.IsGenerating, result.Modes };

}finally{scene.Destroy();}
