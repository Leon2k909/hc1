const string relative="scenes/maps/spillway/spillway.scene";
if(Game.IsPlaying && Game.ActiveScene?.Source?.ResourcePath?.Equals(relative,StringComparison.OrdinalIgnoreCase)==true)
    throw new InvalidOperationException("Stop the running Spillway match before updating its scene resource.");
var asset=Editor.AssetSystem.FindByPath(relative) ?? throw new InvalidOperationException("Create the scene with create_asset first.");
if(Editor.SceneEditorSession.All.Any(s=>s.Scene.Source?.ResourcePath?.Equals(relative,StringComparison.OrdinalIgnoreCase)==true))
    throw new InvalidOperationException("Close the Spillway editor tab before initial scene assembly.");
var root=Sandbox.Project.Current.GetRootPath();
var resource=asset.LoadResource<SceneFile>();
resource.LoadFromJson(System.IO.File.ReadAllText(System.IO.Path.Combine(root,"output/spillway/scene-import.json")));
if(!asset.SaveToDisk(resource))throw new InvalidOperationException("Editor could not save Spillway.");
return new {Saved=relative};
