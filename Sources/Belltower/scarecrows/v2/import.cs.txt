var folder=System.IO.Path.Combine(root,"output/belltower/scarecrow-v2");
var request=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder,"import-request.json")));

var session=Editor.SceneEditorSession.Active;var playing=Game.IsPlaying;var game=Game.ActiveScene?.Id;var dirty=session?.HasUnsavedChanges;
var scenePath=System.IO.Path.Combine(root,"Assets/scenes/maps/Belltower/belltower.scene");var sceneHash=Hash(scenePath);

{
 var path=System.IO.Path.Combine(root,pair.Key);var text=System.IO.File.ReadAllText(path);documents.Add(path,Hash(path));
 var mesh=pair.Value["mesh"].GetValue<string>();
 if(!text.Contains("filename = \""+mesh.Substring(7)+"\"",StringComparison.Ordinal))throw new InvalidOperationException("The dedicated scarecrow import changed.");
 foreach(var alias in pair.Value["required_aliases"].AsArray())
  if(!text.Contains(alias.GetValue<string>(),StringComparison.Ordinal))throw new InvalidOperationException("The scarecrow's material binding changed.");
}
foreach(var pair in manifest["source_sha256"].AsObject())
 if(Hash(System.IO.Path.Combine(root,pair.Key))!=pair.Value.GetValue<string>())throw new InvalidOperationException("Scarecrow source changed: "+pair.Key);
foreach(var pair in manifest["required_source_snippets"].AsObject())

{
 var item=pair.Value;var source=System.IO.Path.Combine(root,item["source"].GetValue<string>());
 var target=System.IO.Path.Combine(root,pair.Key);var backup=System.IO.Path.Combine(root,item["backup"].GetValue<string>());

}

{

  try{System.IO.File.Copy(item.Source,incoming,true);if(Hash(incoming)!=item.After)throw new InvalidOperationException("Incoming hash differs.");changed.Add((item.Target,item.Backup));System.IO.File.Move(incoming,item.Target,true);}
  finally{if(System.IO.File.Exists(incoming))System.IO.File.Delete(incoming);}
 }
}
catch{foreach(var item in changed.AsEnumerable().Reverse())System.IO.File.Copy(item.Backup,item.Target,true);throw;}

var docsPreserved=documents.All(p=>Hash(p.Key)==p.Value);
