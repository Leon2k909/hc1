var manifest=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(directory,"manifest.json")));

string Hash(string p) {using var stream=System.IO.File.OpenRead(p);return Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(stream)).ToLowerInvariant();}
foreach(var pair in manifest["guards"].AsObject())
 if(Hash(System.IO.Path.Combine(root,pair.Key))!=pair.Value.GetValue<string>())throw new InvalidOperationException("Shared source changed: "+pair.Key);
if(Hash(System.IO.Path.Combine(root,"Sources/Belltower/bistro-table-v2/build.py"))!=manifest["source_sha256"].GetValue<string>())throw new InvalidOperationException("Builder changed.");
var previous=Editor.SceneEditorSession.Active;
var playing=Game.IsPlaying;var runtime=Game.ActiveScene?.Id;

foreach(var pair in manifest["files"].AsObject())

 if(!pair.Key.StartsWith("Assets/models/maps/belltower/bistro-table-v2/") || pair.Key.Contains(".."))throw new InvalidOperationException("Unexpected target.");
 var source=System.IO.Path.Combine(directory,"stage",pair.Key);var target=System.IO.Path.Combine(root,pair.Key);var hash=pair.Value.GetValue<string>();

 if(System.IO.File.Exists(target) && Hash(target)!=hash)throw new InvalidOperationException("Private target already differs: "+pair.Key);
 plan.Add((source,target,hash));
}
foreach(var item in plan.OrderBy(p=>p.Target.EndsWith(".vmdl") ? 1 : 0))
{
 cancellationToken.ThrowIfCancellationRequested();

 Editor.AssetSystem.RegisterFile(item.Target);
}
