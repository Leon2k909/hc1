var imported=new System.Collections.Generic.List<string>();
var stage=System.IO.Path.Combine(root,"output/belltower/quality/wall-v3");
void Publish(string relative,byte[] bytes)
{
    var target=System.IO.Path.Combine(root,"Assets",relative);
    if(System.IO.File.Exists(target) && System.IO.File.ReadAllBytes(target).SequenceEqual(bytes)) return;
    var backup=System.IO.Path.Combine(stage,"before/Assets",relative);
    if(System.IO.File.Exists(target) && !System.IO.File.Exists(backup))
    {
        System.IO.Directory.CreateDirectory(System.IO.Path.GetDirectoryName(backup));
        System.IO.File.Copy(target,backup);
    }
    System.IO.Directory.CreateDirectory(System.IO.Path.GetDirectoryName(target));
    var incoming=target+".wall-v3-import";
    System.IO.File.WriteAllBytes(incoming,bytes);
    System.IO.File.Move(incoming,target,true);
    Editor.AssetSystem.RegisterFile(target);
    imported.Add(relative);
}
