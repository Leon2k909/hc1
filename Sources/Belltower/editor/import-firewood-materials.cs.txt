// Import original ImageGen firewood albedos and derived PBR maps as new assets.
// This never overwrites accepted materials or changes a scene or game object.
var root=Sandbox.Project.Current.GetRootPath();
var stage=System.IO.Path.Combine(root,"output/belltower/quality/firewood-production-v1");
var manifestPath=System.IO.Path.Combine(stage,"manifest.json");
var manifest=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(manifestPath));
if(manifest["revision"].GetValue<string>()!="firewood-production-v1")
    throw new InvalidOperationException("Unexpected firewood revision");
string Hash(string p)
{
    using var s=System.IO.File.OpenRead(p);
    return Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(s)).ToLowerInvariant();
}
foreach(var source in manifest["sources"].AsObject())
    if(Hash(System.IO.Path.Combine(root,source.Key))!=source.Value.GetValue<string>())
        throw new InvalidOperationException("Original ImageGen source differs: "+source.Key);
foreach(var file in new[]{"geometry_generator","staging_generator"})
    if(Hash(System.IO.Path.Combine(root,manifest[file].GetValue<string>()))!=manifest[file+"_sha256"].GetValue<string>())
        throw new InvalidOperationException("Firewood generator changed during stage");
var allowed=new System.Collections.Generic.HashSet<string>(StringComparer.Ordinal);
foreach(var kind in new[]{"bark","cuts","split"})
    foreach(var suffix in new[]{"_color.png","_normal.png","_rough.png",".vmat"}) allowed.Add(kind+suffix);
var files=manifest["files"].AsArray();
if(files.Count!=12) throw new InvalidOperationException("Expected exactly 12 firewood trial resources");
var seen=new System.Collections.Generic.HashSet<string>();
foreach(var file in files)
{
    var relative=file["path"].GetValue<string>();
    var name=System.IO.Path.GetFileName(relative);
    var expectedSource="output/belltower/quality/firewood-production-v1/"+name;
    if(!allowed.Contains(name) || !seen.Add(name)
        || relative!="materials/belltower/firewood-v2/"+name
        || file["source"].GetValue<string>()!=expectedSource)
        throw new InvalidOperationException("Unexpected firewood resource: "+relative);
    var source=System.IO.Path.Combine(root,expectedSource);
    var destination=System.IO.Path.Combine(root,"Assets",relative);
    if(Hash(source)!=file["sha256"].GetValue<string>()
        || System.IO.File.Exists(destination) && Hash(destination)!=file["sha256"].GetValue<string>())
        throw new InvalidOperationException("Firewood resource changed or conflicts: "+name);
}
var game=Game.ActiveScene?.Id;
var playing=Game.IsPlaying;
var selection=Editor.SceneEditorSession.Active;
var imported=0;
foreach(var file in files)
{
    cancellationToken.ThrowIfCancellationRequested();
    var relative=file["path"].GetValue<string>();
    var source=System.IO.Path.Combine(root,file["source"].GetValue<string>());
    var target=System.IO.Path.Combine(root,"Assets",relative);
    if(System.IO.File.Exists(target)) continue;
    System.IO.Directory.CreateDirectory(System.IO.Path.GetDirectoryName(target));
    System.IO.File.Copy(source,target);
    Editor.AssetSystem.RegisterFile(target);
    if(Hash(target)!=file["sha256"].GetValue<string>())
        throw new InvalidOperationException("Installed firewood resource differs: "+relative);
    imported++;
}
var report=new {Revision="firewood-production-v1",Imported=imported,
    OriginalAssetsReplaced=false,
    GamePreserved=Game.ActiveScene?.Id==game && Game.IsPlaying==playing,
    SelectionPreserved=ReferenceEquals(selection,Editor.SceneEditorSession.Active)};
System.IO.File.WriteAllText(System.IO.Path.Combine(stage,"native-import-result.json"),System.Text.Json.JsonSerializer.Serialize(report));
return report;
