var session=Editor.SceneEditorSession.Active;
if (Game.IsPlaying || session is null || session.HasUnsavedChanges ||
    !session.Scene.Source.ResourcePath.Equals("scenes/maps/belltower/belltower.scene",StringComparison.OrdinalIgnoreCase))
    throw new InvalidOperationException("Requires active, stopped, saved Belltower.");
var root=Sandbox.Project.Current.GetRootPath();
var directory=System.IO.Path.Combine(root,"output/belltower/long-bed");
var plan=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(directory,"plant-plan.json")));
var hash=Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(System.IO.File.ReadAllBytes(System.IO.Path.Combine(root,"Assets/scenes/maps/Belltower/belltower.scene")))).ToLowerInvariant();
if(hash!=plan["before_scene_sha256"].GetValue<string>()) throw new InvalidOperationException("Shared scene changed since planning.");
var objects=session.Scene.GetAllObjects(true).ToDictionary(o=>o.Id);
foreach(var row in plan["objects"].AsArray())
{
    var obj=objects[Guid.Parse(row["id"].GetValue<string>())];
    if(Vector3.DistanceBetween(obj.WorldPosition,Vector3.Parse(row["old_position"].GetValue<string>()))>.001f ||
        Vector3.DistanceBetween(obj.WorldScale,Vector3.Parse(row["old_scale"].GetValue<string>()))>.00001f)
        throw new InvalidOperationException("A target plant changed.");
}
System.IO.File.WriteAllText(System.IO.Path.Combine(directory,"native-object-ids-before.json"),System.Text.Json.JsonSerializer.Serialize(objects.Keys.OrderBy(x=>x).ToArray()));
return new {Verified=32,Saved=true,LiveObjectCount=objects.Count};
