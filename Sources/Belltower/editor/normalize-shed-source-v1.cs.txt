// Publish the guarded, shed-only OBJ edits through the existing editor.
var root = Sandbox.Project.Current.GetRootPath();
if (!System.IO.Path.GetFullPath(root).TrimEnd('\\').Equals(@"C:\sbox-projects\hc1\forked-shooter-main", StringComparison.OrdinalIgnoreCase))
    throw new InvalidOperationException("Wrong project.");
var stage = System.IO.Path.Combine(root, "output/belltower/shed-corners-v1/r6/stage");
var manifest = System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(stage,"manifest.json")));
string Hash(string path) { using var stream = System.IO.File.OpenRead(path); return Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(stream)).ToLowerInvariant(); }
foreach (var pair in manifest["source_sha256"].AsObject())
        if (Hash(System.IO.Path.Combine(root,pair.Key)) != pair.Value.GetValue<string>())
            throw new InvalidOperationException("Shared input changed: "+pair.Key);
// Other work may add unrelated render nodes. We never replace a model definition;
// require that each edited timber import is still referenced by the current one.
var untouchedDefinitions = manifest["input_guards"].AsObject().ToDictionary(p=>p.Key,p=>Hash(System.IO.Path.Combine(root,p.Key)));
foreach (var side in new[] { "west", "east" })
{
    var definition = System.IO.File.ReadAllText(System.IO.Path.Combine(root,"Assets/models/maps/belltower/belltower_"+side+".vmdl"));
    foreach (var pair in manifest["models"][side]["imports"].AsArray())
        if (!definition.Contains("filename = \""+pair["asset"].GetValue<string>().Substring("Assets/".Length)+"\""))
            throw new InvalidOperationException("A staged timber import is no longer used.");
}
var selected = Editor.SceneEditorSession.Active;
var unsaved = selected?.HasUnsavedChanges;
var playing = Game.IsPlaying; var gameId = Game.ActiveScene?.Id;
var scenePath = System.IO.Path.Combine(root,"Assets/scenes/maps/Belltower/belltower.scene");
var sceneHash = Hash(scenePath);
var plan = new System.Collections.Generic.List<(string Source,string Target,string Backup,string Before,string After)>();
foreach (var pair in manifest["files"].AsObject())
{
    if (!System.Text.RegularExpressions.Regex.IsMatch(pair.Key,@"^Assets/models/maps/belltower/(belltower_(west|east)\.obj|compiler-safe/(west|east)/(west|east)_\d+_furniture_wood(_dark)?\.obj)$"))
        throw new InvalidOperationException("Unexpected asset: "+pair.Key);
    var source = System.IO.Path.Combine(stage,pair.Key); var target = System.IO.Path.Combine(root,pair.Key);
    var backup = System.IO.Path.Combine(stage,"before",pair.Key);
    var before = pair.Value["before"].GetValue<string>(); var after = pair.Value["after"].GetValue<string>();
    if (Hash(source)!=after || Hash(backup)!=before) throw new InvalidOperationException("Staging evidence changed.");
    var current = Hash(target);
    if (current==after) continue;
    if (current!=before) throw new InvalidOperationException("Shared asset changed: "+pair.Key);
    plan.Add((source,target,backup,before,after));
}
if (manifest["files"].AsObject().Count!=2) throw new InvalidOperationException("Unexpected asset count.");
var changed = new System.Collections.Generic.List<(string Target,string Backup,string Before)>();
try
{
    foreach (var item in plan)
    {
        var incoming = item.Target+".shed-corners-incoming";
        try
        {
            System.IO.File.Copy(item.Source,incoming,true);
            if (Hash(incoming)!=item.After || Hash(item.Target)!=item.Before)
                throw new InvalidOperationException("Asset changed during import.");
            changed.Add((item.Target,item.Backup,item.Before));
            try { System.IO.File.Move(incoming,item.Target,true); }
            catch (UnauthorizedAccessException)
            {
                // The editor may retain a read handle without FILE_SHARE_DELETE.
                // A normal source-file save can still be available in that case.
                if (Hash(item.Target)!=item.Before) throw;
                System.IO.File.Copy(incoming,item.Target,true);
            }
            if (Hash(item.Target)!=item.After) throw new InvalidOperationException("Published asset differs.");
        }
        finally { if (System.IO.File.Exists(incoming)) System.IO.File.Delete(incoming); }
    }
}
catch
{
    foreach (var item in changed.AsEnumerable().Reverse())
        if (Hash(item.Target)!=item.Before) System.IO.File.Copy(item.Backup,item.Target,true);
    throw;
}
foreach (var item in plan) Editor.AssetSystem.RegisterFile(item.Target);
if (!ReferenceEquals(selected,Editor.SceneEditorSession.Active) || selected?.HasUnsavedChanges!=unsaved || Game.IsPlaying!=playing || Game.ActiveScene?.Id!=gameId || Hash(scenePath)!=sceneHash || untouchedDefinitions.Any(p=>Hash(System.IO.Path.Combine(root,p.Key))!=p.Value))
    throw new InvalidOperationException("Editor state changed during asset publication.");
var result = new { Imported=plan.Count, Sheds=6, SceneSha256=sceneHash, SceneUnchanged=true, EditorStatePreserved=true, PhysicsUnchanged=true, ModelDefinitionsUnchanged=true };
System.IO.File.WriteAllText(System.IO.Path.Combine(stage,"native-import.json"),System.Text.Json.JsonSerializer.Serialize(result));
return result;
