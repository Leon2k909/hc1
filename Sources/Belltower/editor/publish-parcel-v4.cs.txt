// Guarded all-or-nothing parcel-v4 publication. Edits Belltower scene source
// only when no Belltower editor tab or match is active, and keeps backups.
var root=Sandbox.Project.Current.GetRootPath();
var stage=System.IO.Path.Combine(root,"output/belltower/quality/parcel-v4-production");
var manifest=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(stage,"publish-manifest.json")));
if(manifest["revision"].GetValue<string>()!="parcel-v4") throw new InvalidOperationException("Wrong parcel revision");
if(Game.IsPlaying && Game.ActiveScene?.Source?.ResourcePath?.Equals("scenes/maps/belltower/belltower.scene",StringComparison.OrdinalIgnoreCase)==true)
    throw new InvalidOperationException("Cannot replace live Belltower match");
if(Editor.SceneEditorSession.All.Any(s=>s.Scene.IsEditor && s.Scene.Source?.ResourcePath?.Equals("scenes/maps/belltower/belltower.scene",StringComparison.OrdinalIgnoreCase)==true))
    throw new InvalidOperationException("Close the Belltower document before updating the source scene");
string Hash(string path) { using var stream=System.IO.File.OpenRead(path);return Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(stream)).ToLowerInvariant(); }
var staged=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(stage,"native-trial-manifest.json")));
if(Hash(System.IO.Path.Combine(stage,"native-trial-manifest.json"))!=manifest["material_manifest_sha256"].GetValue<string>()) throw new InvalidOperationException("Texture manifest changed");
foreach(var m in staged["source_sha256"].AsObject())
    if(Hash(System.IO.Path.Combine(root,m.Key))!=m.Value.GetValue<string>()) throw new InvalidOperationException("Original input changed: "+m.Key);
var models=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(root,"output/belltower/quality/parcel-v4-model-compile.json"))).AsObject();
if(models.Count!=88 || models.Any(x=>x.Value["Success"]?.GetValue<bool>()!=true)) throw new InvalidOperationException("88 forest models have not compiled");
foreach(var item in models)
{
    var path=System.IO.Path.Combine(root,"Assets",item.Key);
    if(!System.IO.File.Exists(path+"_c")) throw new InvalidOperationException("Missing compiled forest sector: "+item.Key);
}
var uploads=manifest["assets"].AsArray();
if(uploads.Count!=18 || manifest["near_trees"].GetValue<int>()!=1600 || manifest["forest_sectors"].GetValue<int>()!=88)
    throw new InvalidOperationException("Incomplete parcel batch");
var checkedFiles=new System.Collections.Generic.List<(string Source,string Target,string Backup,string Sha)>();
var backupFolder=System.IO.Path.Combine(stage,"before-publication");
foreach(var item in uploads)
{
    var relative=item["target"].GetValue<string>();
    if(relative.Contains("..") || !(relative.StartsWith("Assets/materials/belltower/vista/",StringComparison.Ordinal)
        || relative=="Assets/scenes/maps/Belltower/belltower.scene" || relative=="Sources/Belltower/landscape/trees.json" || relative=="Sources/Belltower/landscape/forest.json"))
        throw new InvalidOperationException("Unexpected output path: "+relative);
    var source=System.IO.Path.Combine(root,item["source"].GetValue<string>());
    var target=System.IO.Path.Combine(root,relative);
    var backup=System.IO.Path.Combine(backupFolder,relative);
    if(Hash(source)!=item["source_sha256"].GetValue<string>()
        || Hash(target)!=item["expected_sha256"].GetValue<string>())
        throw new InvalidOperationException("Source or accepted target changed: "+relative);
    if(System.IO.File.Exists(backup) && Hash(backup)!=item["expected_sha256"].GetValue<string>())
        throw new InvalidOperationException("Existing backup conflict: "+relative);
    checkedFiles.Add((source,target,backup,item["source_sha256"].GetValue<string>()));
}
var game=Game.ActiveScene?.Id;
var playing=Game.IsPlaying;
var selection=Editor.SceneEditorSession.Active;
foreach(var file in checkedFiles)
{
    System.IO.Directory.CreateDirectory(System.IO.Path.GetDirectoryName(file.Backup));
    if(!System.IO.File.Exists(file.Backup)) System.IO.File.Copy(file.Target,file.Backup);
}
var changed=new System.Collections.Generic.List<(string Source,string Target,string Backup,string Sha)>();
try
{
    foreach(var file in checkedFiles)
    {
        cancellationToken.ThrowIfCancellationRequested();
        var incoming=file.Target+".parcel-v4-incoming";
        System.IO.File.Copy(file.Source,incoming,true);
        if(Hash(incoming)!=file.Sha) throw new InvalidOperationException("Incoming texture/scene hash differs");
        System.IO.File.Move(incoming,file.Target,true);
        changed.Add(file);
        if(file.Target.EndsWith(".jpg") || file.Target.EndsWith(".png") || file.Target.EndsWith(".vmat"))
            Editor.AssetSystem.RegisterFile(file.Target);
    }
}
catch
{
    foreach(var file in changed.AsEnumerable().Reverse()) System.IO.File.Copy(file.Backup,file.Target,true);
    throw;
}
var report=new {Updated=changed.Count,BackupDirectory=backupFolder,NearTrees=1600,ForestSectors=88,
    SceneOnDisk=true,GamePreserved=Game.ActiveScene?.Id==game && Game.IsPlaying==playing,
    SelectedDocumentPreserved=ReferenceEquals(selection,Editor.SceneEditorSession.Active)};
System.IO.File.WriteAllText(System.IO.Path.Combine(stage,"native-publish-result.json"),System.Text.Json.JsonSerializer.Serialize(report));
return report;
