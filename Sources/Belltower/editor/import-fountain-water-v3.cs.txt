// Publish only the seven guarded fountain-water resources. Never select a scene.
var root=Sandbox.Project.Current.GetRootPath();
var folder=System.IO.Path.Combine(root,"output/belltower/fountain-water-v3");
var request=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder,"import-request.json")));
var revision=request["revision"].GetValue<string>(); var phase=request["phase"].GetValue<string>();
if(!System.Text.RegularExpressions.Regex.IsMatch(revision,"^[a-z0-9-]+$") || (phase!="shaders" && phase!="publish"))
    throw new InvalidOperationException("Invalid water import request.");
var manifest=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder,revision,"stage/manifest.json")));
if(manifest["revision"].GetValue<string>()!="fountain-water-v3") throw new InvalidOperationException("Wrong staged water revision.");
var allowed=new System.Collections.Generic.HashSet<string>{
    "Assets/shaders/belltower_fountain_surface_v3.shader","Assets/shaders/belltower_fountain_stream_v3.shader",
    "Assets/materials/belltower/fountain_water.vmat","Assets/materials/belltower/fountain_jet.vmat",
    "Assets/models/maps/belltower/belltower_centre.obj",
    "Assets/models/maps/belltower/fountain-ornaments-v2/centre/centre_034_fountain_jet.obj",
    "Assets/models/maps/belltower/fountain-ornaments-v2/centre/centre_035_fountain_water.obj"};
var selected=Editor.SceneEditorSession.Active; var playing=Game.IsPlaying; var game=Game.ActiveScene?.Id;
var dirty=selected?.HasUnsavedChanges;
if(phase=="publish" && playing) throw new InvalidOperationException("Inspect the current play session before changing fountain resources.");
string Hash(string path)=>Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(System.IO.File.ReadAllBytes(path))).ToLowerInvariant();
foreach(var group in new[]{"source_sha256","protected_sha256"})
    foreach(var pair in manifest[group].AsObject())
        if(Hash(System.IO.Path.Combine(root,pair.Key))!=pair.Value.GetValue<string>())
            throw new InvalidOperationException("A shared input changed: "+pair.Key);
var plans=new System.Collections.Generic.List<(string Source,string Target,string Backup,string After)>();
foreach(var pair in manifest["files"].AsObject())
{
    if(!allowed.Contains(pair.Key)) throw new InvalidOperationException("Unexpected water target.");
    if(phase=="shaders" && !pair.Key.EndsWith(".shader",StringComparison.Ordinal)) continue;
    var source=System.IO.Path.Combine(root,pair.Value["source"].GetValue<string>());
    var target=System.IO.Path.Combine(root,pair.Key); var after=pair.Value["after"].GetValue<string>();
    var before=pair.Value["before"]?.GetValue<string>();
    var backup=before is null?null:System.IO.Path.Combine(root,pair.Value["backup"].GetValue<string>());
    if(Hash(source)!=after || (backup is not null && Hash(backup)!=before)) throw new InvalidOperationException("Staged source or backup changed.");
    if(System.IO.File.Exists(target) && Hash(target)==after) continue;
    if(before is null?System.IO.File.Exists(target):!System.IO.File.Exists(target)||Hash(target)!=before)
        throw new InvalidOperationException("A shared target changed: "+pair.Key);
    plans.Add((source,target,backup,after));
}
var changed=new System.Collections.Generic.List<(string Target,string Backup)>();
try
{
    foreach(var item in plans)
    {
        System.IO.Directory.CreateDirectory(System.IO.Path.GetDirectoryName(item.Target));
        var incoming=item.Target+".water-v3-incoming";
        try
        {
            System.IO.File.Copy(item.Source,incoming,true);
            if(Hash(incoming)!=item.After) throw new InvalidOperationException("Incoming checksum failed.");
            System.IO.File.Move(incoming,item.Target,true); changed.Add((item.Target,item.Backup));
        }
        finally {if(System.IO.File.Exists(incoming)) System.IO.File.Delete(incoming);}
    }
}
catch
{
    foreach(var item in changed.AsEnumerable().Reverse())
        if(item.Backup is not null) System.IO.File.Copy(item.Backup,item.Target,true);
        else if(System.IO.File.Exists(item.Target)) System.IO.File.Delete(item.Target);
    throw;
}
foreach(var item in plans) Editor.AssetSystem.RegisterFile(item.Target);
foreach(var pair in manifest["protected_sha256"].AsObject())
    if(Hash(System.IO.Path.Combine(root,pair.Key))!=pair.Value.GetValue<string>()) throw new InvalidOperationException("A protected file changed during water import.");
var preserved=ReferenceEquals(selected,Editor.SceneEditorSession.Active)&&playing==Game.IsPlaying&&game==Game.ActiveScene?.Id&&dirty==selected?.HasUnsavedChanges;
var result=new {Phase=phase,Revision=revision,Imported=plans.Count,EditorStatePreserved=preserved,
    SceneChanged=false,PhysicsChanged=false,CarvingsAndBellChanged=false,AllNonWaterSourceBytesPreserved=true};
System.IO.File.WriteAllText(System.IO.Path.Combine(folder,revision,"native-import-"+phase+".json"),System.Text.Json.JsonSerializer.Serialize(result));
if(!preserved) throw new InvalidOperationException("Editor state changed during publication; inspect the receipt.");
return result;
