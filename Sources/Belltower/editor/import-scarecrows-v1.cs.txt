// Publish only checked scarecrow mesh/material additions. Scene and viewport
// state are preserved; neither foliage transforms nor collision are edited.
var root=Sandbox.Project.Current.GetRootPath();
var stage=System.IO.Path.Combine(root,"output/belltower/scarecrow-v1/stage");
var manifest=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(stage,"manifest.json")));
var session=Editor.SceneEditorSession.Active;var playing=Game.IsPlaying;var dirty=session?.HasUnsavedChanges;
var scenePath=System.IO.Path.Combine(root,"Assets/scenes/maps/Belltower/belltower.scene");
string Hash(string path)
{
    using var stream=System.IO.File.OpenRead(path);
    return Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(stream)).ToLowerInvariant();
}
var sceneHash=Hash(scenePath);
foreach(var group in new[]{"source_sha256","texture_sha256"})
    foreach(var item in manifest[group].AsObject())
        if(Hash(System.IO.Path.Combine(root,item.Key))!=item.Value.GetValue<string>())
            throw new InvalidOperationException("Staged input changed: "+item.Key);
var allowed=new System.Collections.Generic.HashSet<string>();
foreach(var side in new[]{"west","east"})
    foreach(var extension in new[]{"obj","mtl","vmdl"})
        allowed.Add("Assets/models/maps/belltower/belltower_"+side+"."+extension);
foreach(var name in new[]{"hat","patch","stitch"})allowed.Add("Assets/materials/belltower/scarecrow-v1/"+name+".vmat");
foreach(var extension in new[]{"obj","mtl","vmdl"})allowed.Add("Assets/models/maps/belltower/scarecrow-v1/scarecrow_v1."+extension);
if(!allowed.SetEquals(manifest["files"].AsObject().Select(p=>p.Key)))throw new InvalidOperationException("Unexpected scarecrow asset set.");
var plan=new System.Collections.Generic.List<(string Source,string Target,string Backup,string Before,string After)>();
foreach(var item in manifest["files"].AsObject())
{
    var source=System.IO.Path.Combine(stage,item.Key);var target=System.IO.Path.Combine(root,item.Key);
    var after=item.Value["after"].GetValue<string>();var before=item.Value["before"]?.GetValue<string>();
    if(Hash(source)!=after)throw new InvalidOperationException("Staged asset changed: "+item.Key);
    var current=System.IO.File.Exists(target)?Hash(target):null;
    if(current==after)continue;
    if(current!=before)throw new InvalidOperationException("Installed asset changed: "+item.Key);
    var backup=System.IO.Path.Combine(stage,"before",item.Key);
    if(before is not null && (!System.IO.File.Exists(backup) || Hash(backup)!=before))
        throw new InvalidOperationException("Original asset backup differs: "+item.Key);
    plan.Add((source,target,backup,before,after));
}
var changed=new System.Collections.Generic.List<(string Target,string Backup,string Before)>();
try
{
    foreach(var item in plan.OrderBy(p=>p.Target.EndsWith(".vmat")?0:p.Target.EndsWith(".mtl")?1:p.Target.EndsWith(".vmdl")?2:3))
    {
        var incoming=item.Target+".scarecrow-incoming";
        try
        {
            System.IO.File.Copy(item.Source,incoming);
            if(Hash(incoming)!=item.After)throw new InvalidOperationException("Incoming bytes differ.");
            changed.Add((item.Target,item.Backup,item.Before));System.IO.File.Move(incoming,item.Target,true);
        }
        finally{if(System.IO.File.Exists(incoming))System.IO.File.Delete(incoming);}
    }
}
catch
{
    foreach(var item in changed.AsEnumerable().Reverse())
        if(item.Before is null)System.IO.File.Delete(item.Target);else System.IO.File.Copy(item.Backup,item.Target,true);
    throw;
}
foreach(var item in plan)Editor.AssetSystem.RegisterFile(item.Target);
if(!ReferenceEquals(session,Editor.SceneEditorSession.Active) || Game.IsPlaying!=playing
    || session?.HasUnsavedChanges!=dirty || Hash(scenePath)!=sceneHash)
    throw new InvalidOperationException("Scene or editor state changed during import.");
var result=new {Imported=plan.Count,SceneSha256=sceneHash,SceneUnchanged=true,
    EditorStatePreserved=true,ViewportUntouched=true,PhysicsUnchanged=true,FoliageUnchanged=true,ForcedTownCompile=false};
System.IO.File.WriteAllText(System.IO.Path.Combine(stage,"native-import.json"),System.Text.Json.JsonSerializer.Serialize(result));
return result;
