// Publish only the guarded, additive casement imports into the existing models.
var root = Sandbox.Project.Current.GetRootPath();
var folder = System.IO.Path.Combine(root, "output/belltower/open-windows-v1");
var revision = System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder, "import-request.json")))["revision"].GetValue<string>();
if (!System.Text.RegularExpressions.Regex.IsMatch(revision, "^[a-z0-9-]+$")) throw new InvalidOperationException("Invalid revision.");
var manifest = System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder, revision, "stage/manifest.json")));
if (manifest["revision"].GetValue<string>() != "open-windows-v1") throw new InvalidOperationException("Wrong manifest.");
var selected = Editor.SceneEditorSession.Active;
var playing = Game.IsPlaying;
var game = Game.ActiveScene?.Id;
var dirty = selected?.HasUnsavedChanges;
if (dirty == true) throw new InvalidOperationException("Inspect the unsaved session before importing the windows.");
string Hash(string path) => Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(System.IO.File.ReadAllBytes(path))).ToLowerInvariant();
foreach (var group in new[] { "source_sha256", "protected_sha256" })
    foreach (var pair in manifest[group].AsObject())
        if (Hash(System.IO.Path.Combine(root, pair.Key)) != pair.Value.GetValue<string>()) throw new InvalidOperationException("Shared input changed: " + pair.Key);
var plans = new System.Collections.Generic.List<(string Source, string Target, string Backup, string After, int Order)>();
foreach (var pair in manifest["files"].AsObject())
{
    var relative = pair.Key;
    if (!(relative.StartsWith("Assets/models/maps/belltower/open-windows-v1/", StringComparison.Ordinal)
        || new[] { "centre", "west", "east" }.Any(part => relative == "Assets/models/maps/belltower/belltower_" + part + ".vmdl"))
        || relative.Contains("..")) throw new InvalidOperationException("Unexpected target.");
    var source = System.IO.Path.Combine(root, pair.Value["source"].GetValue<string>());
    var target = System.IO.Path.Combine(root, relative);
    var after = pair.Value["after"].GetValue<string>();
    var before = pair.Value["before"]?.GetValue<string>();
    var backup = before is null ? null : System.IO.Path.Combine(root, pair.Value["backup"].GetValue<string>());
    if (Hash(source) != after || (backup is not null && Hash(backup) != before)) throw new InvalidOperationException("Stage or backup differs.");
    if (System.IO.File.Exists(target) && Hash(target) == after) continue;
    if (before is null ? System.IO.File.Exists(target) : !System.IO.File.Exists(target) || Hash(target) != before)
        throw new InvalidOperationException("Shared target changed: " + relative);
    plans.Add((source, target, backup, after, relative.EndsWith(".vmdl", StringComparison.Ordinal) ? 1 : 0));
}
// During play, allow only a visual correction to our already-installed private
// mesh. Descriptions, physics, scene documents and the play session stay intact.
if (playing && plans.Any(p => !p.Target.Replace('\\', '/').Contains("/Assets/models/maps/belltower/open-windows-v1/", StringComparison.Ordinal)
    || !p.Target.EndsWith(".obj", StringComparison.OrdinalIgnoreCase)))
    throw new InvalidOperationException("Only private render-mesh updates are allowed during play.");
var changed = new System.Collections.Generic.List<(string Target, string Backup)>();
try
{
    foreach (var item in plans.OrderBy(p => p.Order))
    {
        System.IO.Directory.CreateDirectory(System.IO.Path.GetDirectoryName(item.Target));
        var incoming = item.Target + ".open-windows-v1-incoming";
        try
        {
            System.IO.File.Copy(item.Source, incoming, true);
            if (Hash(incoming) != item.After) throw new InvalidOperationException("Incoming checksum failed.");
            System.IO.File.Move(incoming, item.Target, true);
            changed.Add((item.Target, item.Backup));
        }
        finally { if (System.IO.File.Exists(incoming)) System.IO.File.Delete(incoming); }
    }
}
catch
{
    foreach (var item in changed.AsEnumerable().Reverse())
        if (item.Backup is not null) System.IO.File.Copy(item.Backup, item.Target, true);
        else if (System.IO.File.Exists(item.Target)) System.IO.File.Delete(item.Target);
    throw;
}
foreach (var item in plans.OrderBy(p => p.Order)) Editor.AssetSystem.RegisterFile(item.Target);
foreach (var pair in manifest["protected_sha256"].AsObject())
    if (Hash(System.IO.Path.Combine(root, pair.Key)) != pair.Value.GetValue<string>()) throw new InvalidOperationException("Protected resource changed: " + pair.Key);
var preserved = ReferenceEquals(selected, Editor.SceneEditorSession.Active) && playing == Game.IsPlaying && game == Game.ActiveScene?.Id && dirty == selected?.HasUnsavedChanges;
var result = new { Revision = revision, Imported = plans.Count, Windows = manifest["windows"].GetValue<int>(), Leaves = manifest["leaves"].GetValue<int>(),
    EditorStatePreserved = preserved, SceneChanged = false, PhysicsChanged = false };
System.IO.File.WriteAllText(System.IO.Path.Combine(folder, revision, "native-import.json"), System.Text.Json.JsonSerializer.Serialize(result));
if (!preserved) throw new InvalidOperationException("Editor state changed during publication; inspect before continuing.");
return result;
