// Run only in the coordinated editor window. These versioned resources are
// unreferenced until a temporary material comparison assigns the trial material.
var root=Sandbox.Project.Current.GetRootPath();
var stage=System.IO.Path.Combine(root,"output/belltower/quality/forest-detail-v1");
string Hash(string path)
{
    using var stream=System.IO.File.OpenRead(path);
    return Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(stream)).ToLowerInvariant();
}
var manifestPath=System.IO.Path.Combine(stage,"manifest.json");
var manifest=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(manifestPath));
var checks=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(stage,"checks.json")));
if(manifest["revision"].GetValue<string>()!="forest-detail-v1" || !checks["passed"].GetValue<bool>()
    || checks["manifest_sha256"].GetValue<string>()!=Hash(manifestPath)
    || checks["checker_sha256"].GetValue<string>()!=Hash(System.IO.Path.Combine(root,"Tools/belltower_check_forest_detail.py"))
    || manifest["geometry_changed"].GetValue<bool>() || manifest["texture_inputs_changed"].GetValue<bool>()
    || manifest["calibration_sha256"].GetValue<string>()!=Hash(System.IO.Path.Combine(stage,"calibration.json")))
    throw new InvalidOperationException("Current forest-detail source checks are required.");
foreach(var source in manifest["source_sha256"].AsObject())
    if(Hash(System.IO.Path.Combine(root,source.Key))!=source.Value.GetValue<string>())
        throw new InvalidOperationException("Forest-detail input changed: "+source.Key);
var allowed=new System.Collections.Generic.HashSet<string>(StringComparer.Ordinal)
{
    "shaders/belltower_forest_detail_trial.shader",
    "materials/belltower/forest/trials/forest_detail_v1.vmat"
};
var seen=new System.Collections.Generic.HashSet<string>(StringComparer.Ordinal);
var files=manifest["files"].AsArray();
if(files.Count!=2) throw new InvalidOperationException("Unexpected forest-detail asset count.");
foreach(var file in files)
{
    var relative=file["path"].GetValue<string>();
    if(!allowed.Contains(relative) || !seen.Add(relative)
        || file["source"].GetValue<string>()!="output/belltower/quality/forest-detail-v1/Assets/"+relative
        || Hash(System.IO.Path.Combine(root,file["source"].GetValue<string>()))!=file["sha256"].GetValue<string>())
        throw new InvalidOperationException("Unexpected or changed forest-detail source.");
    var target=System.IO.Path.Combine(root,"Assets",relative);
    if(System.IO.File.Exists(target) && Hash(target)!=file["sha256"].GetValue<string>())
        throw new InvalidOperationException("Different trial asset already exists: "+relative);
}
var game=Game.ActiveScene?.Id;
var playing=Game.IsPlaying;
var selected=Editor.SceneEditorSession.Active;
var written=0;
foreach(var file in files)
{
    cancellationToken.ThrowIfCancellationRequested();
    var relative=file["path"].GetValue<string>();
    var target=System.IO.Path.Combine(root,"Assets",relative);
    if(System.IO.File.Exists(target)) continue;
    var incoming=System.IO.Path.Combine(stage,"native-incoming",System.IO.Path.GetFileName(relative));
    System.IO.Directory.CreateDirectory(System.IO.Path.GetDirectoryName(incoming));
    System.IO.Directory.CreateDirectory(System.IO.Path.GetDirectoryName(target));
    System.IO.File.Copy(System.IO.Path.Combine(root,file["source"].GetValue<string>()),incoming,true);
    if(Hash(incoming)!=file["sha256"].GetValue<string>()) throw new InvalidOperationException("Incoming source changed.");
    System.IO.File.Move(incoming,target);
    written++;
}
foreach(var relative in allowed) Editor.AssetSystem.RegisterFile(System.IO.Path.Combine(root,"Assets",relative));
var result=new {Imported=2,NewFiles=written,ManifestSha256=Hash(manifestPath),SceneSaved=false,SceneObjectsAdded=0,
    GamePreserved=Game.ActiveScene?.Id==game && Game.IsPlaying==playing,
    SelectedDocumentPreserved=ReferenceEquals(selected,Editor.SceneEditorSession.Active),NativeVisualAcceptance=false};
System.IO.File.WriteAllText(System.IO.Path.Combine(stage,"native-import-result.json"),System.Text.Json.JsonSerializer.Serialize(result));
return result;
