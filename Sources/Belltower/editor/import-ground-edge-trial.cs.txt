// Import unreferenced, versioned trial assets in a coordinated editor window.
// This does not insert scene objects, stop play, change selection, or save scenes.
var root=Sandbox.Project.Current.GetRootPath();
var stage=System.IO.Path.Combine(root,"output/belltower/quality/ground-edge-v1");
string Hash(string path)
{
    using var input=System.IO.File.OpenRead(path);
    return Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(input)).ToLowerInvariant();
}
var manifestPath=System.IO.Path.Combine(stage,"manifest.json");
var manifest=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(manifestPath));
var checks=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(stage,"geometry-checks.json")));
if(manifest["revision"].GetValue<string>()!="ground-edge-v1" || !checks["passed"].GetValue<bool>()
    || checks["manifest_sha256"].GetValue<string>()!=Hash(manifestPath)
    || checks["checker_sha256"].GetValue<string>()!=Hash(System.IO.Path.Combine(root,"Tools/belltower_check_ground_edges.py"))
    || !checks["flat_paving_containment"].GetValue<bool>() || !checks["protected_thresholds_clear"].GetValue<bool>()
    || !checks["original_pixels_and_alpha_preserved"].GetValue<bool>()
    || checks["authored_physics_shapes"].GetValue<int>()!=0)
    throw new InvalidOperationException("Current ground-detail source/placement validation is required.");
foreach(var input in manifest["source_sha256"].AsObject())
    if(Hash(System.IO.Path.Combine(root,input.Key))!=input.Value.GetValue<string>())
        throw new InvalidOperationException("A checked ground-detail input changed: "+input.Key);
var allowed=new System.Collections.Generic.HashSet<string>(StringComparer.Ordinal)
{
    "shaders/belltower_ground_edge_trial.shader",
    "materials/belltower/ground-edge-v1/ground_edge.png",
    "materials/belltower/ground-edge-v1/ground_edge.vmat"
};
foreach(var side in new[]{"west","east"})
    foreach(var extension in new[]{"obj","mtl","vmdl"})
        allowed.Add("models/maps/belltower/ground-edge-v1/belltower_ground_edge_"+side+"."+extension);
var files=manifest["files"].AsArray();
var seen=new System.Collections.Generic.HashSet<string>(StringComparer.Ordinal);
if(files.Count!=allowed.Count) throw new InvalidOperationException("Unexpected trial asset count.");
foreach(var file in files)
{
    var relative=file["path"].GetValue<string>();
    var expected=file["sha256"].GetValue<string>();
    if(!allowed.Contains(relative) || !seen.Add(relative)
        || file["source"].GetValue<string>()!="output/belltower/quality/ground-edge-v1/Assets/"+relative
        || Hash(System.IO.Path.Combine(root,file["source"].GetValue<string>()))!=expected)
        throw new InvalidOperationException("Unexpected or changed ground-detail source: "+relative);
    var target=System.IO.Path.Combine(root,"Assets",relative);
    if(System.IO.File.Exists(target) && Hash(target)!=expected)
        throw new InvalidOperationException("A different trial asset already exists: "+relative);
}
var game=Game.ActiveScene?.Id;
var playing=Game.IsPlaying;
var selected=Editor.SceneEditorSession.Active;
var written=0;
foreach(var file in files)
{
    cancellationToken.ThrowIfCancellationRequested();
    var relative=file["path"].GetValue<string>();
    var target=System.IO.Path.Combine(root,"Assets",relative);
    if(System.IO.File.Exists(target)) continue;
    var incoming=System.IO.Path.Combine(stage,"native-incoming",relative);
    System.IO.Directory.CreateDirectory(System.IO.Path.GetDirectoryName(incoming));
    System.IO.Directory.CreateDirectory(System.IO.Path.GetDirectoryName(target));
    System.IO.File.Copy(System.IO.Path.Combine(root,file["source"].GetValue<string>()),incoming,true);
    if(Hash(incoming)!=file["sha256"].GetValue<string>()) throw new InvalidOperationException("Incoming file differs: "+relative);
    // Do not replace a destination another operation may have created after preflight.
    System.IO.File.Move(incoming,target);
    if(Hash(target)!=file["sha256"].GetValue<string>()) throw new InvalidOperationException("Published file differs: "+relative);
    written++;
}
foreach(var relative in new[]{"shaders/belltower_ground_edge_trial.shader","materials/belltower/ground-edge-v1/ground_edge.vmat",
    "models/maps/belltower/ground-edge-v1/belltower_ground_edge_west.vmdl","models/maps/belltower/ground-edge-v1/belltower_ground_edge_east.vmdl"})
    Editor.AssetSystem.RegisterFile(System.IO.Path.Combine(root,"Assets",relative));
var result=new {Imported=files.Count,NewFiles=written,ExistingIdentical=files.Count-written,ManifestSha256=Hash(manifestPath),
    SceneObjectsAdded=0,SceneSaved=false,NativeVisualAcceptance=false,
    GamePreserved=Game.ActiveScene?.Id==game && Game.IsPlaying==playing,
    SelectedDocumentPreserved=ReferenceEquals(selected,Editor.SceneEditorSession.Active)};
System.IO.File.WriteAllText(System.IO.Path.Combine(stage,"native-import-result.json"),System.Text.Json.JsonSerializer.Serialize(result));
return result;
