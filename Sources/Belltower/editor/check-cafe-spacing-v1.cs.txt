// Read-only physics check: moved tables collide at their new positions; old positions are clear.
var root=Sandbox.Project.Current.GetRootPath();
var session=Editor.SceneEditorSession.All.Single(s=>s.Scene.IsEditor&&s.Scene.Source?.ResourcePath?.Equals("scenes/maps/belltower/belltower.scene",StringComparison.OrdinalIgnoreCase)==true);
var scene=session.Scene;
var samples=new (string Name,Vector2 Plan,float Height)[]{
    ("new-0",new Vector2(565,-340),29),("new-1",new Vector2(520,-200),29),("new-2",new Vector2(565,-60),29),
    ("old-0",new Vector2(600,-330),0),("old-1",new Vector2(520,-250),0),
    ("old-2",new Vector2(600,-170),0),("removed-3",new Vector2(520,-90),0)};
var results=new System.Collections.Generic.List<object>();
var passed=true;
using(scene.Push())
foreach(var sample in samples)
{
    var from=new Vector3(sample.Plan.y,-sample.Plan.x,70);
    var hit=scene.Trace.Ray(from,from.WithZ(-5)).UsePhysicsWorld(true).UseRenderMeshes(false).Run();
    var ok=hit.Hit&&!hit.StartedSolid&&MathF.Abs(hit.EndPosition.z-sample.Height)<1;
    passed&=ok;
    results.Add(new {sample.Name,Plan=sample.Plan.ToString(),ExpectedHeight=sample.Height,hit.Hit,hit.StartedSolid,
        Height=hit.EndPosition.z,Object=hit.GameObject?.Name,Passed=ok});
}
var result=new {Passed=passed,Checks=results,Scene=session.Scene.Source.ResourcePath};
System.IO.File.WriteAllText(System.IO.Path.Combine(root,"output/belltower/cafe-spacing-v1/native-collision-check.json"),System.Text.Json.JsonSerializer.Serialize(result));
if(!passed) throw new InvalidOperationException("Inspect the recorded cafe collision samples.");
return result;
