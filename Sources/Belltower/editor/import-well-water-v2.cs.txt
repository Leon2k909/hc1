// Publish only the well's private shader and water material; preserve the document.
var root = Sandbox.Project.Current.GetRootPath();
if (!System.IO.Path.GetFullPath(root).TrimEnd('\\').Equals(@"C:\sbox-projects\hc1\forked-shooter-main", StringComparison.OrdinalIgnoreCase))
    throw new InvalidOperationException("Wrong project for the well-water update.");
var folder = System.IO.Path.Combine(root, "output/belltower/well-water-v2");
var request = System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder, "import-request.json")));
var revision = request["revision"].GetValue<string>();
if (!System.Text.RegularExpressions.Regex.IsMatch(revision, "^[a-z0-9-]+$")) throw new InvalidOperationException("Invalid revision.");
var manifest = System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder, revision, "stage/manifest.json")));
var allowed = new System.Collections.Generic.HashSet<string> {
    "Assets/shaders/belltower_well_water_v2.shader", "Assets/materials/belltower/well-v1/well_water.vmat" };
var selected = Editor.SceneEditorSession.Active;
var playing = Game.IsPlaying;
var game = Game.ActiveScene?.Id;
var dirty = selected?.HasUnsavedChanges;
if (playing || dirty != false || selected?.Scene.Source?.ResourcePath?.Equals("scenes/maps/belltower/belltower.scene", StringComparison.OrdinalIgnoreCase) != true)
    throw new InvalidOperationException("Well-water publication requires the existing saved, stopped Belltower document.");
string Hash(string path) => Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(System.IO.File.ReadAllBytes(path))).ToLowerInvariant();
void CheckState()
{
    if (!ReferenceEquals(selected, Editor.SceneEditorSession.Active) || playing != Game.IsPlaying || game != Game.ActiveScene?.Id || dirty != selected?.HasUnsavedChanges)
        throw new InvalidOperationException("Editor state changed; do not continue the well-water publication.");
}
var imported = new System.Collections.Generic.List<string>();
foreach (var relative in new[] { "Assets/shaders/belltower_well_water_v2.shader", "Assets/materials/belltower/well-v1/well_water.vmat" })
{
    CheckState();
    var file = manifest["files"][relative];
    if (!allowed.Contains(relative) || file is null) throw new InvalidOperationException("Unexpected well-water target.");
    var source = System.IO.Path.Combine(root, file["source"].GetValue<string>());
    var target = System.IO.Path.Combine(root, relative);
    var after = file["after"].GetValue<string>();
    var before = file["before"]?.GetValue<string>();
    if (Hash(source) != after) throw new InvalidOperationException("Staged water source changed.");
    if (!System.IO.File.Exists(target) || Hash(target) != after)
    {
        if (before is null ? System.IO.File.Exists(target) : !System.IO.File.Exists(target) || Hash(target) != before)
            throw new InvalidOperationException("Shared water target changed: " + relative);
        if (before is not null && Hash(System.IO.Path.Combine(root, file["backup"].GetValue<string>())) != before)
            throw new InvalidOperationException("Water backup changed.");
        var incoming = target + ".well-water-incoming";
        try
        {
            System.IO.File.Copy(source, incoming, true);
            if (Hash(incoming) != after) throw new InvalidOperationException("Incoming checksum failed.");
            System.IO.File.Move(incoming, target, true);
        }
        finally { if (System.IO.File.Exists(incoming)) System.IO.File.Delete(incoming); }
        Editor.AssetSystem.RegisterFile(target);
        imported.Add(relative);
    }
    if (relative.EndsWith(".shader", StringComparison.Ordinal))
    {
        await Editor.EditorUtility.CompileShader("shaders/belltower_well_water_v2.shader", new(), cancellationToken);
        if (!System.IO.File.Exists(target + "_c") || System.IO.File.GetLastWriteTimeUtc(target + "_c") < System.IO.File.GetLastWriteTimeUtc(target))
            throw new InvalidOperationException("Private well-water shader did not compile; the material has not been switched.");
    }
}
CheckState();
var result = new { Revision = revision, Imported = imported, ShaderCompiled = true, EditorStatePreserved = true, SceneChanged = false };
System.IO.File.WriteAllText(System.IO.Path.Combine(folder, revision, "native-import.json"), System.Text.Json.JsonSerializer.Serialize(result));
return result;
