// Remove only the four square clock patches and publish their round replacement.
// The original dial textures, model resource, physics and scene stay untouched.
var root=Sandbox.Project.Current.GetRootPath();
var stage=System.IO.Path.Combine(root,"output/belltower/clock-v1/stage");
var manifest=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(stage,"manifest.json")));
var session=Editor.SceneEditorSession.Active;
if (Game.IsPlaying || session is null || session.HasUnsavedChanges
    || !session.Scene.Source.ResourcePath.Equals("scenes/maps/belltower/belltower.scene",StringComparison.OrdinalIgnoreCase))
    throw new InvalidOperationException("Requires the active, stopped, saved Belltower document.");
string Hash(string path)=>Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(System.IO.File.ReadAllBytes(path))).ToLowerInvariant();
var scenePath=System.IO.Path.Combine(root,"Assets/scenes/maps/Belltower/belltower.scene");
if (Hash(scenePath)!=manifest["scene_sha256"].GetValue<string>()) throw new InvalidOperationException("Saved scene changed since staging.");
foreach (var section in new[]{"source_sha256","texture_sha256"})
    foreach (var pair in manifest[section].AsObject())
        if (Hash(System.IO.Path.Combine(root,pair.Key))!=pair.Value.GetValue<string>())
            throw new InvalidOperationException("Staged input changed: "+pair.Key);
var relative=manifest["path"].GetValue<string>();
if (relative!="Assets/models/maps/belltower/belltower_centre.obj") throw new InvalidOperationException("Unexpected target.");
var source=System.IO.Path.Combine(stage,relative);
var target=System.IO.Path.Combine(root,relative);
var before=manifest["before"].GetValue<string>();
var after=manifest["after"].GetValue<string>();
if (Hash(source)!=after) throw new InvalidOperationException("Staged mesh changed.");
if (Hash(System.IO.Path.ChangeExtension(target,"vmdl"))!=manifest["vmdl_sha256"].GetValue<string>()
    || Hash(System.IO.Path.ChangeExtension(target,"mtl"))!=manifest["mtl_sha256"].GetValue<string>())
    throw new InvalidOperationException("Model or material mapping changed since staging.");
var current=Hash(target);
var backup=System.IO.Path.Combine(stage,"before",relative);
if (current!=after)
{
    if (current!=before) throw new InvalidOperationException("Installed mesh changed since staging.");
    System.IO.Directory.CreateDirectory(System.IO.Path.GetDirectoryName(backup));
    if (!System.IO.File.Exists(backup)) System.IO.File.Copy(target,backup);
    if (Hash(backup)!=before) throw new InvalidOperationException("Preserved original differs.");
    var incoming=target+".clock-incoming";
    try
    {
        System.IO.File.Copy(source,incoming,true);
        if (Hash(incoming)!=after) throw new InvalidOperationException("Incoming mesh differs.");
        System.IO.File.Move(incoming,target,true);
    }
    finally { if (System.IO.File.Exists(incoming)) System.IO.File.Delete(incoming); }
    Editor.AssetSystem.RegisterFile(target);
}
if (Hash(scenePath)!=manifest["scene_sha256"].GetValue<string>() || session.HasUnsavedChanges || Game.IsPlaying
    || !ReferenceEquals(session,Editor.SceneEditorSession.Active)) throw new InvalidOperationException("Editor state changed during import.");
var result=new {Imported=current==after?0:1,Path=relative,Sha256=after,
    SceneUnchanged=true,PhysicsUnchanged=true,TexturesUnchanged=true,ForcedModelCompile=false};
System.IO.File.WriteAllText(System.IO.Path.Combine(stage,"native-import.json"),System.Text.Json.JsonSerializer.Serialize(result));
return result;
