// Isolate macro/fine-normal contributions using private material copies only.
var root=Sandbox.Project.Current.GetRootPath();
var folder=System.IO.Path.Combine(root,"output/firstperson-clothing-v3-20261021/normal-inspection");
System.IO.Directory.CreateDirectory(folder);
var selected=Editor.SceneEditorSession.Active;var playing=Game.IsPlaying;var dirty=selected.HasUnsavedChanges;
var scene=Scene.CreateEditorScene();
try
{
    CameraComponent camera=null;SkinnedModelRenderer renderer=null;Material source=null;
    using(scene.Push())
    {
        var light=new GameObject(true,"Normal inspection light").Components.Create<DirectionalLight>();
        light.WorldRotation=Rotation.From(new Angles(35,25,0));light.LightColor=Color.White*2.2f;light.SkyColor=new Color(.72f,.78f,.85f)*.65f;
        camera=new GameObject(true,"Normal inspection camera").Components.Create<CameraComponent>();
        camera.BackgroundColor=new Color(.075f,.09f,.11f);camera.FieldOfView=44;camera.FovAxis=CameraComponent.Axis.Horizontal;
        camera.CustomSize=new Vector2(1400,1000);camera.ZNear=.15f;camera.ZFar=500;camera.EnablePostProcessing=false;camera.AutoExposure.Enabled=false;
        typeof(Scene).GetProperty("Camera",System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.Public|System.Reflection.BindingFlags.NonPublic).SetValue(scene,camera);
        var model=Model.Load("models/citizen_clothes/shirt/army_shirt/model/army_shirt_m_human.vmdl");
        renderer=new GameObject(true,"Normal inspection shirt").Components.Create<SkinnedModelRenderer>();renderer.Model=model;renderer.UseAnimGraph=false;renderer.Tint=new Color(.76f,.86f,.66f);
        var target=(model.Bones.GetBone("arm_lower_R_twist1").LocalTransform.Position+model.Bones.GetBone("hand_R").LocalTransform.Position)*.5f;
        camera.WorldPosition=target+new Vector3(7,-7,5);camera.WorldRotation=Rotation.LookAt(target-camera.WorldPosition);
        source=Material.Load("materials/operators/first_person/v3/shirt.vmat");
    }
    foreach(var mode in new[]{"default","macro-only","fine-only","albedo-only"})
    {
        Material material=null;
        using(scene.Push())
        {
            material=source.CreateCopy("first-person-normal-diagnostic-"+mode);
            if(mode=="macro-only"||mode=="albedo-only")material.Set("g_flFineNormalStrength",0f);
            if(mode=="fine-only"||mode=="albedo-only")material.Set("g_tNormal",Texture.Load("materials/default/default_normal.tga"));
            renderer.Materials.SetOverride(0,material);
        }
        using var bitmap=new Bitmap(1400,1000);
        for(var f=0;f<12;f++)
        {
            using(scene.Push())
            {
                foreach(var key in new[]{"g_tColor","g_tNormal","g_tRma","g_tFineNormal","g_tFineData","g_tSleeveLeft","g_tSleeveRight","g_tCuffMain","g_tCuffStrip"})
                {var t=material.GetTexture(key);if(t is not null&&t.IsValid&&!t.IsError)t.MarkUsed(Math.Max(t.Width,t.Height));}
                scene.EditorTick(.016f,.016f);camera.RenderToBitmap(bitmap,false);
            }
            await Task.Delay(80,cancellationToken);
        }
        System.IO.File.WriteAllBytes(System.IO.Path.Combine(folder,mode+".png"),bitmap.ToPng());
    }
}
finally{scene.Destroy();}
return new{SharedEditorStatePreserved=ReferenceEquals(selected,Editor.SceneEditorSession.Active)&&playing==Game.IsPlaying&&dirty==selected.HasUnsavedChanges,TemporarySceneDestroyed=true};
