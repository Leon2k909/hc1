// Run a copy in sbox-cli's exchange directory. Installs only the three screen assets and the scoped mesh changes
// staged by ares_screen_refresh.py. Compile ar_screens.vmat, then the hall, dressing and reactor models.
var project = Sandbox.Project.Current.GetRootPath();
var directory = System.IO.Path.Combine( project, "output/ares/screens-20261020" );
var stage = System.IO.Path.Combine( directory, "stage" );
var before = System.Text.Json.Nodes.JsonNode.Parse( System.IO.File.ReadAllText(
    System.IO.Path.Combine( directory, "before/manifest.json" ) ) ).AsObject();
var installedPath = System.IO.Path.Combine( directory, "installed.json" );
var expected = System.IO.File.Exists( installedPath )
    ? System.Text.Json.Nodes.JsonNode.Parse( System.IO.File.ReadAllText( installedPath ) ).AsObject() : before;
var scene = Editor.SceneEditorSession.Active?.Scene;
if ( scene?.Name != "ares" || Game.IsPlaying ) throw new InvalidOperationException( "Open Ares in edit mode before importing" );
string Hash( string path ) => Convert.ToHexString( System.Security.Cryptography.SHA256.HashData( System.IO.File.ReadAllBytes( path ) ) ).ToLowerInvariant();
var allowed = new System.Collections.Generic.HashSet<string>( StringComparer.OrdinalIgnoreCase ) {
    "Assets/models/maps/ares/ares_hall.obj", "Assets/models/maps/ares/ares_dressing.obj", "Assets/models/maps/ares/ares_reactor.obj",
    "Assets/materials/ares/ar_screens.vmat", "Assets/materials/ares/ar_screens_color.png", "Assets/materials/ares/ar_screens_illum.png"
};
var plan = new System.Collections.Generic.List<(string Source, string Target, string Relative)>();
foreach ( var source in System.IO.Directory.GetFiles( stage, "*", System.IO.SearchOption.AllDirectories ) )
{
    var relative = "Assets/" + System.IO.Path.GetRelativePath( stage, source ).Replace( '\\', '/' );
    if ( !allowed.Contains( relative ) ) throw new InvalidOperationException( "Unexpected staged asset: " + relative );
    var target = System.IO.Path.Combine( project, relative );
    if ( System.IO.File.Exists( target ) && Hash( target ) != Hash( source )
        && ( expected[relative] is null || Hash( target ) != expected[relative].GetValue<string>() ) )
        throw new InvalidOperationException( "Changed since staging: " + relative );
    plan.Add( (source, target, relative) );
}
if ( plan.Count != allowed.Count ) throw new InvalidOperationException( "Incomplete stage" );
foreach ( var item in plan )
{
    cancellationToken.ThrowIfCancellationRequested();
    if ( !System.IO.File.Exists( item.Target ) || Hash( item.Target ) != Hash( item.Source ) )
        System.IO.File.Copy( item.Source, item.Target, true );
    Editor.AssetSystem.RegisterFile( item.Target );
}
foreach ( var name in new[] { "ares_hall", "ares_dressing", "ares_reactor" } )
    Editor.AssetSystem.RegisterFile( System.IO.Path.Combine( project, "Assets/models/maps/ares/" + name + ".vmdl" ) );
var installed = plan.ToDictionary( item => item.Relative, item => Hash( item.Target ) );
System.IO.File.WriteAllText( installedPath, System.Text.Json.JsonSerializer.Serialize( installed,
    new System.Text.Json.JsonSerializerOptions { WriteIndented = true } ) );
var preserved = System.Text.Json.Nodes.JsonNode.Parse( System.IO.File.ReadAllText(
    System.IO.Path.Combine( directory, "preserve-manifest.json" ) ) ).AsObject();
var changed = preserved.Where( item => Hash( System.IO.Path.Combine( project, item.Key ) ) != item.Value.GetValue<string>() )
    .Select( item => item.Key ).ToArray();
var report = new { Imported = installed.Keys, SourceSceneUnchanged = scene == Editor.SceneEditorSession.Active?.Scene,
    Game.IsPlaying, PreservedAssetsChanged = changed };
System.IO.File.WriteAllText( System.IO.Path.Combine( directory, "import.json" ), System.Text.Json.JsonSerializer.Serialize( report,
    new System.Text.Json.JsonSerializerOptions { WriteIndented = true } ) );
return report;
