var root=Sandbox.Project.Current.GetRootPath();
var folder=System.IO.Path.Combine(root,"output/ares/observation-sign-v1");
var plan=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder,"install-plan.json")));
var session=Editor.SceneEditorSession.Active;
var before=new { Id=session.Scene.Id, Game.IsPlaying, Dirty=session.HasUnsavedChanges };
string Hash(string p)=>Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(System.IO.File.ReadAllBytes(p))).ToLowerInvariant();
var target=System.IO.Path.Combine(root,plan["target"].GetValue<string>());
var staged=System.IO.Path.Combine(folder,"install-stage.vmdl");
if(Hash(target)!=plan["before_sha256"].GetValue<string>()||Hash(staged)!=plan["after_sha256"].GetValue<string>())throw new InvalidOperationException("Source changed; inspect before mutation");
System.IO.File.Copy(staged,target,true);
Editor.AssetSystem.RegisterFile(target);
return new { Installed=plan["target"].GetValue<string>(), Sha256=Hash(target), UserStatePreserved=session.Scene.Id==before.Id&&Game.IsPlaying==before.IsPlaying&&session.HasUnsavedChanges==before.Dirty };
