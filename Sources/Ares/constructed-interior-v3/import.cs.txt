var root=Sandbox.Project.Current.GetRootPath();
var folder=System.IO.Path.Combine(root,"output/ares/constructed-interior-v3");
var manifest=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder,"manifest.json")));
string Hash(string p)=>Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(System.IO.File.ReadAllBytes(p))).ToLowerInvariant();
var textures=new System.Collections.Generic.List<object>();
foreach(var channel in new[]{"color","rough"}) {
 var source=System.IO.Path.Combine(root,"Sources/Ares/constructed-interior-v3/generated/skin-"+channel+"-v1.png");
 var target=System.IO.Path.Combine(root,"Assets/materials/ares/constructed-v3/skin_"+channel+".png");
 System.IO.Directory.CreateDirectory(System.IO.Path.GetDirectoryName(target));
 using var image=Bitmap.CreateFromBytes(System.IO.File.ReadAllBytes(source));using var packed=image.Resize(2048,1024,true);
 var bytes=packed.ToPng();var hash=Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(bytes)).ToLowerInvariant();
 if(!System.IO.File.Exists(target)||Hash(target)!=hash)System.IO.File.WriteAllBytes(target,bytes);
 Editor.AssetSystem.RegisterFile(target);
 textures.Add(new{Channel=channel,RawWidth=image.Width,RawHeight=image.Height,Width=2048,Height=1024,Hash=Hash(target)});
}
foreach(var pair in manifest["files"].AsObject()) {
 if(pair.Key.Contains("..")||!pair.Key.Contains("/constructed-v3/"))throw new InvalidOperationException("Unexpected target");
 var source=System.IO.Path.Combine(folder,"stage",pair.Key);
 if(Hash(source)!=pair.Value.GetValue<string>())throw new InvalidOperationException("Staged asset changed");
 var target=System.IO.Path.Combine(root,pair.Key);System.IO.Directory.CreateDirectory(System.IO.Path.GetDirectoryName(target));
 if(!System.IO.File.Exists(target)||Hash(target)!=pair.Value.GetValue<string>())System.IO.File.Copy(source,target,true);
 Editor.AssetSystem.RegisterFile(target);
}
System.IO.File.WriteAllText(System.IO.Path.Combine(folder,"skin-import.json"),System.Text.Json.JsonSerializer.Serialize(textures));
return new {Files=manifest["files"].AsObject().Count,Textures=textures,PreviewOnly=true};
