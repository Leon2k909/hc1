// Ares in an isolated editor scene: its visual roots only (no game modes, no navmesh), its reflection probes rendered,
// then each view of Sources/Ares/editor/capture-config.json rendered with settle frames for the auto exposure, to the
// config's directory (relative to the project). A view with "art": true is also written to Assets/ui/maps/ares.png
// (the menus' picture). Leaves the editor's own scene and play state alone, and destroys its scene at the end.
// Run it with sbox-cli's run_csharp (copy it into the session exchange directory as a .cs file). Views are in plan
// coordinates ( east, north, z ), as Tools/ares_map.py's.
var projectRoot = Sandbox.Project.Current.GetRootPath();
var config = System.Text.Json.Nodes.JsonNode.Parse( System.IO.File.ReadAllText(
	System.IO.Path.Combine( projectRoot, "Sources/Ares/editor/capture-config.json" ) ) ).AsObject();
var destination = System.IO.Path.Combine( projectRoot, config["directory"]?.GetValue<string>() ?? "output/ares" );
System.IO.Directory.CreateDirectory( destination );
var source = Editor.SceneEditorSession.Active?.Scene;
var before = new { Game.IsPlaying, SceneId = source?.Id, GameSceneId = Game.ActiveScene?.Id };
var visual = new System.Collections.Generic.HashSet<string>( StringComparer.OrdinalIgnoreCase ) {
	"Sun", "Sky", "Fog", "Look", "Ares", "Lights", "Reflection Probes" };
var root = System.Text.Json.Nodes.JsonNode.Parse( System.IO.File.ReadAllText( System.IO.Path.Combine( projectRoot, "Assets/scenes/maps/Ares/ares.scene" ) ) ).AsObject();
var objects = new System.Text.Json.Nodes.JsonArray();
foreach ( var item in root["GameObjects"].AsArray() )
	if ( visual.Contains( item["Name"]?.GetValue<string>() ?? "" ) ) objects.Add( item.DeepClone() );
root["GameObjects"] = objects;
root["SceneProperties"]["NavMesh"]["Enabled"] = false;
root["SceneProperties"]["WantsSystemScene"] = false;
var scene = Scene.CreateEditorScene();
var flags = System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.Public | System.Reflection.BindingFlags.NonPublic;
// RenderToBitmap alone doesn't run the scene's render hooks: run them as the app does, so fog and lights update.
var preRender = typeof( Scene ).GetMethod( "PreRender", flags, null, Type.EmptyTypes, null );
var preCameraRender = typeof( Scene ).GetMethod( "PreCameraRender", flags, null, Type.EmptyTypes, null );
var log = new System.Collections.Generic.List<string>();
try
{
	CameraComponent camera;
	using ( scene.Push() )
	{
		scene.Deserialize( root );
		var holder = new GameObject( false, "Ares capture camera" );
		camera = holder.Components.Create<CameraComponent>();
		camera.ZNear = 4; camera.ZFar = 120000;
		holder.Enabled = true;
		scene.EditorTick( .016f, .016f );
		preRender.Invoke( scene, null );
		preCameraRender.Invoke( scene, null );
	}
	var models = scene.GetAllComponents<ModelRenderer>().Select( r => new { r.GameObject.Name, Model = r.Model?.ResourcePath, Error = r.Model?.IsError ?? true } ).ToArray();
	var renderCubemap = typeof( EnvmapProbe ).GetMethod( "RenderCubemap", flags, null, Type.EmptyTypes, null );
	foreach ( var probe in scene.GetAllComponents<EnvmapProbe>() )
	{
		cancellationToken.ThrowIfCancellationRequested();
		try { using ( scene.Push() ) renderCubemap?.Invoke( probe, null ); } catch ( Exception e ) { log.Add( "probe: " + e.Message ); }
		await Task.Delay( 16, cancellationToken );
	}
	Vector3 World( System.Text.Json.Nodes.JsonNode p ) => new( p[1].GetValue<float>(), -p[0].GetValue<float>(), p[2].GetValue<float>() );
	using var bitmap = new Bitmap( config["width"].GetValue<int>(), config["height"].GetValue<int>() );
	foreach ( var view in config["views"].AsArray() )
	{
		cancellationToken.ThrowIfCancellationRequested();
		var name = view["name"].GetValue<string>();
		for ( var frame = 0; frame < config["settle"].GetValue<int>(); frame++ )
		{
			using ( scene.Push() )
			{
				camera.WorldPosition = World( view["eye"] );
				camera.WorldRotation = Rotation.LookAt( World( view["target"] ) - World( view["eye"] ) );
				camera.FieldOfView = view["fov"].GetValue<float>();
				scene.EditorTick( .016f, .016f );
				preRender.Invoke( scene, null );
				preCameraRender.Invoke( scene, null );
				camera.RenderToBitmap( bitmap );
			}
			await Task.Delay( 16, cancellationToken );
		}
		System.IO.File.WriteAllBytes( System.IO.Path.Combine( destination, name + ".png" ), bitmap.ToPng() );
		if ( view["art"]?.GetValue<bool>() == true )
			System.IO.File.WriteAllBytes( System.IO.Path.Combine( projectRoot, "Assets/ui/maps/ares.png" ), bitmap.ToPng() );
		log.Add( "wrote " + name );
	}
	return new { Before = before, SourceUnchanged = Editor.SceneEditorSession.Active?.Scene?.Id == before.SceneId && Game.ActiveScene?.Id == before.GameSceneId,
		Models = models, Log = log };
}
catch ( Exception error )
{
	System.IO.File.WriteAllText( System.IO.Path.Combine( destination, "capture-error.txt" ), error.ToString() );
	throw;
}
finally { scene.Destroy(); }
