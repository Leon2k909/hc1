// Publish only new case assets and the unreferenced, byte-preserved dressing
// base. The active map integration is a separate guarded editor operation.
var root=Sandbox.Project.Current.GetRootPath();
if(!string.Equals(System.IO.Path.GetFullPath(root).TrimEnd('\\','/'),@"C:\sbox-projects\hc1\forked-shooter-main",StringComparison.OrdinalIgnoreCase))throw new InvalidOperationException("Wrong project.");
var folder=System.IO.Path.Combine(root,"output/ares/crate-v2");
var request=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder,"import-request.json")));
var stage=request["stage"].GetValue<string>();
if(!System.Text.RegularExpressions.Regex.IsMatch(stage,"^stage-[a-z0-9-]+$"))throw new InvalidOperationException("Invalid stage.");
var manifest=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder,stage,"manifest.json")));
if(manifest["revision"].GetValue<string>()!="ares-cargo-case-v2")throw new InvalidOperationException("Wrong revision.");
var session=Editor.SceneEditorSession.Active;var playing=Game.IsPlaying;var game=Game.ActiveScene?.Id;var dirty=session.HasUnsavedChanges;
if(playing||dirty||!string.Equals(session.Scene.Source?.ResourcePath,"scenes/maps/ares/ares.scene",StringComparison.OrdinalIgnoreCase))throw new InvalidOperationException("Inspect changed scene/play state before importing.");
string Hash(string path)=>Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(System.IO.File.ReadAllBytes(path))).ToLowerInvariant();
foreach(var group in new[]{"source_sha256","protected_sha256"})
    foreach(var pair in manifest[group].AsObject())
        if(Hash(System.IO.Path.Combine(root,pair.Key))!=pair.Value.GetValue<string>())throw new InvalidOperationException("Shared input changed: "+pair.Key);
foreach(var pair in manifest["required_source_snippets"].AsObject())
{
    var source=System.IO.File.ReadAllText(System.IO.Path.Combine(root,pair.Key)).Replace("\r\n","\n");
    foreach(var part in pair.Value.AsArray())if(!source.Contains(part.GetValue<string>().Replace("\r\n","\n"),StringComparison.Ordinal))throw new InvalidOperationException("Regeneration hook changed: "+pair.Key);
}
var files=manifest["files"].AsObject().Where(p=>p.Value["phase"].GetValue<string>()=="assets").ToArray();
foreach(var pair in files)
{
    var path=pair.Key;
    if(path.Contains("..")||!(path.StartsWith("Assets/materials/ares/crate-v2/",StringComparison.Ordinal)||path.StartsWith("Assets/models/props/ares/crate-v2/",StringComparison.Ordinal)||path.StartsWith("Assets/models/maps/ares/crate-v2/",StringComparison.Ordinal)))throw new InvalidOperationException("Unexpected target: "+path);
    var item=pair.Value;var target=System.IO.Path.Combine(root,path);
    if(Hash(System.IO.Path.Combine(root,item["source"].GetValue<string>()))!=item["after"].GetValue<string>())throw new InvalidOperationException("Stage changed.");
    var current=System.IO.File.Exists(target)?Hash(target):null;
    if(current!=item["before"]?.GetValue<string>()&&current!=item["after"].GetValue<string>())throw new InvalidOperationException("Shared target changed: "+path);
}
var imported=new System.Collections.Generic.List<string>();
foreach(var pair in files)
{
    var item=pair.Value;var target=System.IO.Path.Combine(root,pair.Key);
    if(!System.IO.File.Exists(target)||Hash(target)!=item["after"].GetValue<string>())
    {
        System.IO.Directory.CreateDirectory(System.IO.Path.GetDirectoryName(target));var incoming=target+".crate-incoming";
        try{System.IO.File.Copy(System.IO.Path.Combine(root,item["source"].GetValue<string>()),incoming,true);System.IO.File.Move(incoming,target,true);}
        finally{if(System.IO.File.Exists(incoming))System.IO.File.Delete(incoming);}
        imported.Add(pair.Key);
    }
    Editor.AssetSystem.RegisterFile(target);
}
var preserved=ReferenceEquals(session,Editor.SceneEditorSession.Active)&&playing==Game.IsPlaying&&game==Game.ActiveScene?.Id&&dirty==session.HasUnsavedChanges;
var result=new{Stage=stage,ImportedFiles=imported,MapIntegrationApplied=false,EditorStatePreserved=preserved};
System.IO.File.WriteAllText(System.IO.Path.Combine(folder,"asset-import.json"),System.Text.Json.JsonSerializer.Serialize(result));
if(!preserved)throw new InvalidOperationException("Editor state changed during asset publication.");
return result;
