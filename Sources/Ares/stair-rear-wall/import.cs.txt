string Hash(string p)=>Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(System.IO.File.ReadAllBytes(p))).ToLowerInvariant();
foreach(var group in new[]{"protected","source_sha256"})
 foreach(var pair in manifest[group].AsObject())
  if(Hash(System.IO.Path.Combine(root,pair.Key))!=pair.Value.GetValue<string>())throw new InvalidOperationException("Shared source changed: "+pair.Key);
var files=manifest["files"].AsObject();
foreach(var pair in files)
{
 if(!pair.Key.StartsWith("Assets/models/maps/ares/stair-rear-wall/",StringComparison.Ordinal)||pair.Key.Contains(".."))throw new InvalidOperationException("Unexpected target.");
 var target=System.IO.Path.Combine(root,pair.Key);var source=System.IO.Path.Combine(root,pair.Value["source"].GetValue<string>());
 if(Hash(source)!=pair.Value["sha256"].GetValue<string>())throw new InvalidOperationException("Staged asset changed.");
 if(System.IO.File.Exists(target)&&Hash(target)!=Hash(source))throw new InvalidOperationException("Target already exists.");
}
foreach(var pair in files.OrderBy(p=>p.Key.EndsWith(".vmdl",StringComparison.Ordinal)?1:0))
{

 if(!System.IO.File.Exists(target))
 {

  var incoming=target+".rear-wall-incoming";

 }

}
