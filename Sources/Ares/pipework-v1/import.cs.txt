// Publish pipe geometry and private finishes. Only five material aliases are
// added to the current hall document; every other import and all physics remain.

var folder=System.IO.Path.Combine(root,"output/ares/pipework-v1");
var request=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder,"import-request.json")));
var stage=request["stage"].GetValue<string>();
var preview=request["preview"]?.GetValue<bool>()??true;
if(!System.Text.RegularExpressions.Regex.IsMatch(stage,"^stage-[a-z0-9-]+$"))throw new InvalidOperationException("Invalid stage.");
var manifest=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder,stage,"manifest.json")));

var session=Editor.SceneEditorSession.Active;var playing=Game.IsPlaying;var game=Game.ActiveScene?.Id;var dirty=session.HasUnsavedChanges;

string Hash(string p)=>Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(System.IO.File.ReadAllBytes(p))).ToLowerInvariant();
var sourceScene=System.IO.Path.Combine(root,"Assets/scenes/maps/Ares/ares.scene");var sceneHash=Hash(sourceScene);
foreach(var group in new[]{"source_sha256","protected_sha256"})
 foreach(var pair in manifest[group].AsObject())
 {
  if(pair.Key=="Assets/scenes/maps/Ares/ares.scene")continue;
  if(Hash(System.IO.Path.Combine(root,pair.Key))!=pair.Value.GetValue<string>())throw new InvalidOperationException("Shared input changed: "+pair.Key);
 }
foreach(var pair in manifest["required_source_snippets"].AsObject())
{
 var source=System.IO.File.ReadAllText(System.IO.Path.Combine(root,pair.Key)).Replace("\r\n","\n");

  if(!source.Contains(part.GetValue<string>().Replace("\r\n","\n"),StringComparison.Ordinal))throw new InvalidOperationException("Source hook changed.");
}
var files=manifest["files"].AsObject();var active=manifest["active_base"].GetValue<string>();
var activeMtl=manifest["active_mtl"]?.GetValue<string>();
if(!active.StartsWith("Assets/models/maps/ares/",StringComparison.Ordinal)||!active.EndsWith(".obj",StringComparison.Ordinal)||active.Contains(".."))throw new InvalidOperationException("Unexpected live mesh.");
var document=manifest["document"].GetValue<string>();var liveDocument=System.IO.Path.Combine(root,document);

var marker="_class = \"PhysicsShapeList\"";

if(!documentText.Contains("filename = \""+active.Substring("Assets/".Length)+"\"",StringComparison.Ordinal))throw new InvalidOperationException("The active hall import moved; restage the pipe edit.");
foreach(var pair in files)
{
 var path=pair.Key;
 if((path!=active&&path!=activeMtl&&!path.StartsWith("Assets/models/maps/ares/pipework-v1/",StringComparison.Ordinal)&&!path.StartsWith("Assets/materials/ares/pipework-v1/",StringComparison.Ordinal))||path.Contains(".."))throw new InvalidOperationException("Unexpected target.");
 var item=pair.Value;var target=System.IO.Path.Combine(root,path);var staged=System.IO.Path.Combine(root,item["source"].GetValue<string>());
 var before=item["before"]?.GetValue<string>();var after=item["after"].GetValue<string>();
 if(Hash(staged)!=after)throw new InvalidOperationException("Staged source changed.");
 if(before is not null&&Hash(System.IO.Path.Combine(root,item["backup"].GetValue<string>()))!=before)throw new InvalidOperationException("Backup changed.");
 var current=System.IO.File.Exists(target)?Hash(target):null;
 if(current!=before&&current!=after)throw new InvalidOperationException("Target changed: "+path);
}
if(!preview)
{
 var trial=manifest["trial_document"].GetValue<string>();var previewBase=manifest["preview_base"].GetValue<string>();
 if(Hash(System.IO.Path.Combine(root,previewBase))!=files[active]["after"].GetValue<string>())throw new InvalidOperationException("Trial mesh differs.");

 foreach(var pair in manifest["materials"].AsObject())

  var material=Material.Load(pair.Value.GetValue<string>());
  if(material is null||material.IsError)throw new InvalidOperationException("A pipe material did not compile.");
 }
}
var finalDocument=documentText;var aliasesAdded=0;
if(!preview&&activeMtl is not null)
{

 {
  var token="from = \""+pair.Key+".vmat\"";

  if(finalDocument.Contains(token,StringComparison.Ordinal))

 }

 if(finalDocument!=documentText)

  if(!System.IO.File.ReadAllBytes(liveDocument).SequenceEqual(documentBytes))throw new InvalidOperationException("Hall changed before the material patch.");

  var incoming=liveDocument+".pipework-mats-incoming";

  {
   System.IO.File.WriteAllText(incoming,finalDocument,new System.Text.UTF8Encoding(false));
   System.IO.File.Move(incoming,liveDocument,true);
  }
  finally{if(System.IO.File.Exists(incoming))System.IO.File.Delete(incoming);}
  // Extra aliases with the old mesh remain valid. Publish them before the

int Priority(string path)=>path.EndsWith(".png",StringComparison.Ordinal)?0:path.EndsWith(".mtl",StringComparison.Ordinal)?1:path.EndsWith(".obj",StringComparison.Ordinal)?2:path.EndsWith(".vmat",StringComparison.Ordinal)?3:4;

}
foreach(var pair in files.OrderBy(p=>Priority(p.Key)))

if(!preserved)throw new InvalidOperationException("Editor state changed; inspect completed import.");
return result;
