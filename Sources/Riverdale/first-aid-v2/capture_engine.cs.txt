// Before and after of Riverdale's first aid boxes, rendered by s&box: run with sbox-cli --call run_csharp '{"path":...}'.
// Loads riverdale.scene's visual roots into an isolated editor scene, so the open scene and play state are never touched and nothing is saved, with the player camera's post stack and exposure held still.
// The new cabinet is a shade bigger than the old box all round, so "after" stood it over the old one without cutting anything. Since the install the scene has its own, so both variants now draw the new one: installed_shots.cs.txt is the check for that. Writes <view>-<variant>.png into the work folder's engine/.
var projectRoot = Sandbox.Project.Current.GetRootPath();
var destination = System.IO.Path.Combine(projectRoot, "output/asset-improvement-20261030/first-aid/engine");
System.IO.Directory.CreateDirectory(destination);
var keep = new System.Collections.Generic.HashSet<string>(new[] { "Sun", "Sky", "Fog", "Look", "Riverdale", "Lights", "Reflection Probes", "Volumetric Fog", "Indirect Light" }, StringComparer.OrdinalIgnoreCase);
var root = System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(projectRoot, "Assets/scenes/maps/Riverdale/riverdale.scene"))).AsObject();
var objects = new System.Text.Json.Nodes.JsonArray();
foreach (var item in root["GameObjects"].AsArray())
    if (keep.Contains(item["Name"]?.GetValue<string>() ?? "")) objects.Add(item.DeepClone());
var prefab = System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(projectRoot, "Assets/prefabs/player/cameras/default_player_camera.prefab"))).AsObject();
var wanted = new[] { "Sandbox.CameraComponent", "Sandbox.Bloom", "Sandbox.Tonemapping", "Sandbox.Sharpen", "Sandbox.AmbientOcclusion", "Sandbox.Vignette", "Facepunch.PhotographicResponse" };
var components = new System.Text.Json.Nodes.JsonArray();
foreach (var component in prefab["RootObject"]["Components"].AsArray())
    if (wanted.Contains(component["__type"]?.GetValue<string>())) components.Add(component.DeepClone());
objects.Add(new System.Text.Json.Nodes.JsonObject { ["__guid"] = System.Text.Json.Nodes.JsonNode.Parse("\"" + Guid.NewGuid() + "\""), ["Name"] = "Capture camera", ["Enabled"] = true, ["Components"] = components });
root["GameObjects"] = objects;
if (root["SceneProperties"]?["NavMesh"] is System.Text.Json.Nodes.JsonObject nav) nav["Enabled"] = false;
if (root["SceneProperties"] is System.Text.Json.Nodes.JsonObject sceneProperties) sceneProperties["WantsSystemScene"] = false;
var model = Model.Load("models/maps/riverdale/first_aid_v2/first_aid.vmdl");
if (model is null || model.IsError) throw new InvalidOperationException("The new cabinet hasn't compiled.");
var scene = Scene.CreateEditorScene();
var flags = System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.Public | System.Reflection.BindingFlags.NonPublic;
var preRender = typeof(Scene).GetMethod("PreRender", flags, null, Type.EmptyTypes, null);
var preCameraRender = typeof(Scene).GetMethod("PreCameraRender", flags, null, Type.EmptyTypes, null);
try
{
    CameraComponent camera;
    var cabinets = new System.Collections.Generic.List<GameObject>();
    var info = new System.Collections.Generic.List<string>();
    using (scene.Push())
    {
        scene.Deserialize(root);
        var holder = scene.GetAllObjects(false).Single(g => g.Name == "Capture camera");
        camera = holder.Components.Get<CameraComponent>();
        camera.ZNear = 1;
        camera.ZFar = 30000;
        camera.IsMainCamera = true;
        foreach (var tonemap in scene.GetAllComponents<Tonemapping>())
        {
            var hold = ( tonemap.MinimumExposure + tonemap.MaximumExposure ) / 2f;
            info.Add(tonemap.GameObject.Name + " " + tonemap.MinimumExposure + ".." + tonemap.MaximumExposure);
            tonemap.AutoExposureEnabled = true;
            tonemap.MinimumExposure = tonemap.MaximumExposure = hold;
            tonemap.ExposureCompensation = 0;
            tonemap.Rate = 10;
        }
        foreach (var (at, yaw) in new[] { (new Vector3(985f, 1000f, 54f), -90f), (new Vector3(985f, -1000f, 54f), 90f), (new Vector3(2000f, 962f, 54f), 180f), (new Vector3(2000f, 95f, 54f), 180f) })
        {
            var go = new GameObject(false, "New cabinet");
            go.WorldPosition = at;
            go.WorldRotation = Rotation.FromYaw(yaw);
            go.Components.Create<ModelRenderer>().Model = model;
            cabinets.Add(go);
        }
        scene.EditorTick(.016f, .016f);
        preRender.Invoke(scene, null);
        preCameraRender.Invoke(scene, null);
    }
    var textureNames = new[] { "g_tColor", "g_tNormal", "g_tRoughness", "g_tAmbientOcclusion", "g_tMetalness", "SkyTexture" };
    foreach (var material in scene.GetAllComponents<ModelRenderer>().Where(r => r.Model is not null && r.Model != model).SelectMany(r => r.Model.Materials).Distinct())
        foreach (var texture in textureNames.Select(material.GetTexture).Append(material.FirstTexture).Where(t => t is not null).Distinct())
            texture.MarkUsed(1024);
    foreach (var material in model.Materials)
        foreach (var texture in textureNames.Select(material.GetTexture).Append(material.FirstTexture).Where(t => t is not null).Distinct())
            texture.MarkUsed(2048);
    for (var wait = 0; wait < 130; wait++)
    {
        using (scene.Push()) { scene.EditorTick(.016f, .016f); preRender.Invoke(scene, null); }
        await Task.Delay(33, cancellationToken);
    }
    var views = new (string name, Vector3 eye, Vector3 target, float fov)[]
    {
        ("front", new Vector3(985f, 962f, 63f), new Vector3(985f, 1000f, 62.6f), 60f),
        ("angle", new Vector3(1030f, 950f, 72f), new Vector3(985f, 1000f, 62f), 55f),
        ("close", new Vector3(998f, 972f, 60f), new Vector3(986f, 996f, 56f), 40f),
        ("walkup", new Vector3(1055f, 880f, 64f), new Vector3(985f, 1000f, 62f), 70f),
        ("far", new Vector3(1250f, 520f, 70f), new Vector3(985f, 1000f, 62f), 60f),
        ("north", new Vector3(1950f, 95f, 66f), new Vector3(2000f, 95f, 62f), 70f),
    };
    foreach (var variant in new[] { "before", "after" })
    {
        using (scene.Push()) foreach (var go in cabinets) go.Enabled = variant == "after";
        foreach (var view in views)
        {
            cancellationToken.ThrowIfCancellationRequested();
            using var bitmap = new Bitmap(1280, 720);
            for (var frame = 0; frame < 12; frame++)
            {
                using (scene.Push())
                {
                    camera.WorldPosition = view.eye;
                    camera.WorldRotation = Rotation.LookAt(view.target - view.eye);
                    camera.FieldOfView = view.fov;
                    scene.EditorTick(.016f, .016f);
                    preRender.Invoke(scene, null);
                    preCameraRender.Invoke(scene, null);
                    camera.RenderToBitmap(bitmap);
                }
                await Task.Delay(16, cancellationToken);
            }
            System.IO.File.WriteAllBytes(System.IO.Path.Combine(destination, $"{view.name}-{variant}.png"), bitmap.ToPng());
        }
    }
    return new { Game.IsPlaying, Info = info, Bounds = model.Bounds.Size.ToString(), Materials = model.Materials.Select(m => m.ResourcePath).ToArray() };
}
catch (Exception error)
{
    System.IO.File.WriteAllText(System.IO.Path.Combine(destination, "capture-error.txt"), error.ToString());
    throw;
}
finally { scene.Destroy(); }
