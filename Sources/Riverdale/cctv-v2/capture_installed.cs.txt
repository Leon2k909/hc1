// Verify the published camera models in a disposable Riverdale scene.
var rootPath=Sandbox.Project.Current.GetRootPath();
var folder=System.IO.Path.Combine(rootPath,"output/riverdale/cctv-v2");
var editor=Editor.SceneEditorSession.Active;
var before=new { Scene=editor.Scene.Id, Runtime=Game.ActiveScene?.Id, Game.IsPlaying };
var data=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(rootPath,"Assets/scenes/maps/riverdale/riverdale.scene"))).AsObject();
var visual=new System.Collections.Generic.HashSet<string>(StringComparer.OrdinalIgnoreCase) {
 "Sun","Sky","Fog","Look","Riverdale","Lights","Reflection Probes","Volumetric Fog","Indirect Light"
};
var roots=new System.Text.Json.Nodes.JsonArray();
foreach(var node in data["GameObjects"].AsArray())
 if(visual.Contains(node["Name"]?.GetValue<string>()??"")) roots.Add(node.DeepClone());
// Keep the hall and its existing lighting, avoiding the distant yards/backdrop.
void Prune(System.Text.Json.Nodes.JsonArray list) {
 foreach(var node in list.ToArray()) {
  var renderer=node["Components"]?.AsArray().FirstOrDefault(c=>c["__type"]?.GetValue<string>()=="Sandbox.ModelRenderer");
  if(renderer!=null) {
   var model=renderer["Model"];
   var path=model is System.Text.Json.Nodes.JsonObject?model["Path"]?.GetValue<string>():model?.GetValue<string>();
   if(path==null || !path.StartsWith("models/maps/riverdale/riverdale_hall",StringComparison.OrdinalIgnoreCase)) { list.Remove(node); continue; }
  }
  if(node["Children"] is System.Text.Json.Nodes.JsonArray children) Prune(children);
 }
}
Prune(roots);
data["GameObjects"]=roots;
data["SceneProperties"]["NavMesh"]["Enabled"]=false;
data["SceneProperties"]["WantsSystemScene"]=false;
var scene=Scene.CreateEditorScene();
var flags=System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.Public|System.Reflection.BindingFlags.NonPublic;
var preRender=typeof(Scene).GetMethod("PreRender",flags,null,Type.EmptyTypes,null);
var preCamera=typeof(Scene).GetMethod("PreCameraRender",flags,null,Type.EmptyTypes,null);
try {
 CameraComponent camera;
 using(scene.Push()) {
  scene.Deserialize(data);
  var holder=new GameObject(false,"Installed camera verification");
  camera=holder.Components.Create<CameraComponent>();
  camera.IsMainCamera=false; camera.ZNear=.5f; camera.ZFar=120000;
  camera.CustomSize=new Vector2(1400,1000);
  camera.WorldPosition=new Vector3(1306.3215f,602.0884f,256.4190f);
  camera.WorldRotation=Rotation.LookAt(new Vector3(1304.7216f,630.0163f,258.7750f)-camera.WorldPosition);
  camera.FieldOfView=62;
  var tone=holder.Components.Create<Tonemapping>();
  tone.Mode=Tonemapping.TonemappingMode.ACES; tone.AutoExposureEnabled=false; tone.ExposureCompensation=.65f;
  var ao=holder.Components.Create<AmbientOcclusion>();
  ao.Radius=48; ao.Intensity=.45f; ao.FalloffRange=.9f; ao.ThinCompensation=1;
  var bloom=holder.Components.Create<Bloom>(); bloom.Strength=.1f; bloom.Threshold=1.6f;
  holder.Enabled=true;
  scene.EditorTick(.016f,.016f); preRender.Invoke(scene,null); preCamera.Invoke(scene,null);
 }
 var renderers=scene.GetAllComponents<ModelRenderer>().ToArray();
 var detail=renderers.Single(r=>r.Model?.ResourcePath=="models/maps/riverdale/riverdale_hall_detail.vmdl");
 var fine=renderers.Single(r=>r.Model?.ResourcePath=="models/maps/riverdale/riverdale_hall_fine.vmdl");
 var cameraMaterials=detail.Model.Materials.Where(m=>m.ResourcePath.Contains("materials/riverdale/cctv_v2/")).Select(m=>m.ResourcePath).ToArray();
 if(cameraMaterials.Length!=6) throw new InvalidOperationException("Published model does not expose all six camera materials.");
 foreach(var material in renderers.Where(r=>r.Model!=null).SelectMany(r=>r.Model.Materials).Distinct())
  foreach(var texture in new[]{"g_tColor","g_tNormal","g_tRoughness","g_tAmbientOcclusion","g_tMetalness"}.Select(material.GetTexture).Append(material.FirstTexture).Where(t=>t!=null).Distinct()) texture.MarkUsed(1024);
 for(var frame=0;frame<30;frame++) {
  using(scene.Push()) { scene.EditorTick(.016f,.016f); preRender.Invoke(scene,null); }
  await Task.Delay(33,cancellationToken);
 }
 using var bitmap=new Bitmap(1400,1000);
 for(var frame=0;frame<10;frame++) {
  cancellationToken.ThrowIfCancellationRequested();
  using(scene.Push()) { scene.EditorTick(.016f,.016f); preRender.Invoke(scene,null); preCamera.Invoke(scene,null); camera.RenderToBitmap(bitmap,false); }
  await Task.Delay(16,cancellationToken);
 }
 var image=System.IO.Path.Combine(folder,"installed-hero.png");
 System.IO.File.WriteAllBytes(image,bitmap.ToPng());
 var errors=renderers.Where(r=>r.Model==null || r.Model.IsError).Select(r=>r.GameObject.Name).ToArray();
 var report=new {
  Installed=true, DetailPath=detail.Model.ResourcePath, FinePath=fine.Model.ResourcePath,
  CameraMaterials=cameraMaterials, ErrorModels=errors, Screenshot=image,
  UserStatePreserved=editor.Scene.Id==before.Scene && Game.ActiveScene?.Id==before.Runtime && Game.IsPlaying==before.IsPlaying
 };
 System.IO.File.WriteAllText(System.IO.Path.Combine(folder,"installed-native-verification.json"),System.Text.Json.JsonSerializer.Serialize(report,new System.Text.Json.JsonSerializerOptions{WriteIndented=true}));
 return report;
} finally { scene.Destroy(); }
