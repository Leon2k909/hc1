var editor=Editor.SceneEditorSession.Active;
if(!Game.IsPlaying||Game.ActiveScene?.Source?.ResourcePath!="scenes/menu.scene"||Networking.IsActive||Facepunch.GameNetworkManager.EditorLobby)
 throw new InvalidOperationException("Runtime check requires offline menu play, with editor lobby disabled");
var editId=editor.Scene.Id;var dirty=editor.HasUnsavedChanges;
var original=ResourceLibrary.Get<SceneFile>("scenes/menu.scene");
var map=ResourceLibrary.Get<SceneFile>("scenes/maps/reefhall/reefhall.scene");
var folder=System.IO.Path.Combine(Sandbox.Project.Current.GetRootPath(),"output/reefhall/pump-v1/runtime-before");
System.IO.Directory.CreateDirectory(folder);
var written=new System.Collections.Generic.List<string>();
try {
 Game.ActiveScene.Load(map);
 for(int i=0;i<100;i++){cancellationToken.ThrowIfCancellationRequested();await Task.Delay(25,cancellationToken);}
 var scene=Game.ActiveScene;
 if(scene.Source?.ResourcePath!=map.ResourcePath)throw new InvalidOperationException("Lowtide did not become the active runtime map");
 CameraComponent camera;
 using(scene.Push()) {
  var holder=new GameObject(true,"Runtime asset verification camera");
  camera=holder.Components.Create<CameraComponent>();camera.IsMainCamera=false;camera.ZNear=2;camera.ZFar=50000;camera.CustomSize=new Vector2(1600,900);
 }
 var views=new[]{new {Name="pump",Eye=new Vector3(1480,485,80),Target=new Vector3(1610,380,34),Fov=55f},new {Name="inlet",Eye=new Vector3(1520,275,64),Target=new Vector3(1610,375,32),Fov=60f}};
 foreach(var view in views) {
  camera.WorldPosition=view.Eye;camera.WorldRotation=Rotation.LookAt(view.Target-view.Eye);camera.FieldOfView=view.Fov;
  using var bitmap=new Bitmap(1600,900);
  for(int frame=0;frame<120;frame++) {
   cancellationToken.ThrowIfCancellationRequested();camera.RenderToBitmap(bitmap,false);await Task.Delay(20,cancellationToken);
  }
  var path=System.IO.Path.Combine(folder,view.Name+".png");System.IO.File.WriteAllBytes(path,bitmap.ToPng());written.Add(path);
 }
 var report=new { ActualPlayMode=Game.IsPlaying, ActiveRuntimeMap=scene.Source.ResourcePath, Networking.IsActive,
  PlayerCount=scene.GetAllComponents<Facepunch.PlayerPawn>().Count(),
  ErrorModels=scene.GetAllComponents<ModelRenderer>().Where(r=>r.Model?.IsError==true).Select(r=>r.GameObject.Name).ToArray(),Written=written };
 System.IO.File.WriteAllText(System.IO.Path.Combine(folder,"checked.json"),System.Text.Json.JsonSerializer.Serialize(report,new System.Text.Json.JsonSerializerOptions{WriteIndented=true}));
 return report;
} finally {
 Game.ActiveScene.Load(original);
 System.IO.File.WriteAllText(System.IO.Path.Combine(folder,"restored.json"),System.Text.Json.JsonSerializer.Serialize(new{MenuRestored=Game.ActiveScene.Source?.ResourcePath==original.ResourcePath,Game.IsPlaying,EditScenePreserved=editor.Scene.Id==editId&&editor.HasUnsavedChanges==dirty}));
}
