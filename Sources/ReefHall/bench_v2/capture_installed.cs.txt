// Reef Hall's benches and Lowtide's quay bollards as installed, rendered by s&box and tested with rays: run with sbox-cli --call run_csharp '{"path":...}'.
// Based on Sources/Boxyard/ibc_tote_v2/capture_installed.cs.txt. Loads the scene file's visual roots into an isolated editor scene, so the open scene and play state are never touched and nothing is saved.
// Reads output/asset-improvement-20261030/bench-bollard-check/config.json, which check_config.py writes. Writes <view>.png to its out folder.
var projectRoot = Sandbox.Project.Current.GetRootPath();
var folder = System.IO.Path.Combine(projectRoot, "output/asset-improvement-20261030/bench-bollard-check");
var configuration = System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder, "config.json"))).AsObject();
var destination = System.IO.Path.Combine(projectRoot, configuration["out"].GetValue<string>());
System.IO.Directory.CreateDirectory(destination);
var source = Editor.SceneEditorSession.Active?.Scene;
var before = new { Game.IsPlaying, Scene = source?.Name, SceneId = source?.Id, GameSceneId = Game.ActiveScene?.Id };
var visual = new System.Collections.Generic.HashSet<string>(configuration["visual"].AsArray().Select(x => x.GetValue<string>()), StringComparer.OrdinalIgnoreCase);
var root = System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(projectRoot, configuration["scene"].GetValue<string>()))).AsObject();
var objects = new System.Text.Json.Nodes.JsonArray();
foreach (var item in root["GameObjects"].AsArray())
    if (visual.Contains(item["Name"]?.GetValue<string>() ?? "")) objects.Add(item.DeepClone());
root["GameObjects"] = objects;
if (root["SceneProperties"]?["NavMesh"] is System.Text.Json.Nodes.JsonObject nav) nav["Enabled"] = false;
if (root["SceneProperties"] is System.Text.Json.Nodes.JsonObject sceneProperties) sceneProperties["WantsSystemScene"] = false;

var scene = Scene.CreateEditorScene();
var flags = System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.Public | System.Reflection.BindingFlags.NonPublic;
var preRender = typeof(Scene).GetMethod("PreRender", flags, null, Type.EmptyTypes, null) ?? throw new MissingMethodException("Scene.PreRender");
var preCameraRender = typeof(Scene).GetMethod("PreCameraRender", flags, null, Type.EmptyTypes, null) ?? throw new MissingMethodException("Scene.PreCameraRender");
Vector3 Vec(System.Text.Json.Nodes.JsonNode node) { var a = node.AsArray(); return new(a[0].GetValue<float>(), a[1].GetValue<float>(), a[2].GetValue<float>()); }
try
{
    CameraComponent camera;
    using (scene.Push())
    {
        scene.Deserialize(root);
        // Auto exposure wouldn't settle in a few frames, so every shot is held at one value.
        foreach (var tonemap in scene.GetAllComponents<Tonemapping>())
        {
            tonemap.AutoExposureEnabled = true;
            tonemap.MinimumExposure = tonemap.MaximumExposure = configuration["exposure"].GetValue<float>();
            tonemap.ExposureCompensation = 0;
            tonemap.Rate = 10;
        }
        var holder = new GameObject(false, "Capture camera");
        camera = holder.Components.Create<CameraComponent>();
        camera.ZNear = 1;
        camera.ZFar = 60000;
        holder.Enabled = true;
        scene.EditorTick(.016f, .016f);
        preRender.Invoke(scene, null);
        preCameraRender.Invoke(scene, null);
    }

    var wanted = configuration["models"].AsArray().Select(x => x.GetValue<string>()).ToArray();
    bool IsNew(ModelRenderer r) => r.Model is not null && wanted.Any(w => string.Equals(r.Model.ResourcePath, w, StringComparison.OrdinalIgnoreCase));
    // Nothing above 1024 for the map or 2048 for the props: the editor is shared, and a whole map at full size has run it out of video memory.
    var textureNames = new[] { "g_tColor", "g_tNormal", "g_tRoughness", "g_tAmbientOcclusion", "SkyTexture" };
    var renderers = scene.GetAllComponents<ModelRenderer>().ToArray();
    foreach (var material in renderers.Where(r => r.Model is not null && !IsNew(r)).SelectMany(r => r.Model.Materials).Distinct())
        foreach (var texture in textureNames.Select(material.GetTexture).Append(material.FirstTexture).Where(t => t is not null).Distinct())
            texture.MarkUsed(1024);
    foreach (var material in renderers.Where(IsNew).SelectMany(r => r.Model.Materials).Distinct())
        foreach (var texture in textureNames.Select(material.GetTexture).Append(material.FirstTexture).Where(t => t is not null).Distinct())
            texture.MarkUsed(2048);
    for (var wait = 0; wait < configuration["stream_frames"].GetValue<int>(); wait++)
    {
        using (scene.Push()) { scene.EditorTick(.016f, .016f); preRender.Invoke(scene, null); }
        await Task.Delay(33, cancellationToken);
    }

    var rays = new System.Collections.Generic.List<object>();
    using (scene.Push())
    {
        foreach (var ray in configuration["rays"].AsArray())
        {
            var from = Vec(ray["from"]);
            var to = Vec(ray["to"]);
            var hit = scene.Trace.Ray(from, to).Run();
            rays.Add(new { Label = ray["label"].GetValue<string>(), Hit = hit.Hit, Distance = hit.Hit ? MathF.Round(hit.Distance, 2) : -1f, Expect = ray["expect"].GetValue<float>(),
                Object = hit.GameObject?.Name, Surface = hit.Surface?.ResourceName });
        }
    }

    var settle = configuration["settle_frames"].GetValue<int>();
    var written = new System.Collections.Generic.List<string>();
    foreach (var view in configuration["views"].AsArray())
    {
        cancellationToken.ThrowIfCancellationRequested();
        var name = view["name"].GetValue<string>();
        using var bitmap = new Bitmap(configuration["width"].GetValue<int>(), configuration["height"].GetValue<int>());
        var eye = Vec(view["eye"]);
        var target = Vec(view["target"]);
        for (var frame = 0; frame < settle; frame++)
        {
            using (scene.Push())
            {
                camera.WorldPosition = eye;
                camera.WorldRotation = Rotation.LookAt(target - eye);
                camera.FieldOfView = view["fov"].GetValue<float>();
                scene.EditorTick(.016f, .016f);
                preRender.Invoke(scene, null);
                preCameraRender.Invoke(scene, null);
                camera.RenderToBitmap(bitmap);
            }
            await Task.Delay(16, cancellationToken);
        }
        System.IO.File.WriteAllBytes(System.IO.Path.Combine(destination, name + ".png"), bitmap.ToPng());
        written.Add(name);
    }
    return new
    {
        Before = before, Game.IsPlaying,
        SourceUnchanged = Editor.SceneEditorSession.Active?.Scene?.Id == before.SceneId && Game.ActiveScene?.Id == before.GameSceneId,
        Written = written, Renderers = renderers.Length,
        ErrorModels = renderers.Where(r => r.Model is null || r.Model.IsError).Select(r => r.GameObject.Name).ToArray(),
        Placed = renderers.Where(IsNew).Select(r => new { r.GameObject.Name, Model = r.Model.ResourcePath, Position = r.WorldPosition.ToString(), Scale = r.WorldScale.ToString(), Colliders = r.GameObject.Components.GetAll<Collider>().Count(), Bounds = r.Model.Bounds.Size.ToString(), Materials = r.Model.Materials.Select(m => m.ResourcePath).ToArray() }).ToArray(),
        Rays = rays
    };
}
catch (Exception error)
{
    System.IO.File.WriteAllText(System.IO.Path.Combine(folder, "capture-error.txt"), error.ToString());
    throw;
}
finally { scene.Destroy(); }
