// Spillway's offscreen views: run with the native s&box MCP's run_csharp (sbox-cli --call run_csharp '{"path":...}').
// Loads Assets/scenes/maps/sewage/sewage.scene's visual roots into an isolated editor scene, renders the views
// listed in output/sewage/crane-blocks-v1/demonstration/config.json (all when it has none) to PNGs there, then destroys that scene. The
// user's open scene and play state are never touched.
var projectRoot = Sandbox.Project.Current.GetRootPath();
var destination = System.IO.Path.Combine(projectRoot, "output/sewage/crane-blocks-v1/demonstration");
System.IO.Directory.CreateDirectory(destination);
var configPath = System.IO.Path.Combine(destination, "config.json");
var configuration = System.IO.File.Exists(configPath)
    ? System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(configPath)).AsObject()
    : new System.Text.Json.Nodes.JsonObject();
var source = Editor.SceneEditorSession.Active?.Scene;
var before = new { Game.IsPlaying, Scene = source?.Name, SceneId = source?.Id, GameSceneId = Game.ActiveScene?.Id };
var visual = new System.Collections.Generic.HashSet<string>(StringComparer.OrdinalIgnoreCase) {"Waterworks shell","Waterworks tower","Waterworks catwalks","Waterworks pipework","Waterworks details","Power cabinet 1","Power cabinet 2","Power cabinet 3","Waterworks signage","Waterworks water","Waterworks cascades","Practical work lights","Chamber atmosphere","Chamber reflection 0","Chamber reflection 1","Chamber reflection 2","Skylight shaft 0","Skylight shaft 1","Skylight shaft 2"};
var root = System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(projectRoot, "Assets/scenes/maps/sewage/sewage.scene"))).AsObject();
var objects = new System.Text.Json.Nodes.JsonArray();
foreach (var item in root["GameObjects"].AsArray())
    if (visual.Contains(item["Name"]?.GetValue<string>() ?? "")) objects.Add(item.DeepClone());
root["GameObjects"] = objects;
root["SceneProperties"]["NavMesh"]["Enabled"] = false;
root["SceneProperties"]["WantsSystemScene"] = false;
var scene = Scene.CreateEditorScene();
var flags = System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.Public | System.Reflection.BindingFlags.NonPublic;
var preRender = typeof(Scene).GetMethod("PreRender", flags, null, Type.EmptyTypes, null) ?? throw new MissingMethodException("Scene.PreRender");
var preCameraRender = typeof(Scene).GetMethod("PreCameraRender", flags, null, Type.EmptyTypes, null) ?? throw new MissingMethodException("Scene.PreCameraRender");
try {
    CameraComponent camera;
    var hidden = new System.Collections.Generic.List<string>();
    using (scene.Push()) {
        scene.Deserialize(root);
        foreach(var item in new[]{new Vector3(600,-300,900),new Vector3(730,400,850)}) {
            var fill=new GameObject(true,"Temporary inspection fill");
            fill.WorldPosition=new Vector3(item.y,-item.x,item.z);
            var light=fill.Components.Create<PointLight>();light.LightColor=new Color(5f,4.8f,4.4f);
            light.Radius=1200;light.Shadows=false;
        }

        // "hide": objects to leave out of the views, by name (to see through the pool's water, say).
        if (configuration["hide"] is System.Text.Json.Nodes.JsonArray hide)
            foreach (var name in hide.Select(x => x.GetValue<string>()))
                foreach (var go in scene.GetAllObjects(false).Where(g => g.Name == name).ToArray()) { go.Enabled = false; hidden.Add(go.Name); }
        var holder = new GameObject(false, "Spillway capture camera");
        camera = holder.Components.Create<CameraComponent>();
        camera.CustomSize = new Vector2(configuration["width"]?.GetValue<int>() ?? 1600, configuration["height"]?.GetValue<int>() ?? 1000);
        camera.FovAxis=CameraComponent.Axis.Horizontal;camera.EnablePostProcessing=true;
        camera.ZNear = 4; camera.ZFar = 120000;
        holder.Enabled = true;
        scene.EditorTick(.016f, .016f);
        preRender.Invoke(scene, null);
        preCameraRender.Invoke(scene, null);
    }
    // Textures stream in: a render straight after loading samples low mips, so the lettering reads as blobs. Ask for
    // every map at full size, then give the streamer a few seconds.
    var marked = 0;
    var textureNames = new[] { "g_tColor", "g_tNormal", "g_tRoughness", "g_tSelfIllumMask", "g_tTransmissiveColor", "g_tAmbientOcclusion" };
    foreach (var material in scene.GetAllComponents<ModelRenderer>().Where(r => r.Model is not null).SelectMany(r => r.Model.Materials).Distinct())
        foreach (var texture in textureNames.Select(material.GetTexture).Append(material.FirstTexture).Where(t => t is not null).Distinct())
        { texture.MarkUsed(4096); marked++; }
    for (var wait = 0; wait < (configuration["stream_frames"]?.GetValue<int>() ?? 90); wait++)
    {
        using (scene.Push()) { scene.EditorTick(.016f, .016f); preRender.Invoke(scene, null); }
        await Task.Delay(33, cancellationToken);
    }
    // Plan ( e, n, z ) to world ( n, -e, z ).
    Vector3 World(Vector3 plan) => new(plan.y, -plan.x, plan.z);
    var views = new[] {
        (Name: "location", Eye: new Vector3(700f,-850f,520f), Target: new Vector3(470f,240f,670f), Fov: 85f),
        (Name: "mounting", Eye: new Vector3(800f,-430f,930f), Target: new Vector3(610f,-20f,840f), Fov: 70f),
        (Name: "catwalk", Eye: new Vector3(860f,-320f,608f), Target: new Vector3(610f,-20f,780f), Fov: 78f),
    };
    var wanted = configuration["views"] is System.Text.Json.Nodes.JsonArray list
        ? list.Select(x => x.GetValue<string>()).ToHashSet() : views.Select(x => x.Name).ToHashSet();
    using var bitmap = new Bitmap(configuration["width"]?.GetValue<int>() ?? 1600, configuration["height"]?.GetValue<int>() ?? 900);
    var settle = configuration["settle_frames"]?.GetValue<int>() ?? 20;
    foreach (var view in views.Where(x => wanted.Contains(x.Name))) {
        cancellationToken.ThrowIfCancellationRequested();
        foreach(var light in scene.GetAllComponents<PointLight>().Where(l=>l.GameObject.Name=="Temporary inspection fill"))light.GameObject.Enabled=view.Name!="catwalk";
        for (var frame = 0; frame < settle; frame++)
        {
            using (scene.Push()) {
                camera.WorldPosition = World(view.Eye);
                camera.WorldRotation = Rotation.LookAt(World(view.Target) - World(view.Eye));
                camera.FieldOfView = view.Fov;
                scene.EditorTick(.016f, .016f);
                preRender.Invoke(scene, null);
                preCameraRender.Invoke(scene, null);
                camera.CustomSize = new Vector2(configuration["width"]?.GetValue<int>() ?? 1600, configuration["height"]?.GetValue<int>() ?? 1000);
                camera.RenderToBitmap(bitmap);
            }
            await Task.Delay(16, cancellationToken);
        }
        System.IO.File.WriteAllBytes(System.IO.Path.Combine(destination, view.Name + "-" + configuration["phase"].GetValue<string>() + ".png"), bitmap.ToPng());
    }
    var missing = scene.GetAllComponents<ModelRenderer>().Where(r => r.Model is null || r.Model.IsError).Select(r => r.GameObject.Name).ToArray();
    return new { Before = before, Game.IsPlaying,
        SourceUnchanged = Editor.SceneEditorSession.Active?.Scene?.Id == before.SceneId && Game.ActiveScene?.Id == before.GameSceneId,
        Directory = destination, Renderers = scene.GetAllComponents<ModelRenderer>().Count(), ErrorModels = missing,
        Lights = scene.GetAllComponents<PointLight>().Count(), Probes = scene.GetAllComponents<EnvmapProbe>().Count(), MarkedTextures = marked, Hidden = hidden, Objects = scene.GetAllObjects(false).Where(g => g.Parent is not null && g.Parent.Name == "Sewage").Select(g => g.Name + (g.Enabled ? "" : " (off)")).ToArray() };
} catch (Exception error) {
    System.IO.File.WriteAllText(System.IO.Path.Combine(destination, "capture-error.txt"), error.ToString());
    throw;
} finally { scene.Destroy(); }
