var model=Model.Load("models/maps/sewage/sewage_details.vmdl");
if(model.IsError)throw new InvalidOperationException("Public details model failed");
var paths=model.Materials.Select(m=>m.ResourcePath).ToArray();
foreach(var mat in new[]{"paint","steel","dark","label","stripes"})if(!paths.Contains("materials/sewage/crane-blocks-v1/"+mat+".vmat"))throw new InvalidOperationException("Missing crane material: "+mat);
var old=Model.Load("models/maps/sewage/crane-blocks-v1/before.vmdl");
if(old.IsError)throw new InvalidOperationException("Old alternate unavailable");
var report=new{Model=model.ResourcePath,Error=model.IsError,CraneMaterials=5,OldAlternate=old.ResourcePath};
System.IO.File.WriteAllText(System.IO.Path.Combine(Sandbox.Project.Current.GetRootPath(),"output/sewage/crane-blocks-v1/public-resource.json"),System.Text.Json.JsonSerializer.Serialize(report));
return report;
