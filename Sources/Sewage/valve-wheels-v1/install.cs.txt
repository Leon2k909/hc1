if(Game.IsPlaying) throw new Exception("Wait for edit mode before installing valve wheels.");
var root=Sandbox.Project.Current.GetRootPath();
var plan=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(root,"output/sewage/valve-wheels-v1/plan.json")));
var hash=Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(System.IO.File.ReadAllBytes(System.IO.Path.Combine(root,"Assets/models/maps/sewage/sewage_pipework.obj")))).ToLowerInvariant();
if(hash!=plan["original_sha256"].GetValue<string>()) throw new Exception("Original pipework changed; rebuild the targeted splice before installing.");
var session=Editor.SceneEditorSession.All.Single(x=>x.Scene.Name=="sewage");
var pipework=session.Scene.GetAllObjects(true).Single(x=>x.Name=="Waterworks pipework").Components.Get<ModelRenderer>();
if(pipework.Model?.ResourcePath!="models/maps/sewage/sewage_pipework.vmdl") throw new Exception("Unexpected pipework model; inspect before installing.");
if(session.Scene.GetAllObjects(true).Any(x=>x.Name=="Waterworks valve wheels")) throw new Exception("Valve wheels are already present.");
var body=Model.Load("models/maps/sewage/valve-wheels-v1/base_pipework.vmdl");
var wheels=Model.Load("models/maps/sewage/valve-wheels-v1/wheels.vmdl");
if(body is null || body.IsError || wheels is null || wheels.IsError) throw new Exception("Valve preview models failed to load.");
using(session.Scene.Push())
{
    var holder=new GameObject(false,"Waterworks valve wheels");
    holder.Tags.Add("sewage");
    holder.Components.Create<ModelRenderer>().Model=wheels;
    pipework.Model=body;
    holder.Enabled=true;
}
session.HasUnsavedChanges=true;
session.Save(false);
var disk=System.IO.File.ReadAllText(System.IO.Path.Combine(root,"Assets/scenes/maps/sewage/sewage.scene"));
return new {Saved=!session.HasUnsavedChanges,OnDisk=disk.Contains("Waterworks valve wheels")&&disk.Contains("models/maps/sewage/valve-wheels-v1/base_pipework.vmdl"),ActiveScene=Editor.SceneEditorSession.Active.Scene.Name};
