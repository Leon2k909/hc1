var project=Sandbox.Project.Current.GetRootPath();
var output=System.IO.Path.Combine(project,"output/sewage/warning-sign-v1");
var editor=Editor.SceneEditorSession.Active;
var state=new{Scene=editor.Scene.Id,Runtime=Game.ActiveScene?.Id,Game.IsPlaying,Dirty=editor.HasUnsavedChanges};

Vector3 W(float e,float n,float z)=>new(n,-e,z);
var scene=Scene.CreateEditorScene();var report=new System.Collections.Generic.List<object>();
try {
 CameraComponent camera;ModelRenderer signage;
 using(scene.Push()) {
  scene.Deserialize(root);
  signage=scene.GetAllObjects(true).Single(g=>g.Name=="Waterworks signage").Components.Get<ModelRenderer>();
  var holder=new GameObject(false,"Warning sign review camera");camera=holder.Components.Create<CameraComponent>();
  camera.IsMainCamera=false;camera.EnablePostProcessing=true;camera.FovAxis=CameraComponent.Axis.Horizontal;camera.CustomSize=new Vector2(1440,1000);camera.ZNear=1;camera.ZFar=20000;holder.Enabled=true;
 }
 foreach(var mat in scene.GetAllComponents<ModelRenderer>().Where(r=>r.Model!=null).SelectMany(r=>r.Model.Materials).Distinct())
  foreach(var tex in new[]{"g_tColor","g_tNormal","g_tRoughness","g_tSelfIllumMask"}.Select(mat.GetTexture).Append(mat.FirstTexture).Where(t=>t!=null).Distinct())tex.MarkUsed(4096);
 float time=0;
 var views=new[]{(Name:"close",Eye:W(-790,-510,95),Target:W(-615,-400,35),Fov:43f),(Name:"rear",Eye:W(-510,-310,80),Target:W(-611,-400,35),Fov:51f)};
 foreach(var view in views)foreach(var variant in new[]{"before","after"}) {
  using(scene.Push()) {
   signage.Model=Model.Load(variant=="after"?"models/maps/sewage/sewage_signage.vmdl":"models/maps/sewage/warning-sign-v1/before.vmdl");
   if(signage.Model==null||signage.Model.IsError)throw new Exception("Failed model: "+variant);
   camera.WorldPosition=view.Eye;camera.WorldRotation=Rotation.LookAt(view.Target-view.Eye);camera.FieldOfView=view.Fov;
   foreach(var mat in signage.Model.Materials)foreach(var tex in new[]{"g_tColor","g_tNormal","g_tRoughness"}.Select(mat.GetTexture).Append(mat.FirstTexture).Where(t=>t!=null).Distinct())tex.MarkUsed(2048);
  }
  using var bitmap=new Bitmap(1440,1000);
  for(var frame=0;frame<45;frame++) {
   cancellationToken.ThrowIfCancellationRequested();
   using(scene.Push()){scene.EditorTick(time+=.016f,.016f);preRender.Invoke(scene,null);preCamera.Invoke(scene,null);camera.RenderToBitmap(bitmap,false);}
   await Task.Delay(20,cancellationToken);
  }
  var path=System.IO.Path.Combine(output,view.Name+"-"+variant+".png");System.IO.File.WriteAllBytes(path,bitmap.ToPng());
  report.Add(new{File=path,Variant=variant,View=view.Name,Model=signage.Model.ResourcePath,Camera=camera.WorldPosition.ToString(),Rotation=camera.WorldRotation.ToString(),camera.FieldOfView});
 }
}finally{scene.Destroy();}
var result=new{Captures=report,MatchedCameraAndLighting=true,UserStatePreserved=editor.Scene.Id==state.Scene&&Game.ActiveScene?.Id==state.Runtime&&Game.IsPlaying==state.IsPlaying&&editor.HasUnsavedChanges==state.Dirty};
