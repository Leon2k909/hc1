var output = System.IO.Path.Combine(Sandbox.Project.Current.GetRootPath(), "output/sewage/work-lamps-v1/current-review");
var material = Material.Load("materials/sewage/pipe-texture-v1/pipe.vmat");
if (material is null || material.IsError) throw new Exception("Pipe material has not compiled successfully.");
var root = System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(Sandbox.Project.Current.GetRootPath(), "Assets/scenes/maps/sewage/sewage.scene"))).AsObject();
var objects = root["GameObjects"].AsArray();
foreach (var item in objects.ToArray())
{
    var name = item["Name"]?.GetValue<string>() ?? "";
    if (name == "zombies" || name == "game_setup" || name.Contains("spawn", StringComparison.OrdinalIgnoreCase)) objects.Remove(item);
}
if (root["SceneProperties"] is System.Text.Json.Nodes.JsonObject props)
{
    props["WantsSystemScene"] = false;
    if (props["NavMesh"] is System.Text.Json.Nodes.JsonObject nav) nav["Enabled"] = false;
}
var editor=Editor.SceneEditorSession.Active;
var savedState=new{Scene=editor.Scene.Id,Runtime=Game.ActiveScene?.Id,Game.IsPlaying,Dirty=editor.HasUnsavedChanges};
var scene = Scene.CreateEditorScene();
try
{
    using (scene.Push()) scene.Deserialize(root);
    var renderer = scene.GetAllObjects(true).Single(x => x.Name == "Waterworks details").Components.Get<ModelRenderer>();
    GameObject added;
    using (scene.Push())
    {
        added = new GameObject(false, "Existing caged work lamp preview");
        var flangeModel = Model.Load("models/maps/sewage/work-lamps-v1/lamps.vmdl");
        if (flangeModel is null || flangeModel.IsError) throw new Exception("Lamp model failed to compile.");
        added.Components.Create<ModelRenderer>().Model = flangeModel;
    }
    var flags = System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Public;
    var preRender = typeof(Scene).GetMethod("PreRender", flags, null, Type.EmptyTypes, null);
    var preCameraRender = typeof(Scene).GetMethod("PreCameraRender", flags, null, Type.EmptyTypes, null);
    CameraComponent camera;
    using (scene.Push())
    {
        var holder = new GameObject(false, "Caged work lamp review camera");
        camera = holder.Components.Create<CameraComponent>();
        camera.IsMainCamera = false;
        camera.ZNear = 1;
        camera.ZFar = 20000;
        camera.WorldPosition = new Vector3(-200f,620f,225f);
        camera.WorldRotation = Rotation.LookAt(new Vector3(-110f,680f,205f)-camera.WorldPosition);
        camera.FieldOfView = 42;
        holder.Enabled = true;
    }
    foreach (var variant in new[] { "before", "after" })
    {
        using (scene.Push())
        {
            added.Enabled = variant == "after";
            renderer.Model = Model.Load(variant == "after" ? "models/maps/sewage/work-lamps-v1/base_details.vmdl" : "models/maps/sewage/sewage_details.vmdl");
            if(renderer.Model.IsError) throw new Exception("Details variant failed to compile.");
        }
        using var bitmap = new Bitmap(1280,720);
        for (var frame = 0; frame < 24; frame++)
        {
            cancellationToken.ThrowIfCancellationRequested();
            using (scene.Push())
            {
                scene.EditorTick(.016f,.016f);
                preRender.Invoke(scene,null);
                preCameraRender.Invoke(scene,null);
                camera.RenderToBitmap(bitmap,false);
            }
            await Task.Delay(16,cancellationToken);
        }
        System.IO.File.WriteAllBytes(System.IO.Path.Combine(output,variant+".png"),bitmap.ToPng());
    }
    var result=new{Success=true,ReusedExistingModel=true,MatchedCameraAndLighting=true,UserStatePreserved=editor.Scene.Id==savedState.Scene&&Game.ActiveScene?.Id==savedState.Runtime&&Game.IsPlaying==savedState.IsPlaying&&editor.HasUnsavedChanges==savedState.Dirty,Output=output};
System.IO.File.WriteAllText(System.IO.Path.Combine(output,"capture-report.json"),System.Text.Json.JsonSerializer.Serialize(result));return result;
}
finally { scene.Destroy(); }
