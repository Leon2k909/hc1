var root=Sandbox.Project.Current.GetRootPath();
var stage=System.IO.Path.Combine(root,"output/sewage/work-lamps-v1/stage/Assets/models/maps/sewage/work-lamps-v1");
var target=System.IO.Path.Combine(root,"Assets/models/maps/sewage/work-lamps-v1");
System.IO.Directory.CreateDirectory(target);
var written=new System.Collections.Generic.List<string>();
foreach(var source in System.IO.Directory.GetFiles(stage).Where(x=>new[]{".obj",".mtl",".vmdl"}.Contains(System.IO.Path.GetExtension(x))))
{
    var path=System.IO.Path.Combine(target,System.IO.Path.GetFileName(source));
    System.IO.File.Copy(source,path,true);
    Editor.AssetSystem.RegisterFile(path);
    written.Add(path);
}
var incomplete=System.IO.Path.Combine(target,"lamps.vmdl.writing");
if(System.IO.File.Exists(incomplete)) System.IO.File.Delete(incomplete);
return new { Imported=written, SceneChanged=false };
