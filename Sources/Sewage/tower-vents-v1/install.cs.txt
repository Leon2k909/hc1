var root=Sandbox.Project.Current.GetRootPath();
var plan=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(root,"output/sewage/tower-vents-v1/plan.json")));
var hash=Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(System.IO.File.ReadAllBytes(System.IO.Path.Combine(root,"Assets/models/maps/sewage/sewage_tower.obj")))).ToLowerInvariant();
if(hash!=plan["original_sha256"].GetValue<string>())throw new Exception("Tower changed; rebuild the targeted vent splice first.");
var session=Editor.SceneEditorSession.All.SingleOrDefault(x=>x.Scene.Name=="sewage");
if(session is null)throw new Exception("Open the Sewage edit scene before installing.");
if(Game.IsPlaying&&Editor.SceneEditorSession.Active==session)throw new Exception("Wait for edit mode before installing into the active play scene.");
var tower=session.Scene.GetAllObjects(true).Single(x=>x.Name=="Waterworks tower").Components.Get<ModelRenderer>();
if(tower.Model?.ResourcePath!="models/maps/sewage/sewage_tower.vmdl")throw new Exception("Unexpected tower model; inspect before installing.");
if(session.Scene.GetAllObjects(true).Any(x=>x.Name=="Waterworks louvred vent housings"))throw new Exception("Vent housings already exist.");
var body=Model.Load("models/maps/sewage/tower-vents-v1/base_tower.vmdl");
var vents=Model.Load("models/maps/sewage/tower-vents-v1/vents.vmdl");
if(body is null||body.IsError||vents is null||vents.IsError)throw new Exception("Vent models failed to load.");
using(session.Scene.Push())
{
    var holder=new GameObject(false,"Waterworks louvred vent housings");
    holder.Tags.Add("sewage");
    holder.Components.Create<ModelRenderer>().Model=vents;
    tower.Model=body;
    holder.Enabled=true;
}
session.HasUnsavedChanges=true;
session.Save(false);
var disk=System.IO.File.ReadAllText(System.IO.Path.Combine(root,"Assets/scenes/maps/sewage/sewage.scene"));
return new{Saved=!session.HasUnsavedChanges,OnDisk=disk.Contains("Waterworks louvred vent housings")&&disk.Contains("models/maps/sewage/tower-vents-v1/base_tower.vmdl"),OldAssetRetained=System.IO.File.Exists(System.IO.Path.Combine(root,"Assets/models/maps/sewage/sewage_tower.vmdl")),ActiveScene=Editor.SceneEditorSession.Active.Scene.Name};
