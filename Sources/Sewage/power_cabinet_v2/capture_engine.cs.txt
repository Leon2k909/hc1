// Before and after of Sewage's power cabinets, rendered by s&box: run with sbox-cli --call run_csharp '{"path":...}'.
// Loads sewage.scene's visual objects into an isolated editor scene, so the open scene and play state are never touched and nothing is saved.
// "Before" is the map as it is. "After" draws sewage_details from its OBJ, built here at run time without the old cabinets' faces (so no big
// map model is compiled), stands the new model on each cabinet's spot and moves each cabinet's green light to the new status lamp.
// Reads config.json from the work folder; writes <view>-before.png and <view>-after.png beside it.
var projectRoot = Sandbox.Project.Current.GetRootPath();
var destination = System.IO.Path.Combine(projectRoot, "output/asset-improvement-20261030/sewage-power-cabinet/engine");
var configuration = System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(destination, "config.json"))).AsObject();
var source = Editor.SceneEditorSession.Active?.Scene;
var before = new { Game.IsPlaying, Scene = source?.Name, SceneId = source?.Id, GameSceneId = Game.ActiveScene?.Id };
Vector3 Vec(System.Text.Json.Nodes.JsonNode node) { var a = node.AsArray(); return new(a[0].GetValue<float>(), a[1].GetValue<float>(), a[2].GetValue<float>()); }

var root = System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(projectRoot, "Assets/scenes/maps/sewage/sewage.scene"))).AsObject();
var objects = new System.Text.Json.Nodes.JsonArray();
foreach (var item in root["GameObjects"].AsArray())
{
    var name = item["Name"]?.GetValue<string>() ?? "";
    if (name.StartsWith("Co-op") || name.StartsWith("Reference") || name.StartsWith("Sewage map") || name == "") continue;
    objects.Add(item.DeepClone());
}
root["GameObjects"] = objects;
if (root["SceneProperties"]?["NavMesh"] is System.Text.Json.Nodes.JsonObject nav) nav["Enabled"] = false;
if (root["SceneProperties"] is System.Text.Json.Nodes.JsonObject sceneProperties) sceneProperties["WantsSystemScene"] = false;

var cabinets = configuration["cabinets"].AsArray();
var model = Model.Load(configuration["model"].GetValue<string>());
if (model is null || model.IsError) throw new InvalidOperationException("The new cabinet hasn't compiled.");

// sewage_details without the old cabinets. The OBJ's axes are the world's ( y, z, x ).
bool InCut(Vector3 p)
{
    foreach (var cabinet in cabinets)
    {
        var at = Vec(cabinet["at"]);
        var plinth = Math.Abs(p.x - at.x) <= 38.2f && Math.Abs(p.y - at.y) <= 19.2f && p.z >= at.z - 0.1f && p.z <= at.z + 115.1f;
        // The old lamp sticks out of its south end: plan n - 40 is world x - 40.
        var lamp = p.x >= at.x - 46.9f && p.x <= at.x - 34.8f && Math.Abs(p.y - at.y) <= 10.1f && p.z >= at.z + 98.2f && p.z <= at.z + 109.1f;
        if (plinth || lamp) return true;
    }
    return false;
}
var positions = new System.Collections.Generic.List<Vector3>();
var uvs = new System.Collections.Generic.List<Vector2>();
var normals = new System.Collections.Generic.List<Vector3>();
var meshes = new System.Collections.Generic.List<Mesh>();
var kept = 0; var cut = 0;
{
    var vertices = new System.Collections.Generic.List<Vertex>();
    string currentMaterial = null;
    void Flush()
    {
        if (currentMaterial is null || vertices.Count == 0) return;
        var material = Material.Load($"materials/sewage/{currentMaterial}.vmat");
        var mesh = new Mesh(material);
        mesh.CreateVertexBuffer(vertices.Count, Vertex.Layout, vertices);
        mesh.Bounds = BBox.FromPoints(vertices.Select(v => v.Position));
        meshes.Add(mesh);
        vertices.Clear();
    }
    int[] Corner(string token) { var bits = token.Split('/'); return new[] { int.Parse(bits[0]) - 1, int.Parse(bits[1]) - 1, int.Parse(bits[2]) - 1 }; }
    foreach (var line in System.IO.File.ReadLines(System.IO.Path.Combine(projectRoot, "Assets/models/maps/sewage/sewage_details.obj")))
    {
        if (line.StartsWith("v "))
        {
            var bits = line.Split(' ');
            positions.Add(new Vector3(float.Parse(bits[3]), float.Parse(bits[1]), float.Parse(bits[2])));
        }
        else if (line.StartsWith("vt "))
        {
            var bits = line.Split(' ');
            uvs.Add(new Vector2(float.Parse(bits[1]), 1 - float.Parse(bits[2])));
        }
        else if (line.StartsWith("vn "))
        {
            var bits = line.Split(' ');
            normals.Add(new Vector3(float.Parse(bits[3]), float.Parse(bits[1]), float.Parse(bits[2])));
        }
        else if (line.StartsWith("usemtl "))
        {
            Flush();
            currentMaterial = line.Substring(7).Trim();
        }
        else if (line.StartsWith("f "))
        {
            var bits = line.Split(' ');
            var c0 = Corner(bits[1]); var c1 = Corner(bits[2]); var c2 = Corner(bits[3]);
            var p0 = positions[c0[0]]; var p1 = positions[c1[0]]; var p2 = positions[c2[0]];
            if (InCut(p0) && InCut(p1) && InCut(p2)) { cut++; continue; }
            kept++;
            var t0 = uvs[c0[1]]; var t1 = uvs[c1[1]]; var t2 = uvs[c2[1]];
            var e1 = p1 - p0; var e2 = p2 - p0;
            var du1 = t1.x - t0.x; var dv1 = t1.y - t0.y; var du2 = t2.x - t0.x; var dv2 = t2.y - t0.y;
            var det = du1 * dv2 - du2 * dv1;
            var tangent = Math.Abs(det) > 1e-8f ? ((e1 * dv2 - e2 * dv1) / det).Normal : e1.Normal;
            foreach (var c in new[] { c0, c1, c2 })
                vertices.Add(new Vertex(positions[c[0]], normals[c[2]], tangent, new Vector4(uvs[c[1]].x, uvs[c[1]].y, 0, 0)));
        }
    }
    Flush();
}
var builder = Model.Builder;
foreach (var mesh in meshes) builder.AddMesh(mesh);
var detailsCut = builder.Create();

var scene = Scene.CreateEditorScene();
var flags = System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.Public | System.Reflection.BindingFlags.NonPublic;
var preRender = typeof(Scene).GetMethod("PreRender", flags, null, Type.EmptyTypes, null) ?? throw new MissingMethodException("Scene.PreRender");
var preCameraRender = typeof(Scene).GetMethod("PreCameraRender", flags, null, Type.EmptyTypes, null) ?? throw new MissingMethodException("Scene.PreCameraRender");
try
{
    CameraComponent camera;
    ModelRenderer details;
    Model detailsOld;
    var news = new System.Collections.Generic.List<GameObject>();
    var moved = new System.Collections.Generic.List<(PointLight light, Vector3 from, Vector3 to)>();
    using (scene.Push())
    {
        scene.Deserialize(root);
        details = scene.GetAllObjects(false).Single(g => g.Name == "Waterworks details").Components.Get<ModelRenderer>();
        detailsOld = details.Model;
        var lights = scene.GetAllComponents<PointLight>().ToArray();
        foreach (var cabinet in cabinets)
        {
            var at = Vec(cabinet["at"]);
            var yaw = cabinet["yaw"].GetValue<float>();
            var go = new GameObject(false, "New cabinet");
            go.WorldPosition = at;
            go.WorldRotation = Rotation.FromYaw(yaw);
            var renderer = go.Components.Create<ModelRenderer>();
            renderer.Model = model;
            news.Add(go);
            var oldLight = new Vector3(at.x - 49, at.y, at.z + 98);
            var light = lights.OrderBy(l => l.WorldPosition.Distance(oldLight)).First();
            var to = at + Rotation.FromYaw(yaw) * Vec(configuration["light_local"]);
            moved.Add((light, light.WorldPosition, to));
        }
        var holder = new GameObject(false, "Capture camera");
        camera = holder.Components.Create<CameraComponent>();
        camera.ZNear = 1;
        camera.ZFar = 30000;
        // The gameplay camera's look, at a fixed exposure: auto exposure, even held to one value, left the first shots darker.
        var tone = holder.Components.Create<Tonemapping>();
        tone.Mode = Tonemapping.TonemappingMode.ACES;
        tone.AutoExposureEnabled = false;
        tone.ExposureCompensation = MathF.Log2(configuration["exposure"].GetValue<float>());
        var bloom = holder.Components.Create<Bloom>();
        bloom.Strength = 0.1f;
        bloom.Threshold = 1.6f;
        var occlusion = holder.Components.Create<AmbientOcclusion>();
        occlusion.Radius = 48;
        occlusion.Intensity = 0.45f;
        occlusion.FalloffRange = 0.9f;
        occlusion.ThinCompensation = 1;
        var vignette = holder.Components.Create<Vignette>();
        vignette.Intensity = 0.08f;
        holder.Enabled = true;
        scene.EditorTick(.016f, .016f);
        preRender.Invoke(scene, null);
        preCameraRender.Invoke(scene, null);
    }

    // Nothing above 1024 for the map or 2048 for the cabinet: the editor is shared, and a whole map at full size has run it out of video memory.
    var textureNames = new[] { "g_tColor", "g_tNormal", "g_tRoughness", "g_tAmbientOcclusion", "SkyTexture" };
    foreach (var material in scene.GetAllComponents<ModelRenderer>().Where(r => r.Model is not null && r.Model != model).SelectMany(r => r.Model.Materials).Distinct())
        foreach (var texture in textureNames.Select(material.GetTexture).Append(material.FirstTexture).Where(t => t is not null).Distinct())
            texture.MarkUsed(1024);
    foreach (var material in model.Materials)
        foreach (var texture in textureNames.Select(material.GetTexture).Append(material.FirstTexture).Where(t => t is not null).Distinct())
            texture.MarkUsed(2048);
    for (var wait = 0; wait < configuration["stream_frames"].GetValue<int>(); wait++)
    {
        using (scene.Push()) { scene.EditorTick(.016f, .016f); preRender.Invoke(scene, null); }
        await Task.Delay(33, cancellationToken);
    }

    var settle = configuration["settle_frames"].GetValue<int>();
    var written = new System.Collections.Generic.List<string>();
    foreach (var variant in configuration["variants"].AsArray().Select(v => v.GetValue<string>()))
    {
        using (scene.Push())
        {
            var after = variant != "before";
            foreach (var go in news) go.Enabled = after;
            details.Model = after ? detailsCut : detailsOld;
            foreach (var m in moved) m.light.WorldPosition = after ? m.to : m.from;
        }
        foreach (var view in configuration["views"].AsArray())
        {
            cancellationToken.ThrowIfCancellationRequested();
            var name = view["name"].GetValue<string>();
            using var bitmap = new Bitmap(configuration["width"].GetValue<int>(), configuration["height"].GetValue<int>());
            var eye = Vec(view["eye"]);
            var target = Vec(view["target"]);
            for (var frame = 0; frame < settle; frame++)
            {
                using (scene.Push())
                {
                    camera.WorldPosition = eye;
                    camera.WorldRotation = Rotation.LookAt(target - eye);
                    camera.FieldOfView = view["fov"].GetValue<float>();
                    scene.EditorTick(.016f, .016f);
                    preRender.Invoke(scene, null);
                    preCameraRender.Invoke(scene, null);
                    camera.RenderToBitmap(bitmap);
                }
                await Task.Delay(16, cancellationToken);
            }
            System.IO.File.WriteAllBytes(System.IO.Path.Combine(destination, $"{name}-{variant}.png"), bitmap.ToPng());
            written.Add($"{name}-{variant}");
        }
    }
    return new
    {
        Before = before, Game.IsPlaying,
        SourceUnchanged = Editor.SceneEditorSession.Active?.Scene?.Id == before.SceneId && Game.ActiveScene?.Id == before.GameSceneId,
        Written = written, KeptTriangles = kept, CutTriangles = cut, Meshes = meshes.Count,
        Lights = moved.Select(m => $"{m.from} -> {m.to}").ToArray(),
        ErrorModels = scene.GetAllComponents<ModelRenderer>().Where(r => r.Model is null || r.Model.IsError).Select(r => r.GameObject.Name).ToArray(),
        ModelMaterials = model.Materials.Select(m => m.ResourcePath).ToArray()
    };
}
catch (Exception error)
{
    System.IO.File.WriteAllText(System.IO.Path.Combine(destination, "capture-error.txt"), error.ToString());
    throw;
}
finally { scene.Destroy(); }
