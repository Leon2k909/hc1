var project=Sandbox.Project.Current.GetRootPath();
var folder=System.IO.Path.Combine(project,"output/sewage/stair-treads-v1");
var plan=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder,"plan.json")));
string Hash(string p)=>Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(System.IO.File.ReadAllBytes(p))).ToLowerInvariant();
var files=plan["preview_files"].AsObject();
foreach(var p in files) {
 var source=System.IO.Path.Combine(folder,"stage",p.Key);
 if(Hash(source)!=p.Value.GetValue<string>())throw new Exception("Stage changed: "+p.Key);
 var target=System.IO.Path.Combine(project,p.Key);
 if(System.IO.File.Exists(target)&&Hash(target)!=p.Value.GetValue<string>())throw new Exception("Preview target already differs: "+p.Key);
}
foreach(var p in files) {
 var target=System.IO.Path.Combine(project,p.Key);
 System.IO.Directory.CreateDirectory(System.IO.Path.GetDirectoryName(target));
 System.IO.File.Copy(System.IO.Path.Combine(folder,"stage",p.Key),target,true);
 Editor.AssetSystem.RegisterFile(target);
}
return new{Imported=files.Count,PublicModelUnchanged=true};
