var project=Sandbox.Project.Current.GetRootPath();
var folder=System.IO.Path.Combine(project,"output/sewage/stair-treads-v1");
var plan=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder,"plan.json")));
string Hash(string p)=>Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(System.IO.File.ReadAllBytes(p))).ToLowerInvariant();
var files=plan["install_files"].AsObject();
foreach(var p in files) {
 var source=System.IO.Path.Combine(folder,"install",p.Key);var target=System.IO.Path.Combine(project,"Assets/models/maps/sewage",p.Key);
 if(Hash(source)!=p.Value.GetValue<string>())throw new Exception("Stage changed: "+p.Key);
 var current=Hash(target);
 if(current!=plan["baseline"][p.Key].GetValue<string>()&&current!=p.Value.GetValue<string>())throw new Exception("Public asset was edited meanwhile: "+p.Key);
}
var editor=Editor.SceneEditorSession.Active;var initial=new{Scene=editor.Scene.Id,GameScene=Game.ActiveScene?.Id,Game.IsPlaying,Dirty=editor.HasUnsavedChanges};
foreach(var p in files) {
 var target=System.IO.Path.Combine(project,"Assets/models/maps/sewage",p.Key);
 System.IO.File.Copy(System.IO.Path.Combine(folder,"install",p.Key),target,true);Editor.AssetSystem.RegisterFile(target);
}
var result=new{Installed=files.Select(p=>p.Key).ToArray(),SceneFileChanged=false,CollisionChanged=false,
 UserStatePreserved=editor.Scene.Id==initial.Scene&&Game.ActiveScene?.Id==initial.GameScene&&Game.IsPlaying==initial.IsPlaying&&editor.HasUnsavedChanges==initial.Dirty};
System.IO.File.WriteAllText(System.IO.Path.Combine(folder,"installed.json"),System.Text.Json.JsonSerializer.Serialize(result,new System.Text.Json.JsonSerializerOptions{WriteIndented=true}));return result;
