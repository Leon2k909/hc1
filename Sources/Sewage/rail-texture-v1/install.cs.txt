var root=Sandbox.Project.Current.GetRootPath();
var folder=System.IO.Path.Combine(root,"output/sewage/rail-texture-v1");
var plan=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder,"plan.json")));
string Hash(string p)=>Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(System.IO.File.ReadAllBytes(p))).ToLowerInvariant();
var current=System.IO.Path.Combine(root,"Assets/models/maps/sewage/sewage_catwalks.vmdl");
if(Hash(current)!=plan["public_before_sha"].GetValue<string>())throw new Exception("Public catwalk model changed meanwhile");
foreach(var item in plan["files"].AsObject()) {
 var source=System.IO.Path.Combine(folder,"stage",item.Key);
 if(Hash(source)!=item.Value.GetValue<string>())throw new Exception("Stage changed: "+item.Key);
 var target=System.IO.Path.Combine(root,item.Key);
 if(target!=current&&System.IO.File.Exists(target))throw new Exception("New asset already exists: "+item.Key);
}
var session=Editor.SceneEditorSession.Active;var state=new{Scene=session.Scene.Id,Game.IsPlaying,Dirty=session.HasUnsavedChanges};
foreach(var item in plan["files"].AsObject().OrderBy(p=>p.Key.EndsWith("sewage_catwalks.vmdl")?1:0)) {
 var target=System.IO.Path.Combine(root,item.Key);System.IO.Directory.CreateDirectory(System.IO.Path.GetDirectoryName(target));
 System.IO.File.Copy(System.IO.Path.Combine(folder,"stage",item.Key),target,true);Editor.AssetSystem.RegisterFile(target);
}
var result=new{Installed=true,GeometryChanged=false,CollisionChanged=false,UserStatePreserved=session.Scene.Id==state.Scene&&Game.IsPlaying==state.IsPlaying&&session.HasUnsavedChanges==state.Dirty};
System.IO.File.WriteAllText(System.IO.Path.Combine(folder,"installed.json"),System.Text.Json.JsonSerializer.Serialize(result));return result;
