var root=Sandbox.Project.Current.GetRootPath();
var plan=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(root,"output/sewage/tower-rails-v1/plan.json")));
var path=System.IO.Path.Combine(root,"Assets/models/maps/sewage/tower-vents-v1/base_tower.obj");
var hash=Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(System.IO.File.ReadAllBytes(path))).ToLowerInvariant();
if(hash!=plan["original_sha256"].GetValue<string>())throw new Exception("Tower changed; rebuild the railing splice first.");
var session=Editor.SceneEditorSession.All.Single(x=>x.Scene.Name=="sewage");
if(Game.IsPlaying&&Editor.SceneEditorSession.Active==session)throw new Exception("Wait for edit mode.");
var tower=session.Scene.GetAllObjects(true).Single(x=>x.Name=="Waterworks tower").Components.Get<ModelRenderer>();
if(tower.Model?.ResourcePath!="models/maps/sewage/tower-vents-v1/base_tower.vmdl")throw new Exception("Unexpected tower model.");
if(session.Scene.GetAllObjects(true).Any(x=>x.Name=="Waterworks tower safety railing"))throw new Exception("Railing already installed.");
var body=Model.Load("models/maps/sewage/tower-rails-v1/base_tower.vmdl");
var rails=Model.Load("models/maps/sewage/tower-rails-v1/rails.vmdl");
if(body is null||body.IsError||rails is null||rails.IsError)throw new Exception("Railing model failed to load.");
using(session.Scene.Push())
{
    var holder=new GameObject(false,"Waterworks tower safety railing");
    holder.Tags.Add("sewage");
    holder.Components.Create<ModelRenderer>().Model=rails;
    tower.Model=body;
    holder.Enabled=true;
}
session.HasUnsavedChanges=true;session.Save(false);
var disk=System.IO.File.ReadAllText(System.IO.Path.Combine(root,"Assets/scenes/maps/sewage/sewage.scene"));
return new{Saved=!session.HasUnsavedChanges,OnDisk=disk.Contains("Waterworks tower safety railing")&&disk.Contains("models/maps/sewage/tower-rails-v1/base_tower.vmdl"),OldAssetRetained=System.IO.File.Exists(path),ActiveScene=Editor.SceneEditorSession.Active.Scene.Name};
