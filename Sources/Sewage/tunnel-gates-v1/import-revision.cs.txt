var project=Sandbox.Project.Current.GetRootPath();var folder=System.IO.Path.Combine(project,"output/sewage/tunnel-gates-v1");
var plan=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder,"plan.json")));
var previous=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder,"previous-plan.json")));
string Hash(string p)=>Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(System.IO.File.ReadAllBytes(p))).ToLowerInvariant();
var files=plan["preview_files"].AsObject();
foreach(var p in files) {
 var source=System.IO.Path.Combine(folder,"stage",p.Key);var target=System.IO.Path.Combine(project,p.Key);
 if(Hash(source)!=p.Value.GetValue<string>())throw new Exception("Stage changed: "+p.Key);
 if(System.IO.File.Exists(target)) {
  var current=Hash(target);
  if(current!=p.Value.GetValue<string>()&&current!=previous["preview_files"][p.Key].GetValue<string>())throw new Exception("Preview edited externally: "+p.Key);
 }
}
foreach(var p in files) {
 var target=System.IO.Path.Combine(project,p.Key);
 System.IO.File.Copy(System.IO.Path.Combine(folder,"stage",p.Key),target,true);Editor.AssetSystem.RegisterFile(target);
}
return new{RevisedPreviewFiles=files.Count,PublicModelUnchanged=true};
