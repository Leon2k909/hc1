var session=Editor.SceneEditorSession.All.Single(s=>s.Scene.IsEditor && s.Scene.Source?.ResourcePath?.Equals("scenes/maps/sewage/sewage.scene",StringComparison.OrdinalIgnoreCase)==true);
var scene=session.Scene;
var root=Sandbox.Project.Current.GetRootPath();
var config=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(root,"output/sewage/capture.json")));
var name=config["name"].GetValue<string>();
if(!System.Text.RegularExpressions.Regex.IsMatch(name,"^[a-z0-9-]+$"))throw new InvalidOperationException("Invalid capture name");
foreach(var r in scene.GetAllComponents<ModelRenderer>())
for(var slot=0;slot<r.Materials.Count;slot++)
{
    var material=r.Materials.GetOverride(slot)??r.Materials.GetOriginal(slot);
    if(material is null||material.IsError)continue;
    foreach(var key in new[]{"g_tColor","g_tNormal","g_tRma"})material.GetTexture(key)?.MarkUsed(2048);
}
await Task.Delay(1000,cancellationToken);
var camera=scene.GetAllComponents<CameraComponent>().FirstOrDefault(c=>c.GameObject.Name=="editor_camera" && c.GameObject.Flags.HasFlag(GameObjectFlags.EditorOnly)) ?? scene.GetAllComponents<CameraComponent>().First(c=>c.GameObject.Name=="Reference composition camera");
var oldPosition=camera.WorldPosition;var oldRotation=camera.WorldRotation;var oldFov=camera.FieldOfView;
var flags=System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic|System.Reflection.BindingFlags.Public;
try
{
    using(scene.Push())
    {
        camera.WorldPosition=Vector3.Parse(config["position"].GetValue<string>());
        camera.WorldRotation=Rotation.From(Angles.Parse(config["angles"].GetValue<string>()));
        camera.FieldOfView=config["fov"]?.GetValue<float>()??84;
        typeof(Scene).GetMethod("PreRender",flags,null,Type.EmptyTypes,null).Invoke(scene,null);
        typeof(Scene).GetMethod("PreCameraRender",flags,null,Type.EmptyTypes,null).Invoke(scene,null);
        var post=scene.GetSystem<PostProcessSystem>();
        typeof(PostProcessSystem).GetMethod("UpdateCamera",flags).Invoke(post,new object[]{camera});
        using var bitmap=new Bitmap(1600,900);
        camera.RenderToBitmap(bitmap,false);
        var target=System.IO.Path.Combine(root,"output/sewage",name+".png");
        System.IO.File.WriteAllBytes(target,bitmap.ToPng());
        return new{Path=target,Scene=scene.Name,CameraScene=camera.Scene.Name,camera.FieldOfView};
    }
}
finally{camera.WorldPosition=oldPosition;camera.WorldRotation=oldRotation;camera.FieldOfView=oldFov;}
