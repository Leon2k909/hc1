// Apply only the isolated pergola planting and its own material sources.
var root=Sandbox.Project.Current.GetRootPath();
if(!string.Equals(System.IO.Path.GetFullPath(root).TrimEnd('\\','/'),@"C:\sbox-projects\hc1\forked-shooter-main",StringComparison.OrdinalIgnoreCase))throw new InvalidOperationException("Wrong project.");
var folder=System.IO.Path.Combine(root,"output/veridant/pergola-flowers-v2");
var request=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder,"import-request.json")));
var stage=request["stage"].GetValue<string>();
if(!System.Text.RegularExpressions.Regex.IsMatch(stage,"^stage-[a-z0-9-]+$"))throw new InvalidOperationException("Invalid stage.");
var manifest=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder,stage,"manifest.json")));
if(manifest["revision"].GetValue<string>()!="veridant-pergola-flowers-v2")throw new InvalidOperationException("Wrong revision.");
var session=Editor.SceneEditorSession.Active;var playing=Game.IsPlaying;var game=Game.ActiveScene?.Id;var dirty=session.HasUnsavedChanges;
if(playing)throw new InvalidOperationException("A play session started; inspect before changing planting.");
string Hash(string path)=>Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(System.IO.File.ReadAllBytes(path))).ToLowerInvariant();
foreach(var group in new[]{"source_sha256","protected_sha256"})
    foreach(var pair in manifest[group].AsObject())
        if(Hash(System.IO.Path.Combine(root,pair.Key))!=pair.Value.GetValue<string>())throw new InvalidOperationException("Shared input changed: "+pair.Key);
var files=manifest["files"].AsObject();
foreach(var pair in files)
{
    var path=pair.Key;
    var allowed=path.StartsWith("Assets/materials/veridant/pergola-flowers-v2/",StringComparison.Ordinal)
        ||path.StartsWith("Assets/models/maps/veridant/pergola-flowers-v2/",StringComparison.Ordinal)
        ||path=="Assets/models/maps/veridant/veridant_foliage.vmdl";
    if(!allowed||path.Contains(".."))throw new InvalidOperationException("Unexpected target: "+path);
    var item=pair.Value;var target=System.IO.Path.Combine(root,path);var after=item["after"].GetValue<string>();
    var before=item["before"]?.GetValue<string>();
    if(Hash(System.IO.Path.Combine(root,item["source"].GetValue<string>()))!=after)throw new InvalidOperationException("Staging changed.");
    if(before is not null&&Hash(System.IO.Path.Combine(root,item["backup"].GetValue<string>()))!=before)throw new InvalidOperationException("Backup changed.");
    var current=System.IO.File.Exists(target)?Hash(target):null;
    if(current!=before&&current!=after)throw new InvalidOperationException("Shared target changed: "+path);
}
var imported=new System.Collections.Generic.List<string>();
foreach(var pair in files)
{
    var item=pair.Value;var target=System.IO.Path.Combine(root,pair.Key);var after=item["after"].GetValue<string>();
    if(System.IO.File.Exists(target)&&Hash(target)==after)continue;
    System.IO.Directory.CreateDirectory(System.IO.Path.GetDirectoryName(target));
    var incoming=target+".pergola-incoming";
    try
    {
        System.IO.File.Copy(System.IO.Path.Combine(root,item["source"].GetValue<string>()),incoming,true);
        if(Hash(incoming)!=after)throw new InvalidOperationException("Incoming hash differs.");
        System.IO.File.Move(incoming,target,true);imported.Add(pair.Key);
    }
    finally{if(System.IO.File.Exists(incoming))System.IO.File.Delete(incoming);}
    Editor.AssetSystem.RegisterFile(target);
}
foreach(var pair in manifest["protected_sha256"].AsObject())
    if(Hash(System.IO.Path.Combine(root,pair.Key))!=pair.Value.GetValue<string>())throw new InvalidOperationException("Protected file changed: "+pair.Key);
var preserved=ReferenceEquals(session,Editor.SceneEditorSession.Active)&&playing==Game.IsPlaying&&game==Game.ActiveScene?.Id&&dirty==session.HasUnsavedChanges;
var result=new{Stage=stage,ImportedFiles=imported,EditorStatePreserved=preserved,SceneChanged=false,PhysicsChanged=false,OldAtlasChanged=false,UnrelatedFoliagePreserved=true};
System.IO.File.WriteAllText(System.IO.Path.Combine(folder,"native-import.json"),System.Text.Json.JsonSerializer.Serialize(result));
if(!preserved)throw new InvalidOperationException("Editor state changed; inspect the completed import.");
return result;
