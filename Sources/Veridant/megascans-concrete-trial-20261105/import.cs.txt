var root=Sandbox.Project.Current.GetRootPath();
if(!string.Equals(System.IO.Path.GetFullPath(root).TrimEnd('\\'),@"C:\sbox-projects\hc1\forked-shooter-main",StringComparison.OrdinalIgnoreCase)) throw new InvalidOperationException("Wrong checkout");
var manifest=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(root,"output/veridant/megascans-concrete-trial-20261105/manifest.json")));
string Hash(string path)=>Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(System.IO.File.ReadAllBytes(path))).ToLowerInvariant();
foreach(var kv in manifest["protected"].AsObject()) if(Hash(System.IO.Path.Combine(root,kv.Key))!=kv.Value.GetValue<string>()) throw new InvalidOperationException("Shared file changed: "+kv.Key);
var targetDir=System.IO.Path.Combine(root,"Assets",manifest["relative_asset_directory"].GetValue<string>());
var sourceDir=manifest["source"].GetValue<string>();
foreach(var kv in manifest["files"].AsObject()) {
    if(Hash(System.IO.Path.Combine(sourceDir,kv.Key))!=kv.Value.GetValue<string>()) throw new InvalidOperationException("Staging changed");
    var target=System.IO.Path.Combine(targetDir,kv.Key);
    if(System.IO.File.Exists(target) && Hash(target)!=kv.Value.GetValue<string>()) throw new InvalidOperationException("Trial path already occupied");
}
System.IO.Directory.CreateDirectory(targetDir);
foreach(var kv in manifest["files"].AsObject()) {
    var target=System.IO.Path.Combine(targetDir,kv.Key);
    if(!System.IO.File.Exists(target)) System.IO.File.Copy(System.IO.Path.Combine(sourceDir,kv.Key),target);
    Editor.AssetSystem.RegisterFile(target);
}
return new { TrialFiles=manifest["files"].AsObject().Count, MapChanged=false };
