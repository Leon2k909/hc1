var folder=System.IO.Path.Combine(root,"output/veridant/cases-v4");
var request=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder,"import-request.json")));

    throw new InvalidOperationException("Requires the saved, stopped Veridant document; no editor state was changed.");
string Hash(string path)=>Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(System.IO.File.ReadAllBytes(path))).ToLowerInvariant();
foreach(var group in new[]{"source_sha256","protected_sha256"})
    foreach(var pair in manifest[group].AsObject())
        if(Hash(System.IO.Path.Combine(root,pair.Key))!=pair.Value.GetValue<string>())throw new InvalidOperationException("Shared input changed: "+pair.Key);

var files=manifest["files"].AsObject();
foreach(var pair in files)
{
    var path=pair.Key;
    var allowed=path.StartsWith("Assets/materials/veridant/cases-v4/",StringComparison.Ordinal)
        ||path.StartsWith("Assets/models/maps/veridant/cases-v4/",StringComparison.Ordinal)||documents.Contains(path);
    if(!allowed||path.Contains(".."))throw new InvalidOperationException("Unexpected file: "+path);

    var source=System.IO.Path.Combine(root,item["source"].GetValue<string>());
    var after=item["after"].GetValue<string>();var before=item["before"]?.GetValue<string>();
    if(Hash(source)!=after)throw new InvalidOperationException("Stage changed: "+path);
    if(before is not null&&Hash(System.IO.Path.Combine(root,item["backup"].GetValue<string>()))!=before)throw new InvalidOperationException("Backup changed.");
    var current=System.IO.File.Exists(target)?Hash(target):null;
    if(current!=before&&current!=after)throw new InvalidOperationException("Shared target changed: "+path);
    if(documents.Contains(path))
    {
        var oldText=System.IO.File.ReadAllText(System.IO.Path.Combine(root,item["backup"].GetValue<string>()));
        var newText=System.IO.File.ReadAllText(source);var marker="_class = \"PhysicsShapeList\"";

    }
}

}
