// Publish the independent planting mesh without touching the shared furniture model.
var root = Sandbox.Project.Current.GetRootPath();
var folder = System.IO.Path.Combine(root, "output/veridant/palm-terrace-v1");
var manifest = System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder, "pits-stage/manifest.json")));
if (manifest["revision"].GetValue<string>() != "veridant-palm-pits-v2") throw new InvalidOperationException("Wrong stage.");
var session = Editor.SceneEditorSession.Active;
if (Game.IsPlaying || session?.Scene.Source?.ResourcePath != "scenes/maps/veridant/veridant.scene" || session.HasUnsavedChanges)
    throw new InvalidOperationException("Expected the saved, stopped Veridant scene.");
var sceneId = session.Scene.Id;
string Hash(string path) => Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(System.IO.File.ReadAllBytes(path))).ToLowerInvariant();
var plans = new System.Collections.Generic.List<(string Source, string Target)>();
foreach (var pair in manifest["files"].AsObject())
{
    if (!new[] { ".obj", ".mtl", ".vmdl" }.Any(ext => pair.Key == "Assets/models/maps/veridant/veridant_palm_pits" + ext))
        throw new InvalidOperationException("Unexpected target.");
    var source = System.IO.Path.Combine(root, pair.Value["source"].GetValue<string>());
    var target = System.IO.Path.Combine(root, pair.Key);
    var expected = pair.Value["after"].GetValue<string>();
    if (Hash(source) != expected) throw new InvalidOperationException("Stage changed.");
    if (System.IO.File.Exists(target))
    {
        if (Hash(target) != expected) throw new InvalidOperationException("Another version already exists: " + pair.Key);
        continue;
    }
    plans.Add((source, target));
}
foreach (var plan in plans.OrderBy(p => p.Target.EndsWith(".vmdl") ? 1 : 0))
{
    cancellationToken.ThrowIfCancellationRequested();
    System.IO.File.Copy(plan.Source, plan.Target, false);
    Editor.AssetSystem.RegisterFile(plan.Target);
}
var parent = session.Scene.GetAllObjects(false).Single(go => go.Name == "Veridant");
var existing = session.Scene.GetAllComponents<ModelRenderer>()
    .Where(m => m.Model?.ResourcePath == "models/maps/veridant/veridant_palm_pits.vmdl")
    .Select(m => m.GameObject.Id).ToArray();
if (sceneId != Editor.SceneEditorSession.Active.Scene.Id || Game.IsPlaying || session.HasUnsavedChanges)
    throw new InvalidOperationException("Editor state changed during import; inspect the result.");
return new { Imported = plans.Count, ParentId = parent.Id, Existing = existing, Pits = manifest["pits"].GetValue<int>() };
