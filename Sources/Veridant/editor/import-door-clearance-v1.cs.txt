// Publish the checked shelf-only OBJ patch through the active editor.
var root=Sandbox.Project.Current.GetRootPath();
var folder=System.IO.Path.Combine(root,"output/veridant/door-obstruction");
var manifest=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder,"stage-v1/manifest.json")));
if(manifest["revision"].GetValue<string>()!="veridant-door-clearance-v1") throw new InvalidOperationException("Wrong revision.");
var session=Editor.SceneEditorSession.Active;var playing=Game.IsPlaying;var game=Game.ActiveScene?.Id;var dirty=session.HasUnsavedChanges;
if(playing)throw new InvalidOperationException("A play session started; inspect before replacing its model.");
string Hash(string p)=>Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(System.IO.File.ReadAllBytes(p))).ToLowerInvariant();
foreach(var group in new[]{"source_sha256","protected_sha256"})
    foreach(var pair in manifest[group].AsObject())
        if(Hash(System.IO.Path.Combine(root,pair.Key))!=pair.Value.GetValue<string>())throw new InvalidOperationException("Shared input changed: "+pair.Key);
var relative="Assets/models/maps/veridant/veridant_buildings.obj";
if(manifest["files"].AsObject().Count!=1)throw new InvalidOperationException("Unexpected file count.");
var item=manifest["files"][relative];
var source=System.IO.Path.Combine(root,item["source"].GetValue<string>());
var backup=System.IO.Path.Combine(root,item["backup"].GetValue<string>());
var target=System.IO.Path.Combine(root,relative);
var before=item["before"].GetValue<string>();var after=item["after"].GetValue<string>();
if(Hash(source)!=after||Hash(backup)!=before)throw new InvalidOperationException("Staging changed.");
var imported=false;
if(Hash(target)!=after)
{
    if(Hash(target)!=before)throw new InvalidOperationException("Shared building model changed.");
    var incoming=target+".door-clearance-incoming";
    try
    {
        System.IO.File.Copy(source,incoming,true);
        if(Hash(incoming)!=after)throw new InvalidOperationException("Incoming checksum differs.");
        System.IO.File.Move(incoming,target,true);imported=true;
    }
    finally{if(System.IO.File.Exists(incoming))System.IO.File.Delete(incoming);}
    Editor.AssetSystem.RegisterFile(target);
}
foreach(var pair in manifest["protected_sha256"].AsObject())
    if(Hash(System.IO.Path.Combine(root,pair.Key))!=pair.Value.GetValue<string>())throw new InvalidOperationException("Protected file changed.");
var preserved=ReferenceEquals(session,Editor.SceneEditorSession.Active)&&playing==Game.IsPlaying&&game==Game.ActiveScene?.Id&&dirty==session.HasUnsavedChanges;
var result=new{Imported=imported,DoorwaysCleared=3,EditorStatePreserved=preserved,SceneChanged=false,CollisionChanged=false,MaterialsChanged=false};
System.IO.File.WriteAllText(System.IO.Path.Combine(folder,"native-import.json"),System.Text.Json.JsonSerializer.Serialize(result));
if(!preserved)throw new InvalidOperationException("Editor state changed; inspect the completed import.");
return result;
