            Width=m.GetVector4("g_flAtlasWidth"),Height=m.GetVector4("g_flAtlasHeight"),
            Bottom0=m.GetVector4("g_flBottom0"),Bottom1=m.GetVector4("g_flBottom1"),Bottom2=m.GetVector4("g_flBottom2") }).ToArray() }).ToArray();
