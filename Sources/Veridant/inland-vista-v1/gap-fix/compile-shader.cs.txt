var path = "shaders/hc1_veridant_skirt.shader";
var result = await Editor.EditorUtility.CompileShader(path, new() { ForceRecompile = true }, cancellationToken);
if (result.Success && result.CompiledShader is { Length: > 0 })
    System.IO.File.WriteAllBytes(System.IO.Path.Combine(Sandbox.Project.Current.GetRootPath(), "Assets", path + "_c"), result.CompiledShader);
return new { result.Success, Bytes = result.CompiledShader?.Length,
    Output = result.Programs.SelectMany(p => p.Output ?? new System.Collections.Generic.List<string>()).Take(30).ToArray() };
