// Before and after of Cerro Blanco's cafe cups, rendered by s&box: run with sbox-cli --call run_csharp '{"path":...}'.
// Loads cerroblanco.scene's visual roots into an isolated editor scene, so the open scene and play state are never touched and nothing is saved, with the player camera's post stack and exposure held still.
// "after" swaps the fit-out's detail model for cafe_cup_v2/compare's copy without the old cups (cut_compare.py) and sets the new ones where they were (cups.py): the full cup at 70 degrees round
// each table, the drunk one at 230, each handle along the table's edge. Plan ( e, n ) is world ( n, -e ).
// Writes <view>-<variant>.png into the work folder's engine/.
var projectRoot = Sandbox.Project.Current.GetRootPath();
var destination = System.IO.Path.Combine(projectRoot, "output/asset-improvement-20261030/cerroblanco-cafe-cup/engine");
System.IO.Directory.CreateDirectory(destination);
var keep = new System.Collections.Generic.HashSet<string>(new[] { "Sun", "Sky", "Fog", "Look", "Cerro Blanco", "Trees", "Lights", "Art Lights", "Reflection Probes", "Museum indirect light", "Cerro Blanco construction details" }, StringComparer.OrdinalIgnoreCase);
var root = System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(projectRoot, "Assets/scenes/maps/CerroBlanco/cerroblanco.scene"))).AsObject();
var objects = new System.Text.Json.Nodes.JsonArray();
foreach (var item in root["GameObjects"].AsArray())
    if (keep.Contains(item["Name"]?.GetValue<string>() ?? "")) objects.Add(item.DeepClone());
var prefab = System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(projectRoot, "Assets/prefabs/player/cameras/default_player_camera.prefab"))).AsObject();
var wanted = new[] { "Sandbox.CameraComponent", "Sandbox.Bloom", "Sandbox.Tonemapping", "Sandbox.Sharpen", "Sandbox.AmbientOcclusion", "Sandbox.Vignette", "Facepunch.PhotographicResponse" };
var components = new System.Text.Json.Nodes.JsonArray();
foreach (var component in prefab["RootObject"]["Components"].AsArray())
    if (wanted.Contains(component["__type"]?.GetValue<string>())) components.Add(component.DeepClone());
objects.Add(new System.Text.Json.Nodes.JsonObject { ["__guid"] = System.Text.Json.Nodes.JsonNode.Parse("\"" + Guid.NewGuid() + "\""), ["Name"] = "Capture camera", ["Enabled"] = true, ["Components"] = components });
root["GameObjects"] = objects;
if (root["SceneProperties"]?["NavMesh"] is System.Text.Json.Nodes.JsonObject nav) nav["Enabled"] = false;
if (root["SceneProperties"] is System.Text.Json.Nodes.JsonObject sceneProperties) sceneProperties["WantsSystemScene"] = false;
var chair = Model.Load("models/maps/cerroblanco/cafe_cup_v2/cup.vmdl");
if (chair is null || chair.IsError) throw new InvalidOperationException("The new cup hasn't compiled.");
var topOnly = Model.Load("models/maps/cerroblanco/cafe_cup_v2/cup_empty.vmdl");
if (topOnly is null || topOnly.IsError) throw new InvalidOperationException("The drunk cup hasn't compiled.");
var cut = Model.Load("models/maps/cerroblanco/cafe_cup_v2/compare/cerroblanco_fitout_fine_cut.vmdl");
if (cut is null || cut.IsError) throw new InvalidOperationException("The dressing's cut copy hasn't compiled.");
var scene = Scene.CreateEditorScene();
var flags = System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.Public | System.Reflection.BindingFlags.NonPublic;
var preRender = typeof(Scene).GetMethod("PreRender", flags, null, Type.EmptyTypes, null);
var preCameraRender = typeof(Scene).GetMethod("PreCameraRender", flags, null, Type.EmptyTypes, null);
Vector3 W(float e, float n, float z) => new Vector3(n, -e, z);
try
{
    CameraComponent camera;
    ModelRenderer dressing;
    Model original;
    var added = new System.Collections.Generic.List<GameObject>();
    var info = new System.Collections.Generic.List<string>();
    using (scene.Push())
    {
        scene.Deserialize(root);
        var holder = scene.GetAllObjects(false).Single(g => g.Name == "Capture camera");
        camera = holder.Components.Get<CameraComponent>();
        camera.ZNear = 1;
        camera.ZFar = 30000;
        camera.IsMainCamera = true;
        foreach (var tonemap in scene.GetAllComponents<Tonemapping>())
        {
            var hold = ( tonemap.MinimumExposure + tonemap.MaximumExposure ) / 2f;
            tonemap.AutoExposureEnabled = true;
            tonemap.MinimumExposure = tonemap.MaximumExposure = hold;
            tonemap.ExposureCompensation = 0;
            tonemap.Rate = 10;
        }
        dressing = scene.GetAllObjects(false).Single(g => g.Name == "Fit-out Detail").Components.Get<ModelRenderer>();
        original = dressing.Model;
        info.Add("fit-out detail " + original.ResourcePath);
        // World ( x, y ), yaw and which cup, from cups.py: the full one first at each table.
        foreach (var (x, y, yaw, full) in new[] { (225.638f, -622.052f, 250.0f, true), (215.404f, -616.143f, 50.0f, false), (165.638f, -802.052f, 250.0f, true), (155.404f, -796.143f, 50.0f, false), (425.638f, -862.052f, 250.0f, true), (415.404f, -856.143f, 50.0f, false), (445.638f, -642.052f, 250.0f, true), (435.404f, -636.143f, 50.0f, false), (875.638f, 537.948f, 250.0f, true), (865.404f, 543.857f, 50.0f, false), (945.638f, 417.948f, 250.0f, true), (935.404f, 423.857f, 50.0f, false), (995.638f, 537.948f, 250.0f, true), (985.404f, 543.857f, 50.0f, false), (925.638f, 297.948f, 250.0f, true), (915.404f, 303.857f, 50.0f, false) })
        {
            var go = new GameObject(false, "New cup");
            go.WorldPosition = new Vector3(x, y, 29.5f);
            go.WorldRotation = Rotation.FromYaw(yaw);
            go.Components.Create<ModelRenderer>().Model = full ? chair : topOnly;
            added.Add(go);
        }
        scene.EditorTick(.016f, .016f);
        preRender.Invoke(scene, null);
        preCameraRender.Invoke(scene, null);
    }
    var textureNames = new[] { "g_tColor", "g_tNormal", "g_tRoughness", "g_tAmbientOcclusion", "g_tMetalness", "SkyTexture" };
    foreach (var material in scene.GetAllComponents<ModelRenderer>().Where(r => r.Model is not null && r.Model != chair && r.Model != topOnly).SelectMany(r => r.Model.Materials).Distinct())
        foreach (var texture in textureNames.Select(material.GetTexture).Append(material.FirstTexture).Where(t => t is not null).Distinct())
            texture.MarkUsed(1024);
    foreach (var material in chair.Materials.Concat(topOnly.Materials).Distinct())
        foreach (var texture in textureNames.Select(material.GetTexture).Append(material.FirstTexture).Where(t => t is not null).Distinct())
            texture.MarkUsed(2048);
    for (var wait = 0; wait < 130; wait++)
    {
        using (scene.Push()) { scene.EditorTick(.016f, .016f); preRender.Invoke(scene, null); }
        await Task.Delay(33, cancellationToken);
    }
    // The courtyard table at ( -540, 870 ): its full cup at ( -537.95, 875.64 ) and its drunk one at ( -543.86, 865.40 ); and the cafe's table at ( 620, 220 ).
    var views = new (string name, Vector3 eye, Vector3 target, float fov)[]
    {
        ("table", W(-528f, 852f, 50f), W(-540f, 870f, 29.5f), 50f),
        ("full", W(-530.95f, 869.64f, 36f), W(-537.95f, 875.64f, 30.6f), 34f),
        ("drunk", W(-537.86f, 858.40f, 36.5f), W(-543.86f, 865.40f, 30.5f), 36f),
        ("above", W(-536.45f, 873.14f, 40f), W(-537.95f, 875.64f, 31.5f), 34f),
        ("seated", W(-517.4f, 878.2f, 46f), W(-540f, 870f, 30f), 60f),
        ("cafe", W(680f, 260f, 64f), W(620f, 220f, 26f), 50f),
    };
    // The first frames come out dark while the scene's light settles: render the first view a while and throw it away.
    using (var warm = new Bitmap(640, 360))
        for (var frame = 0; frame < 60; frame++)
        {
            using (scene.Push())
            {
                camera.WorldPosition = views[0].eye;
                camera.WorldRotation = Rotation.LookAt(views[0].target - views[0].eye);
                camera.FieldOfView = views[0].fov;
                scene.EditorTick(.016f, .016f);
                preRender.Invoke(scene, null);
                preCameraRender.Invoke(scene, null);
                camera.RenderToBitmap(warm);
            }
            await Task.Delay(16, cancellationToken);
        }
    foreach (var variant in new[] { "before", "after" })
    {
        using (scene.Push())
        {
            dressing.Model = variant == "after" ? cut : original;
            foreach (var go in added) go.Enabled = variant == "after";
        }
        for (var settle = 0; settle < 20; settle++)
        {
            using (scene.Push()) { scene.EditorTick(.016f, .016f); preRender.Invoke(scene, null); }
            await Task.Delay(33, cancellationToken);
        }
        foreach (var view in views)
        {
            cancellationToken.ThrowIfCancellationRequested();
            using var bitmap = new Bitmap(1280, 720);
            for (var frame = 0; frame < 12; frame++)
            {
                using (scene.Push())
                {
                    camera.WorldPosition = view.eye;
                    camera.WorldRotation = Rotation.LookAt(view.target - view.eye);
                    camera.FieldOfView = view.fov;
                    scene.EditorTick(.016f, .016f);
                    preRender.Invoke(scene, null);
                    preCameraRender.Invoke(scene, null);
                    camera.RenderToBitmap(bitmap);
                }
                await Task.Delay(16, cancellationToken);
            }
            System.IO.File.WriteAllBytes(System.IO.Path.Combine(destination, $"{view.name}-{variant}.png"), bitmap.ToPng());
        }
    }
    return new { Game.IsPlaying, Info = info, Bounds = chair.Bounds.Size.ToString(), Materials = chair.Materials.Select(m => m.ResourcePath).ToArray() };
}
catch (Exception error)
{
    System.IO.File.WriteAllText(System.IO.Path.Combine(destination, "capture-error.txt"), error.ToString());
    throw;
}
finally { scene.Destroy(); }
