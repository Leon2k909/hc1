var project=Sandbox.Project.Current.GetRootPath();
var folder=System.IO.Path.Combine(project,"output/asset-improvement-20261030/riverside-porch-light/installed/legacy-cleanup");
var plan=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder,"plan.json"))).AsArray();
string Hash(byte[] b)=>Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(b)).ToLowerInvariant();
foreach (var row in plan)
{
    var path=System.IO.Path.GetFullPath(System.IO.Path.Combine(project,row["path"].GetValue<string>()));
    if (!path.StartsWith(System.IO.Path.GetFullPath(System.IO.Path.Combine(project,"Assets/models"))+System.IO.Path.DirectorySeparatorChar,StringComparison.OrdinalIgnoreCase)) throw new Exception("Invalid cleanup path");
    var hash=Hash(System.IO.File.ReadAllBytes(path));
    if (hash!=row["before_sha256"].GetValue<string>() && hash!=row["after_sha256"].GetValue<string>()) throw new Exception("Cleanup input changed: "+path);
    if (Hash(System.IO.File.ReadAllBytes(System.IO.Path.Combine(folder,"stage",row["path"].GetValue<string>())))!=row["after_sha256"].GetValue<string>()) throw new Exception("Staged cleanup mismatch");
}
foreach (var row in plan) System.IO.File.WriteAllBytes(System.IO.Path.Combine(project,row["path"].GetValue<string>()),System.IO.File.ReadAllBytes(System.IO.Path.Combine(folder,"stage",row["path"].GetValue<string>())));
return new {CleanedFiles=plan.Count,OldComparisonFaces=plan.Sum(r=>r["removed_faces"].GetValue<int>()),BackupsRetained=true};
