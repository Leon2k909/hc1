var session=Editor.SceneEditorSession.Active;
if (session?.Scene is null || !session.Scene.Name.Equals("riverside",StringComparison.OrdinalIgnoreCase) || session.IsPlaying || session.HasUnsavedChanges) throw new Exception("Need clean Riverside edit mode");
var project=Sandbox.Project.Current.GetRootPath();
var folder=System.IO.Path.Combine(project,"output/asset-improvement-20261030/riverside-porch-light/installed");
var plan=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder,"plan.json")));
string Hash(byte[] b)=>Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(b)).ToLowerInvariant();
foreach (var row in plan["public_models"].AsArray())
    if (Hash(System.IO.File.ReadAllBytes(System.IO.Path.Combine(project,row["path"].GetValue<string>())))!=row["sha256"].GetValue<string>()) throw new Exception("ModelDoc changed after staging: "+row["path"]);
foreach (var row in plan["cuts"].AsArray())
{
    var before=Hash(System.IO.File.ReadAllBytes(System.IO.Path.Combine(project,row["path"].GetValue<string>())));
    if (before!=row["before_sha256"].GetValue<string>() && before!=row["after_sha256"].GetValue<string>()) throw new Exception("Mesh changed after staging: "+row["path"]);
}
var candidate=Editor.AssetSystem.FindByPath("models/props/riverside/porch_light_v2/porch_light.vmdl") ?? throw new Exception("New lamp missing");
await candidate.CompileIfNeededAsync();
if (Model.Load(candidate.Path).IsError) throw new Exception("New lamp failed to load");
foreach (var row in plan["cuts"].AsArray())
{
    var path=row["path"].GetValue<string>();
    var relative=path.Substring("Assets/".Length);
    var source=System.IO.Path.Combine(project,path);
    var staged=System.IO.Path.Combine(folder,"stage",relative);
    if (Hash(System.IO.File.ReadAllBytes(staged))!=row["after_sha256"].GetValue<string>()) throw new Exception("Staged mesh mismatch");
    System.IO.File.WriteAllBytes(source,System.IO.File.ReadAllBytes(staged));
}
using (session.Scene.Push())
{
    var group=session.Scene.GetAllObjects(true).SingleOrDefault(g=>g.Name=="Porch Lights");
    if (group is null) { group=new GameObject(false,"Porch Lights");group.Deserialize(plan["group"].AsObject()); }
    var renderers=group.GetComponentsInChildren<ModelRenderer>().ToArray();
    if (renderers.Length!=26 || renderers.Any(r=>r.Model is null || r.Model.IsError)) throw new Exception("Expected 26 valid lamp renderers");
    if (group.GetComponentsInChildren<Light>().Any()) throw new Exception("Fixture group must not add point lights");
    session.HasUnsavedChanges=true;
    return new { Installed=renderers.Length,OldActiveFacesRemoved=312,AddedColliders=group.GetComponentsInChildren<Collider>().Count(),NeedsSave=true };
}
