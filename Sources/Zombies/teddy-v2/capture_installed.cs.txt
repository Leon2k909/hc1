// Riverside's teddy as installed, from the scene and compiled pad model as they now are: run with sbox-cli --call run_csharp '{"path":...}'.
// Loads riverside.scene's visual roots into an isolated editor scene, so the open scene and play state are never touched and nothing is saved. Exposure is held at the map's ceiling, 3.5.
// capture_engine.cs.txt made the before and after, and needs the old teddy still in the pad. Writes <view>-installed.png to the work folder's engine/.
var projectRoot = Sandbox.Project.Current.GetRootPath();
var destination = System.IO.Path.Combine(projectRoot, "output/asset-improvement-20261030/teddy/engine");
var source = Editor.SceneEditorSession.Active?.Scene;
var before = new { Game.IsPlaying, Scene = source?.Name, SceneId = source?.Id, GameSceneId = Game.ActiveScene?.Id };
var visual = new System.Collections.Generic.HashSet<string>(StringComparer.OrdinalIgnoreCase) { "Moon", "Sky", "Fog", "Wind", "Look", "Riverside", "Lights", "Fires", "Burn Drums", "Cooler", "Teddy", "Reflection Probes", "Volumetric Fog" };
var root = System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(projectRoot, "Assets/scenes/maps/Riverside/riverside.scene"))).AsObject();
var objects = new System.Text.Json.Nodes.JsonArray();
foreach (var item in root["GameObjects"].AsArray())
    if (visual.Contains(item["Name"]?.GetValue<string>() ?? "")) objects.Add(item.DeepClone());
root["GameObjects"] = objects;
if (root["SceneProperties"]?["NavMesh"] is System.Text.Json.Nodes.JsonObject nav) nav["Enabled"] = false;
if (root["SceneProperties"] is System.Text.Json.Nodes.JsonObject sceneProperties) sceneProperties["WantsSystemScene"] = false;

var foot = new Vector3(-215f, 24f, 96f);
var facing = new Vector3(-0.8944f, -0.4472f, 0f);
var left = new Vector3(0.4472f, -0.8944f, 0f);
var views = new (string Name, Vector3 Eye, Vector3 Target, float Fov)[]
{
    ("close", foot + facing * 34f + left * -12f + Vector3.Up * 24f, foot + facing * 2.5f + Vector3.Up * 8f, 55f),
    ("with_cooler", foot + facing * 110f + left * -90f + Vector3.Up * 64f, new Vector3(-190f, 22f, 104f), 55f),
};

var scene = Scene.CreateEditorScene();
var flags = System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.Public | System.Reflection.BindingFlags.NonPublic;
var preRender = typeof(Scene).GetMethod("PreRender", flags, null, Type.EmptyTypes, null) ?? throw new MissingMethodException("Scene.PreRender");
var preCameraRender = typeof(Scene).GetMethod("PreCameraRender", flags, null, Type.EmptyTypes, null) ?? throw new MissingMethodException("Scene.PreCameraRender");
try
{
    CameraComponent camera;
    ModelRenderer teddy, pad;
    using (scene.Push())
    {
        scene.Deserialize(root);
        foreach (var tonemap in scene.GetAllComponents<Tonemapping>())
        {
            tonemap.AutoExposureEnabled = true;
            tonemap.MinimumExposure = tonemap.MaximumExposure = 3.5f;
            tonemap.ExposureCompensation = 0;
            tonemap.Rate = 10;
        }
        teddy = scene.GetAllObjects(false).Single(g => g.Name == "Teddy").Components.Get<ModelRenderer>();
        pad = scene.GetAllObjects(false).Single(g => g.Name == "Pad").Components.Get<ModelRenderer>();
        var holder = new GameObject(false, "Capture camera");
        camera = holder.Components.Create<CameraComponent>();
        camera.ZNear = 1;
        camera.ZFar = 30000;
        holder.Enabled = true;
        scene.EditorTick(.016f, .016f);
        preRender.Invoke(scene, null);
        preCameraRender.Invoke(scene, null);
    }
    var padHasOld = pad.Model.Materials.Any(m => (m.ResourcePath ?? "").Contains("rs_teddy"));
    var textureNames = new[] { "g_tColor", "g_tNormal", "g_tRoughness", "g_tAmbientOcclusion", "SkyTexture" };
    foreach (var material in scene.GetAllComponents<ModelRenderer>().Where(r => r.Model is not null && r != teddy).SelectMany(r => r.Model.Materials).Distinct())
        foreach (var texture in textureNames.Select(material.GetTexture).Append(material.FirstTexture).Where(t => t is not null).Distinct())
            texture.MarkUsed(1024);
    foreach (var material in teddy.Model.Materials)
        foreach (var texture in textureNames.Select(material.GetTexture).Append(material.FirstTexture).Where(t => t is not null).Distinct())
            texture.MarkUsed(2048);
    for (var wait = 0; wait < 90; wait++)
    {
        using (scene.Push()) { scene.EditorTick(.016f, .016f); preRender.Invoke(scene, null); }
        await Task.Delay(33, cancellationToken);
    }
    var written = new System.Collections.Generic.List<string>();
    foreach (var view in views)
    {
        cancellationToken.ThrowIfCancellationRequested();
        using var bitmap = new Bitmap(1536, 1024);
        for (var frame = 0; frame < 24; frame++)
        {
            using (scene.Push())
            {
                camera.WorldPosition = view.Eye;
                camera.WorldRotation = Rotation.LookAt(view.Target - view.Eye);
                camera.FieldOfView = view.Fov;
                scene.EditorTick(.016f, .016f);
                preRender.Invoke(scene, null);
                preCameraRender.Invoke(scene, null);
                camera.RenderToBitmap(bitmap);
            }
            await Task.Delay(16, cancellationToken);
        }
        System.IO.File.WriteAllBytes(System.IO.Path.Combine(destination, $"{view.Name}-installed.png"), bitmap.ToPng());
        written.Add(view.Name);
    }
    return new
    {
        Before = before, Game.IsPlaying,
        SourceUnchanged = Editor.SceneEditorSession.Active?.Scene?.Id == before.SceneId && Game.ActiveScene?.Id == before.GameSceneId,
        Written = written, PadStillHasOldTeddy = padHasOld, TeddyModel = teddy.Model?.ResourcePath, TeddyError = teddy.Model?.IsError,
        TeddyAt = teddy.WorldPosition, ErrorModels = scene.GetAllComponents<ModelRenderer>().Where(r => r.Model is null || r.Model.IsError).Select(r => r.GameObject.Name).ToArray()
    };
}
finally { scene.Destroy(); }
