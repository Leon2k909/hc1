// Railhead's offscreen views without touching the open scene or the play state: run with the native s&box MCP's
// run_csharp. Loads the scene file's visual roots (Assets/scenes/maps/Railhead/railhead.scene, or "scene" in the
// config) into an isolated editor scene, renders the views in output/railhead/captures/config.json to PNGs in its
// "folder", then destroys that scene. Adapted from Cerro Blanco's capture-isolated.
// - Railhead's reflection probes are realtime; they're left as they are (no forced cubemap renders).
// - The indirect light volume's buffer is updated by hand (DDGIVolumeSystem.MarkDirty / UpdateVolumes), as a scene
//   that isn't ticking never does it.
// Config: "views": [ { "name", "eye": [e, n, z], "target": [e, n, z], "fov", optional "width"/"height", "art": true
// (also written to Assets/ui/maps/railhead.png) } ], "width", "height", "settle_frames", "stream_frames",
// "fixed_exposure", "hide": [ object names ], "disable": [ component type names ].
var projectRoot = Sandbox.Project.Current.GetRootPath();
var destination = System.IO.Path.Combine(projectRoot, "output/railhead/wing-poster");
System.IO.Directory.CreateDirectory(destination);
var configuration = System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(destination, "config.json"))).AsObject();
// "folder": where under captures the views go (baseline, round1, ...).
var folder = configuration["folder"]?.GetValue<string>() ?? "latest";
if (!System.Text.RegularExpressions.Regex.IsMatch(folder, "^[a-z0-9-]+$")) throw new InvalidOperationException("Invalid folder.");
var configFolder = destination;
destination = System.IO.Path.Combine(destination, folder);
System.IO.Directory.CreateDirectory(destination);
var source = Editor.SceneEditorSession.Active?.Scene;
var before = new { Game.IsPlaying, Scene = source?.Name, SceneId = source?.Id, GameSceneId = Game.ActiveScene?.Id };
var visual = new System.Collections.Generic.HashSet<string>(StringComparer.OrdinalIgnoreCase) {
    "Sun", "Sky", "Fog", "Look", "Station", "Trees", "Lights", "Reflection Probes", "Station Indirect Light", "Haze"
};
var scenePath = configuration["scene"]?.GetValue<string>() ?? "Assets/scenes/maps/Railhead/railhead.scene";
var root = System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(projectRoot, scenePath))).AsObject();
var objects = new System.Text.Json.Nodes.JsonArray();
foreach (var item in root["GameObjects"].AsArray())
    if (visual.Contains(item["Name"]?.GetValue<string>() ?? "")) objects.Add(item.DeepClone());
root["GameObjects"] = objects;
root["SceneProperties"]["NavMesh"]["Enabled"] = false;
root["SceneProperties"]["WantsSystemScene"] = false;
var scene = Scene.CreateEditorScene();
var flags = System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.Public | System.Reflection.BindingFlags.NonPublic;
var preRender = typeof(Scene).GetMethod("PreRender", flags, null, Type.EmptyTypes, null) ?? throw new MissingMethodException("Scene.PreRender");
var preCameraRender = typeof(Scene).GetMethod("PreCameraRender", flags, null, Type.EmptyTypes, null) ?? throw new MissingMethodException("Scene.PreCameraRender");
void UpdateIndirect()
{
    var systemType = typeof(Scene).Assembly.GetType("Sandbox.DDGIVolumeSystem");
    if (systemType is null) return;
    var getter = typeof(Scene).GetMethods().Single(m => m.Name == "GetSystem" && m.IsGenericMethodDefinition && m.GetParameters().Length == 0);
    var system = getter.MakeGenericMethod(systemType).Invoke(scene, null);
    systemType.GetMethod("MarkDirty", flags)?.Invoke(system, null);
    systemType.GetMethod("UpdateVolumes", flags)?.Invoke(system, null);
}
try {
    CameraComponent camera;
    var hidden = new System.Collections.Generic.List<string>();
    using (scene.Push()) {
        scene.Deserialize(root);
        if (configuration["hide"] is System.Text.Json.Nodes.JsonArray hide)
            foreach (var name in hide.Select(x => x.GetValue<string>()))
                foreach (var go in scene.GetAllObjects(false).Where(g => g.Name == name).ToArray()) { go.Enabled = false; hidden.Add(go.Name); }
        if (configuration["fixed_exposure"] is not null)
            foreach (var tonemap in scene.GetAllComponents<Tonemapping>())
            {
                tonemap.AutoExposureEnabled = true;
                tonemap.MinimumExposure = tonemap.MaximumExposure = configuration["fixed_exposure"].GetValue<float>();
                tonemap.ExposureCompensation = 0;
                tonemap.Rate = 10;
            }
        if (configuration["disable"] is System.Text.Json.Nodes.JsonArray off)
            foreach (var typeName in off.Select(x => x.GetValue<string>()))
                foreach (var component in scene.GetAllObjects(false).SelectMany(go => go.Components.GetAll()).Where(c => c.GetType().Name == typeName).ToArray())
                { component.Enabled = false; hidden.Add(typeName); }
        var holder = new GameObject(false, "Railhead capture camera");
        camera = holder.Components.Create<CameraComponent>();
        camera.ZNear = 4; camera.ZFar = 140000;
        camera.FovAxis = CameraComponent.Axis.Horizontal;
        holder.Enabled = true;
        scene.EditorTick(.016f, .016f);
        preRender.Invoke(scene, null);
        preCameraRender.Invoke(scene, null);
        UpdateIndirect();
    }
    // Textures stream in: ask for the maps at "texture_size" (default 1024), then give the streamer a few seconds. Never
    // ask for a whole map at 4096: on 2026-10-15 that ran the shared editor out of video memory with another map open.
    var textureSize = Math.Clamp(configuration["texture_size"]?.GetValue<int>() ?? 1024, 256, 2048);
    var marked = 0;
    var textureNames = new[] { "g_tColor", "g_tNormal", "g_tRma", "g_tRoughness", "g_tSelfIllumMask", "g_tTransmissiveColor", "g_tAmbientOcclusion", "g_tHeight", "SkyTexture" };
    foreach (var material in scene.GetAllComponents<ModelRenderer>().Where(r => r.Model is not null)
                 .SelectMany(r => Enumerable.Range(0, r.Materials.Count).Select(s => r.Materials.GetOverride(s) ?? r.Materials.GetOriginal(s)))
                 .Where(m => m is not null && !m.IsError).Distinct())
        foreach (var texture in textureNames.Select(material.GetTexture).Append(material.FirstTexture).Where(t => t is not null).Distinct())
        { texture.MarkUsed(textureSize); marked++; }
    for (var wait = 0; wait < (configuration["stream_frames"]?.GetValue<int>() ?? 90); wait++)
    {
        using (scene.Push()) { scene.EditorTick(.016f, .016f); preRender.Invoke(scene, null); }
        await Task.Delay(33, cancellationToken);
    }
    using (scene.Push()) UpdateIndirect();
    // Plan ( e, n, z ) to world ( n, -e, z ).
    Vector3 World(System.Text.Json.Nodes.JsonNode node) { var a = node.AsArray(); return new(a[1].GetValue<float>(), -a[0].GetValue<float>(), a[2].GetValue<float>()); }
    var settle = configuration["settle_frames"]?.GetValue<int>() ?? 20;
    var written = new System.Collections.Generic.List<string>();
    foreach (var view in configuration["views"].AsArray())
    {
        cancellationToken.ThrowIfCancellationRequested();
        var name = view["name"].GetValue<string>();
        var w = view["width"]?.GetValue<int>() ?? configuration["width"]?.GetValue<int>() ?? 1600;
        var h = view["height"]?.GetValue<int>() ?? configuration["height"]?.GetValue<int>() ?? 900;
        using var bitmap = new Bitmap(w, h);
        var eye = World(view["eye"]);
        var target = World(view["target"]);
        // "exposure": pin the tonemapper for this view (the game's own range is 2 outdoors to 3 indoors).
        if (view["exposure"] is not null)
            using (scene.Push())
                foreach (var tonemap in scene.GetAllComponents<Tonemapping>())
                {
                    tonemap.AutoExposureEnabled = true;
                    tonemap.MinimumExposure = tonemap.MaximumExposure = view["exposure"].GetValue<float>();
                    tonemap.ExposureCompensation = 0;
                    tonemap.Rate = 10;
                }
        // The first view also lets the tonemapper settle: it starts from wherever a new scene's exposure is.
        var frames = written.Count == 0 ? settle + 40 : settle;
        for (var frame = 0; frame < frames; frame++)
        {
            using (scene.Push()) {
                camera.WorldPosition = eye;
                camera.WorldRotation = Rotation.LookAt(target - eye);
                camera.FieldOfView = view["fov"].GetValue<float>();
                camera.CustomSize = new Vector2(w, h);
                scene.EditorTick(.016f, .016f);
                preRender.Invoke(scene, null);
                preCameraRender.Invoke(scene, null);
                camera.RenderToBitmap(bitmap);
            }
            await Task.Delay(16, cancellationToken);
        }
        var png = bitmap.ToPng();
        System.IO.File.WriteAllBytes(System.IO.Path.Combine(destination, name + ".png"), png);
        written.Add(name);
        if (view["art"]?.GetValue<bool>() == true)
        {
            var art = System.IO.Path.Combine(projectRoot, "Assets/ui/maps/railhead.png");
            System.IO.File.WriteAllBytes(art, png);
            written.Add(art);
        }
    }
    var missing = scene.GetAllComponents<ModelRenderer>().Where(r => r.Model is null || r.Model.IsError).Select(r => r.GameObject.Name).Distinct().Take(20).ToArray();
    return new { Before = before, Game.IsPlaying,
        SourceUnchanged = Editor.SceneEditorSession.Active?.Scene?.Id == before.SceneId && Game.ActiveScene?.Id == before.GameSceneId,
        Written = written, Renderers = scene.GetAllComponents<ModelRenderer>().Count(), ErrorModels = missing,
        Probes = scene.GetAllComponents<EnvmapProbe>().Select(p => p.Mode.ToString()).Distinct().ToArray(), MarkedTextures = marked, Hidden = hidden };
} catch (Exception error) {
    System.IO.File.WriteAllText(System.IO.Path.Combine(configFolder, "capture-error.txt"), error.ToString());
    throw;
} finally { scene.Destroy(); }
