var results = new System.Collections.Generic.List<object>();
var root = Sandbox.Project.Current.GetRootPath();
Editor.AssetSystem.RegisterFile( System.IO.Path.Combine( root, "Assets/shaders/m4_optic_lens.shader" ) );
var shader = Editor.AssetSystem.FindByPath( "shaders/m4_optic_lens.shader" );
results.Add( new { path = "shader", ok = shader is not null, Shader = string.Join( ",", shader?.GetCompiledFile( true ) ?? "" ), Dust = 0f } );
foreach ( var path in paths )
{
	var asset = Editor.AssetSystem.FindByPath( path );
	var ok = asset is not null && asset.Compile( true );
	var mat = Material.Load( path );
	results.Add( new { path, ok, Shader = mat?.Shader?.ResourcePath, Dust = mat?.Attributes.GetFloat( "g_flDust" ) } );
}
return results;
