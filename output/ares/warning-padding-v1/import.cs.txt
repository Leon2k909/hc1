// Change only the warning atlas colour and its material's colour-texture path.
var root=Sandbox.Project.Current.GetRootPath();
if(!System.IO.Path.GetFullPath(root).TrimEnd('\\','/').Equals(@"C:\sbox-projects\hc1\forked-shooter-main",StringComparison.OrdinalIgnoreCase))throw new InvalidOperationException("Wrong project.");
var folder=System.IO.Path.Combine(root,"output/ares/warning-padding-v1");
var manifest=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder,"manifest.json")));
string Hash(string p)=>Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(System.IO.File.ReadAllBytes(p))).ToLowerInvariant();
var original=System.IO.Path.Combine(root,"Assets/materials/ares/ar_signs_color.jpg");
if(Hash(original)!=manifest["original_atlas_sha256"].GetValue<string>())throw new InvalidOperationException("Atlas changed during staging; keep the other work.");
var sourceText=System.IO.File.ReadAllText(System.IO.Path.Combine(root,"Tools/ares_materials.py")).Replace("\r\n","\n");
if(!sourceText.Contains(manifest["source_snippet"].GetValue<string>().Replace("\r\n","\n")))throw new InvalidOperationException("Warning generator changed.");
var session=Editor.SceneEditorSession.Active;var playing=Game.IsPlaying;var game=Game.ActiveScene?.Id;var dirty=session.HasUnsavedChanges;
var files=manifest["files"].AsObject();
foreach(var pair in files)
{
 if(pair.Key!="Assets/materials/ares/ar_signs_color.png"&&pair.Key!="Assets/materials/ares/ar_signs.vmat")throw new InvalidOperationException("Unexpected asset.");
 var target=System.IO.Path.Combine(root,pair.Key);var stage=System.IO.Path.Combine(folder,"stage",pair.Key);
 var before=pair.Value["before"]?.GetValue<string>();var after=pair.Value["after"].GetValue<string>();
 if(Hash(stage)!=after)throw new InvalidOperationException("Stage changed.");
 var current=System.IO.File.Exists(target)?Hash(target):null;
 if(current!=before&&current!=after)throw new InvalidOperationException("Shared asset changed: "+pair.Key);
}
foreach(var pair in files)
{
 var target=System.IO.Path.Combine(root,pair.Key);var after=pair.Value["after"].GetValue<string>();
 if(!System.IO.File.Exists(target)||Hash(target)!=after)
 {
  var incoming=target+".warning-incoming";
  try
  {
   System.IO.File.Copy(System.IO.Path.Combine(folder,"stage",pair.Key),incoming,true);
   if(Hash(incoming)!=after)throw new InvalidOperationException("Incoming texture changed.");
   System.IO.File.Move(incoming,target,true);
  }
  finally{if(System.IO.File.Exists(incoming))System.IO.File.Delete(incoming);}
 }
 Editor.AssetSystem.RegisterFile(target);
}
var preserved=ReferenceEquals(session,Editor.SceneEditorSession.Active)&&playing==Game.IsPlaying&&game==Game.ActiveScene?.Id&&dirty==session.HasUnsavedChanges;
var result=new{Imported=files.Select(x=>x.Key).ToArray(),EditorStatePreserved=preserved,SceneEdited=false};
System.IO.File.WriteAllText(System.IO.Path.Combine(folder,"imported.json"),System.Text.Json.JsonSerializer.Serialize(result));
if(!preserved)throw new InvalidOperationException("Editor state changed during the texture import.");
return result;
