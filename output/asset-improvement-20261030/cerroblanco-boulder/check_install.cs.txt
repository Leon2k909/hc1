// Are the garden's six boulders right on the map? Run with sbox-cli --call run_csharp, with the config path below filled in by check_config.py.
// Loads the scene file's visual roots into an isolated editor scene, so the open scene and play state are never touched and nothing is saved.
// Shoots the garden and each stone, and traces down through each: new stones should be hit by their own collision, and nothing should be left of the four old hulls.
var configPath = @"C:\sbox-projects\hc1\forked-shooter-main\output\asset-improvement-20261030\cerroblanco-boulder\check_config.json";
var configuration = System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(configPath)).AsObject();
var projectRoot = Sandbox.Project.Current.GetRootPath();
var destination = System.IO.Path.Combine(projectRoot, configuration["out"].GetValue<string>());
System.IO.Directory.CreateDirectory(destination);
var source = Editor.SceneEditorSession.Active?.Scene;
var before = new { Game.IsPlaying, Scene = source?.Name, SceneId = source?.Id };
var visual = new System.Collections.Generic.HashSet<string>(configuration["roots"].AsArray().Select(n => n.GetValue<string>()), StringComparer.OrdinalIgnoreCase);
var root = System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(projectRoot, configuration["scene"].GetValue<string>()))).AsObject();
var objects = new System.Text.Json.Nodes.JsonArray();
foreach (var item in root["GameObjects"].AsArray())
    if (visual.Contains(item["Name"]?.GetValue<string>() ?? "")) objects.Add(item.DeepClone());
root["GameObjects"] = objects;
if (root["SceneProperties"]?["NavMesh"] is System.Text.Json.Nodes.JsonObject nav) nav["Enabled"] = false;
if (root["SceneProperties"] is System.Text.Json.Nodes.JsonObject sceneProperties) sceneProperties["WantsSystemScene"] = false;
var scene = Scene.CreateEditorScene();
var flags = System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.Public | System.Reflection.BindingFlags.NonPublic;
var preRender = typeof(Scene).GetMethod("PreRender", flags, null, Type.EmptyTypes, null) ?? throw new MissingMethodException("Scene.PreRender");
var preCameraRender = typeof(Scene).GetMethod("PreCameraRender", flags, null, Type.EmptyTypes, null) ?? throw new MissingMethodException("Scene.PreCameraRender");
Vector3 Vec(System.Text.Json.Nodes.JsonNode node) { var a = node.AsArray(); return new(a[0].GetValue<float>(), a[1].GetValue<float>(), a[2].GetValue<float>()); }
try
{
    CameraComponent camera;
    var traces = new System.Collections.Generic.List<string>();
    var boulders = new System.Collections.Generic.List<string>();
    using (scene.Push())
    {
        scene.Deserialize(root);
        foreach (var tonemap in scene.GetAllComponents<Tonemapping>())
        {
            tonemap.AutoExposureEnabled = true;
            tonemap.MinimumExposure = tonemap.MaximumExposure = configuration["exposure"].GetValue<float>();
            tonemap.ExposureCompensation = 0;
            tonemap.Rate = 10;
        }
        var holder = new GameObject(false, "Capture camera");
        camera = holder.Components.Create<CameraComponent>();
        camera.ZNear = 1;
        camera.ZFar = 30000;
        holder.Enabled = true;
        scene.EditorTick(.016f, .016f);
        preRender.Invoke(scene, null);
        preCameraRender.Invoke(scene, null);
        foreach (var go in scene.GetAllObjects(false).Where(g => g.Name.StartsWith("Boulder ")))
        {
            var renderer = go.Components.Get<ModelRenderer>();
            boulders.Add($"{go.Name} at {go.WorldPosition} model {renderer?.Model?.ResourcePath} error {renderer?.Model?.IsError} physics {renderer?.Model?.Physics?.Parts?.Count}");
        }
    }
    var textureNames = new[] { "g_tColor", "g_tNormal", "g_tRoughness", "g_tAmbientOcclusion", "SkyTexture" };
    foreach (var material in scene.GetAllComponents<ModelRenderer>().Where(r => r.Model is not null).SelectMany(r => r.Model.Materials).Distinct())
        foreach (var texture in textureNames.Select(material.GetTexture).Append(material.FirstTexture).Where(t => t is not null).Distinct())
            texture.MarkUsed(material.ResourcePath.Contains("boulder_v2") ? 4096 : 1024);
    for (var wait = 0; wait < 150; wait++)
    {
        using (scene.Push()) { scene.EditorTick(.016f, .016f); preRender.Invoke(scene, null); }
        await Task.Delay(33, cancellationToken);
    }
    using (scene.Push())
    {
        foreach (var probe in configuration["probes"].AsArray())
        {
            var at = Vec(probe["at"]);
            var hit = scene.Trace.Ray(at + Vector3.Up * 200, at - Vector3.Up * 50).Run();
            traces.Add($"{probe["name"]}: hit {hit.Hit} z {hit.HitPosition.z:0.0} on {hit.GameObject?.Name}");
        }
    }
    var written = new System.Collections.Generic.List<string>();
    foreach (var view in configuration["views"].AsArray())
    {
        cancellationToken.ThrowIfCancellationRequested();
        var name = view["name"].GetValue<string>();
        using var bitmap = new Bitmap(configuration["width"].GetValue<int>(), configuration["height"].GetValue<int>());
        var eye = Vec(view["eye"]);
        var target = Vec(view["target"]);
        for (var frame = 0; frame < 30; frame++)
        {
            using (scene.Push())
            {
                camera.WorldPosition = eye;
                camera.WorldRotation = Rotation.LookAt(target - eye);
                camera.FieldOfView = view["fov"].GetValue<float>();
                scene.EditorTick(.016f, .016f);
                preRender.Invoke(scene, null);
                preCameraRender.Invoke(scene, null);
                camera.RenderToBitmap(bitmap);
            }
            await Task.Delay(16, cancellationToken);
        }
        System.IO.File.WriteAllBytes(System.IO.Path.Combine(destination, $"{name}.png"), bitmap.ToPng());
        written.Add(name);
    }
    return new
    {
        Before = before, SourceUnchanged = Editor.SceneEditorSession.Active?.Scene?.Id == before.SceneId, Boulders = boulders, Traces = traces, Written = written,
        Errors = scene.GetAllComponents<ModelRenderer>().Where(r => r.Model is null || r.Model.IsError).Select(r => r.GameObject.Name).ToArray()
    };
}
catch (Exception error)
{
    System.IO.File.WriteAllText(System.IO.Path.Combine(destination, "check-error.txt"), error.ToString());
    throw;
}
finally { scene.Destroy(); }
