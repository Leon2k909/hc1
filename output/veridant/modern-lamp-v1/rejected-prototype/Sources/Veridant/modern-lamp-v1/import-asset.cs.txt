// Install the independent fixture only. No map or scene assets are changed.
var root=Sandbox.Project.Current.GetRootPath();
if(!string.Equals(System.IO.Path.GetFullPath(root).TrimEnd('\\','/'),@"C:\sbox-projects\hc1\forked-shooter-main",StringComparison.OrdinalIgnoreCase))throw new InvalidOperationException("Wrong project.");
var folder=System.IO.Path.Combine(root,"output/veridant/modern-lamp-v1");
var manifest=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder,"asset-stage-v1/manifest.json")));
if(manifest["revision"].GetValue<string>()!="veridant-modern-lamp-asset-v1")throw new InvalidOperationException("Wrong revision.");
var session=Editor.SceneEditorSession.Active;var playing=Game.IsPlaying;var game=Game.ActiveScene?.Id;var dirty=session.HasUnsavedChanges;
string Hash(string path)=>Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(System.IO.File.ReadAllBytes(path))).ToLowerInvariant();
var scene=System.IO.Path.Combine(root,"Assets/scenes/maps/Veridant/veridant.scene");var sceneHash=Hash(scene);
foreach(var pair in manifest["files"].AsObject())
{
    var path=pair.Key;
    if(path.Contains("..")||!(path.StartsWith("Assets/materials/veridant/modern-lamp-v1/",StringComparison.Ordinal)||path.StartsWith("Assets/models/props/veridant/modern-lamp-v1/",StringComparison.Ordinal)))throw new InvalidOperationException("Unexpected target.");
    var source=System.IO.Path.Combine(root,pair.Value["source"].GetValue<string>());var target=System.IO.Path.Combine(root,path);var expected=pair.Value["sha256"].GetValue<string>();
    if(Hash(source)!=expected||(System.IO.File.Exists(target)&&Hash(target)!=expected))throw new InvalidOperationException("Shared file changed: "+path);
}
var imported=new System.Collections.Generic.List<string>();
foreach(var pair in manifest["files"].AsObject())
{
    var target=System.IO.Path.Combine(root,pair.Key);
    if(!System.IO.File.Exists(target))
    {
        System.IO.Directory.CreateDirectory(System.IO.Path.GetDirectoryName(target));
        var incoming=target+".lamp-incoming";
        try{System.IO.File.Copy(System.IO.Path.Combine(root,pair.Value["source"].GetValue<string>()),incoming,false);System.IO.File.Move(incoming,target,false);}
        finally{if(System.IO.File.Exists(incoming))System.IO.File.Delete(incoming);}
        imported.Add(pair.Key);
    }
    Editor.AssetSystem.RegisterFile(target);
}
var preserved=ReferenceEquals(session,Editor.SceneEditorSession.Active)&&playing==Game.IsPlaying&&game==Game.ActiveScene?.Id&&dirty==session.HasUnsavedChanges&&Hash(scene)==sceneHash;
var result=new{ImportedFiles=imported,MapChanged=false,EditorStatePreserved=preserved};
System.IO.File.WriteAllText(System.IO.Path.Combine(folder,"asset-import.json"),System.Text.Json.JsonSerializer.Serialize(result));
if(!preserved)throw new InvalidOperationException("Editor state changed during asset import; inspect before continuing.");
return result;
