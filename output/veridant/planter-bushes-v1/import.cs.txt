// Publish only the verified removal of the north pine planter's 120 grass triangles.
var root = Sandbox.Project.Current.GetRootPath();
var folder = System.IO.Path.Combine(root, "output/veridant/planter-bushes-v1/stage");
var manifest = System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder, "manifest.json")));
var session = Editor.SceneEditorSession.Active;
if (Game.IsPlaying || session.HasUnsavedChanges || session.Scene.Source?.ResourcePath != "scenes/maps/veridant/veridant.scene")
    throw new InvalidOperationException("Editor state changed; inspect before importing.");
string Hash(string path) => Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(System.IO.File.ReadAllBytes(path))).ToLowerInvariant();
var relative = manifest["source"].GetValue<string>();
if (relative != "Assets/models/maps/veridant/veridant_foliage.obj") throw new InvalidOperationException("Unexpected target.");
var target = System.IO.Path.Combine(root, relative);
var staged = System.IO.Path.Combine(folder, "veridant_foliage.obj");
var before = manifest["before"].GetValue<string>();
var after = manifest["after"].GetValue<string>();
if (Hash(staged) != after || Hash(target) != before) throw new InvalidOperationException("Shared mesh or stage changed.");
var vmdl = System.IO.Path.ChangeExtension(target, ".vmdl");
var vmdlHash = Hash(vmdl);
var sceneId = session.Scene.Id;
System.IO.File.Copy(staged, target, true);
Editor.AssetSystem.RegisterFile(target);
var result = new { Imported = relative, RemovedTriangles = 120, Before = before, After = Hash(target),
    CollisionSourceUnchanged = Hash(vmdl) == vmdlHash, ScenePreserved = session.Scene.Id == sceneId && !session.HasUnsavedChanges };
System.IO.File.WriteAllText(System.IO.Path.Combine(folder, "native-import.json"), System.Text.Json.JsonSerializer.Serialize(result));
return result;
