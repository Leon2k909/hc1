// Import only the reviewed market-stall cloth faces in the two existing town meshes.
var root=Sandbox.Project.Current.GetRootPath();
var folder=System.IO.Path.Combine(root,"output/belltower/canopy-cloth-v1/stage");
var manifest=System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder,"manifest.json")));
var selected=Editor.SceneEditorSession.Active;
if (Game.IsPlaying || selected is null || selected.HasUnsavedChanges ||
    !selected.Scene.Source.ResourcePath.Equals("scenes/maps/belltower/belltower.scene",StringComparison.OrdinalIgnoreCase))
    throw new InvalidOperationException("Requires the active, stopped, saved Belltower document.");
string Hash(string path) {
    using var stream=System.IO.File.OpenRead(path);
    return Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(stream)).ToLowerInvariant();
}
foreach(var group in new[]{"guards","source_sha256"})
    foreach(var pair in manifest[group].AsObject())
        if(Hash(System.IO.Path.Combine(root,pair.Key))!=pair.Value.GetValue<string>())
            throw new InvalidOperationException("Canopy input changed: "+pair.Key);
var allow=new System.Collections.Generic.HashSet<string>{
    "Assets/models/maps/belltower/belltower_west.obj","Assets/models/maps/belltower/belltower_east.obj"};
if(!allow.SetEquals(manifest["files"].AsObject().Select(p=>p.Key))) throw new InvalidOperationException("Unexpected canopy asset set.");
var scenePath=System.IO.Path.Combine(root,"Assets/scenes/maps/Belltower/belltower.scene");
var sceneHash=Hash(scenePath);var gameId=Game.ActiveScene?.Id;
var ids=selected.Scene.GetAllObjects(true).Select(o=>o.Id).ToHashSet();
var plan=new System.Collections.Generic.List<(string Target,string Source,string Backup,string After)>();
foreach(var file in manifest["files"].AsObject())
{
    var target=System.IO.Path.Combine(root,file.Key);var source=System.IO.Path.Combine(folder,file.Key);
    var before=file.Value["before"].GetValue<string>();var after=file.Value["after"].GetValue<string>();
    if(Hash(target)!=before || Hash(source)!=after) throw new InvalidOperationException("Canopy model changed since staging.");
    var backup=System.IO.Path.Combine(folder,"before",file.Key);
    System.IO.Directory.CreateDirectory(System.IO.Path.GetDirectoryName(backup));
    if(!System.IO.File.Exists(backup))System.IO.File.Copy(target,backup);
    if(Hash(backup)!=before)throw new InvalidOperationException("Canopy backup differs.");
    plan.Add((target,source,backup,after));
}
var changed=new System.Collections.Generic.List<(string Target,string Backup)>();
try {
    foreach(var item in plan) {
        var incoming=item.Target+".canopy-incoming";
        try {
            System.IO.File.Copy(item.Source,incoming,true);
            if(Hash(incoming)!=item.After)throw new InvalidOperationException("Incoming canopy data differs.");
            changed.Add((item.Target,item.Backup));System.IO.File.Move(incoming,item.Target,true);
        } finally {if(System.IO.File.Exists(incoming))System.IO.File.Delete(incoming);}
    }
} catch {foreach(var item in changed.AsEnumerable().Reverse())System.IO.File.Copy(item.Backup,item.Target,true);throw;}
foreach(var item in plan)Editor.AssetSystem.RegisterFile(item.Target);
if(!ReferenceEquals(selected,Editor.SceneEditorSession.Active) || Game.IsPlaying || Game.ActiveScene?.Id!=gameId ||
    selected.HasUnsavedChanges || Hash(scenePath)!=sceneHash || !ids.SetEquals(selected.Scene.GetAllObjects(true).Select(o=>o.Id)))
    throw new InvalidOperationException("Editor state changed during canopy import.");
var result=new {Imported=plan.Count,SceneSha256=sceneHash,SceneUnchanged=true,EditorStatePreserved=true,
    CollisionAndMaterialsUnchanged=true,Files=plan.Select(p=>new{p.Target,Sha256=Hash(p.Target)})};
System.IO.File.WriteAllText(System.IO.Path.Combine(folder,"native-import.json"),System.Text.Json.JsonSerializer.Serialize(result));
return result;
