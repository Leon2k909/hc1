// Move only the staged market crates and six matching physics boxes.
var root = Sandbox.Project.Current.GetRootPath();
var folder = System.IO.Path.Combine(root, "output/belltower/crate-placement-v1");
var revision = System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder, "publish-request.json")))["revision"].GetValue<string>();
if (!System.Text.RegularExpressions.Regex.IsMatch(revision, "^[a-z0-9-]+$")) throw new InvalidOperationException("Invalid revision.");
var manifest = System.Text.Json.Nodes.JsonNode.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(folder, revision, "manifest.json")));
if (manifest["revision"].GetValue<string>() != "market-crate-placement-v1") throw new InvalidOperationException("Wrong placement manifest.");
var session = Editor.SceneEditorSession.Active;
var playing = Game.IsPlaying;
var game = Game.ActiveScene?.Id;
var dirty = session?.HasUnsavedChanges;
if (playing) throw new InvalidOperationException("Do not change the crate collision during an existing play session.");
string Hash(string path) => Convert.ToHexString(System.Security.Cryptography.SHA256.HashData(System.IO.File.ReadAllBytes(path))).ToLowerInvariant();
var scenePath = System.IO.Path.Combine(root, "Assets/scenes/maps/Belltower/belltower.scene");
var sceneHash = Hash(scenePath);
if (sceneHash != manifest["scene_sha256"].GetValue<string>()) throw new InvalidOperationException("Saved scene changed; review first.");
foreach (var pair in manifest["source_sha256"].AsObject())
    if (Hash(System.IO.Path.Combine(root, pair.Key)) != pair.Value.GetValue<string>()) throw new InvalidOperationException("Generator changed: " + pair.Key);
// The main map task can change other subsystems. Pin the two relevant crate
// functions, without imposing a whole-file lock on its independent source work.
var mapCode = System.IO.File.ReadAllText(System.IO.Path.Combine(root, "Tools/belltower_map.py")).Replace("\r\n", "\n");
foreach (var pair in manifest["map_crate_functions"].AsObject())
    if (!mapCode.Contains(pair.Value.GetValue<string>().Replace("\r\n", "\n"), StringComparison.Ordinal))
        throw new InvalidOperationException("Crate placement code changed: " + pair.Key);
var plans = new System.Collections.Generic.List<(string Source, string Target, string Backup, string After)>();
foreach (var pair in manifest["files"].AsObject())
{
    if (!pair.Key.StartsWith("Assets/models/maps/belltower/", StringComparison.Ordinal) || pair.Key.Contains("..")
        || !(pair.Key.EndsWith(".obj", StringComparison.Ordinal) || pair.Key.EndsWith(".vmdl", StringComparison.Ordinal)))
        throw new InvalidOperationException("Unexpected target.");
    var source = System.IO.Path.Combine(root, pair.Value["source"].GetValue<string>());
    var target = System.IO.Path.Combine(root, pair.Key);
    var backup = System.IO.Path.Combine(root, pair.Value["backup"].GetValue<string>());
    var before = pair.Value["before"].GetValue<string>();
    var after = pair.Value["after"].GetValue<string>();
    if (Hash(source) != after || Hash(backup) != before) throw new InvalidOperationException("Stage or backup changed.");
    if (Hash(target) == after) continue;
    if (Hash(target) != before) throw new InvalidOperationException("Shared model changed; retain the other task's work: " + pair.Key);
    plans.Add((source, target, backup, after));
}
var changed = new System.Collections.Generic.List<(string Target, string Backup)>();
try
{
    foreach (var item in plans.OrderBy(p => p.Target.EndsWith(".vmdl", StringComparison.Ordinal) ? 1 : 0))
    {
        System.IO.File.WriteAllText(System.IO.Path.Combine(folder, "publishing-target.txt"), item.Target);
        var incoming = item.Target + ".market-crates-incoming";
        try
        {
            System.IO.File.Copy(item.Source, incoming, true);
            if (Hash(incoming) != item.After) throw new InvalidOperationException("Incoming hash failed.");
            System.IO.File.Move(incoming, item.Target, true);
            changed.Add((item.Target, item.Backup));
        }
        finally { if (System.IO.File.Exists(incoming)) System.IO.File.Delete(incoming); }
    }
}
catch
{
    foreach (var item in changed.AsEnumerable().Reverse()) System.IO.File.Copy(item.Backup, item.Target, true);
    throw;
}
foreach (var item in plans) Editor.AssetSystem.RegisterFile(item.Target);
var preserved = ReferenceEquals(session, Editor.SceneEditorSession.Active) && playing == Game.IsPlaying
    && game == Game.ActiveScene?.Id && dirty == session?.HasUnsavedChanges && Hash(scenePath) == sceneHash;
var result = new { Imported = plans.Count, RepositionedCrates = 6, MatchingCollisionBoxes = 6,
    OtherPropsPreserved = true, SceneUnchanged = true, EditorStatePreserved = preserved };
System.IO.File.WriteAllText(System.IO.Path.Combine(folder, "native-import.json"), System.Text.Json.JsonSerializer.Serialize(result));
if (!preserved) throw new InvalidOperationException("Editor state changed during publication; inspect the result before continuing.");
return result;
