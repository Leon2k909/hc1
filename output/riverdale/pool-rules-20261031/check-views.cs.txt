var scene=Game.ActiveScene;
var id=scene.Id;
var output=System.IO.Path.Combine(Sandbox.Project.Current.GetRootPath(),"output/riverdale/pool-rules-20261031");
var views=new[]{
    (Name:"north-final",Eye:new Vector3(1928,208,64),Target:new Vector3(1998.88f,223,57),Fov:50f),
    (Name:"west-upper",Eye:new Vector3(1860,910,351),Target:new Vector3(1880,998.88f,345),Fov:48f),
    (Name:"east-upper",Eye:new Vector3(1900,-910,351),Target:new Vector3(1880,-998.88f,345),Fov:48f)
};
var results=new System.Collections.Generic.List<string>();
foreach(var view in views){
    cancellationToken.ThrowIfCancellationRequested();
    var holder=scene.CreateObject(true);holder.Name="__pool_rules_temporary_capture";
    try{
        holder.WorldPosition=view.Eye;holder.WorldRotation=Rotation.LookAt(view.Target-view.Eye);
        var camera=holder.Components.Create<CameraComponent>();camera.FieldOfView=view.Fov;camera.ZNear=1;
        using var bitmap=new Bitmap(1200,1000);
        for(int frame=0;frame<10;frame++){camera.RenderToBitmap(bitmap,false);await Task.Delay(30,cancellationToken);}
        var path=System.IO.Path.Combine(output,view.Name+".png");System.IO.File.WriteAllBytes(path,bitmap.ToPng());results.Add(path);
    }finally{holder.Destroy();}
}
return new{Images=results,Game.IsPlaying,SameScene=scene.Id==id};
