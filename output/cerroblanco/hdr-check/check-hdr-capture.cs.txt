if(Game.IsPlaying)throw new InvalidOperationException("Stop play before the HDR capture check.");

var camera=scene.GetAllComponents<CameraComponent>().First(c=>c.GameObject.Name=="editor_camera");

var oldSun=sun.LightColor;
var previous=(camera.WorldPosition,camera.WorldRotation,camera.FieldOfView);
var previousPostProcessing=camera.EnablePostProcessing;
var previousSize=camera.CustomSize;
var flags=System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic|System.Reflection.BindingFlags.Public;
var root=Sandbox.Project.Current.GetRootPath();
var output=System.IO.Path.Combine(root,"output/cerroblanco/hdr-check");
System.IO.Directory.CreateDirectory(output);
try
{
    using(scene.Push())
    {
        camera.WorldPosition=new Vector3(0,80,150);camera.WorldRotation=Rotation.From(new Angles(0,0,0));camera.FieldOfView=90;camera.CustomSize=new Vector2(512,512);

        camera.RenderToTexture(texture,in setup);Graphics.FlushGPU();
        var pixels=new float[512*512*4];

        return new {Width=512,Height=512,Min=rgb.Min(),Max=rgb.Max(),Mean=rgb.Average(),Finite=rgb.All(float.IsFinite),NonZero=rgb.Count(p=>p>0),BoostedMax=boosted.Max(),BoostedMean=boosted.Average(),Directory=output};
    }
