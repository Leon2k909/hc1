// Survey: each gun's 1x optics, aimed and settled; how far each dot sits from the screen's centre.
var guns = new[] { ("m16","weapons/m16") };
var outFile = @"C:\Users\leon2\AppData\Local\Temp\claude\C--sbox-projects-hc1-forked-shooter-main\aba2264e-688e-481c-bccb-037769f12b6a\scratchpad\survey3\m16-cowitness.jsonl";

	cancellationToken.ThrowIfCancellationRequested();
	var resource = ResourceLibrary.Get<EquipmentResource>( $"{folder}/{name}.equip" );
	var manual = SceneUtility.GetPrefabScene( ResourceLibrary.Get<PrefabFile>( $"{folder}/weapon_{name}.prefab" ) )
		.Components.GetAll<Facepunch.ManualOffset>( FindMode.EverythingInSelfAndDescendants ).First();
	var scene = new Scene();
	try
	{
		using var scope = scene.Push();
		var root = SceneUtility.GetPrefabScene( ResourceLibrary.Get<PrefabFile>( $"{folder}/vm_{name}.prefab" ) ).Clone( new CloneConfig { StartEnabled = false } );
		root.Components.GetOrCreate<Facepunch.GunsmithWeapon>().Resource = resource;
		foreach ( var c in root.Components.GetAll<Component>( FindMode.EverythingInSelfAndDescendants ) ) c.Enabled = c is ModelRenderer || c is Facepunch.Gunsmith.GunsmithAttachmentPoint;
		var renderer = root.Components.Get<SkinnedModelRenderer>( FindMode.EverythingInSelf );
		renderer.RenderOptions.Game = false; renderer.RenderOptions.Overlay = true;
		renderer.UseAnimGraph = true; renderer.PlaybackRate = 1; renderer.PlayAnimationsInEditorScene = true;
		Facepunch.AnimBools.SetBool( renderer, "b_grounded", true );
		Facepunch.AnimBools.SetBool( renderer, "b_sprint", false );

		var point = root.Components.GetAll<Facepunch.Gunsmith.GunsmithAttachmentPoint>( FindMode.EverythingInSelfAndDescendants ).First( p => p.Category == Facepunch.Gunsmith.GunsmithPartType.Optic );

		for ( int i = 0; i < 200; i++ ) { scene.EditorTick( .02f, .02f ); renderer.SceneModel.Transform = renderer.WorldTransform; renderer.SceneModel.Update( .02f ); }

		light.WorldRotation = Rotation.From( new Angles( 40, 30, 0 ) ); light.LightColor = Color.White * 1.2f; light.SkyColor = Color.White * 0.4f;
		foreach ( var optic in optics )
		{
			if ( !optic.EligibleEquipment.Contains( resource ) && name != "m4a1" ) continue;
			var previous = point.CurrentGameObject;
			point.SetPart( optic ); previous?.DestroyImmediate();
			Facepunch.OpticSightLine line = null;
			var lowerThird = new[] { "att_vm_2x_west02_holo", "att_vm_holo_west02", "att_vm_minireddot01_tall", "att_vm_minireddot02_tall",
				"att_vm_minireddot03_tall", "att_vm_reflex_east02_tall", "att_vm_reflex_west02_tall" }.Contains( System.IO.Path.GetFileNameWithoutExtension( optic.ResourcePath ) );
			var drop = lowerThird ? 0.14f : 0f;
			var eye = Vector3.Zero;
			(float Up, float Left, float Distance) Tip()
			{
				renderer.TryGetBoneTransform( "tag_barrel_attach", out var barrel );
				var local = camera.WorldRotation.Inverse * (barrel.PointToWorld( new Vector3( 13.872f, 0f, 1.522f ) ) - camera.WorldPosition);
				return (MathF.Atan2( local.z, local.x ), MathF.Atan2( local.y, local.x ), local.Length);
			}
			void Place( Vector3 move )
			{
				var p = optic.OpticAimOffset + manual.OpticAimCorrection + move;
				var t = (optic.OpticAimAngles + new Angles( -0.0547f, 0.0031f, 0f )).ToRotation();
				root.WorldRotation = t; root.WorldPosition = t * p;
				for ( int i = 0; i < 10; i++ ) { scene.EditorTick( .02f, .02f ); renderer.SceneModel.Transform = renderer.WorldTransform; renderer.SceneModel.Update( .02f ); }
				var b = renderer.SceneModel.GetBoneLocalTransform( "camera" );
				camera.WorldPosition = b.Position; camera.WorldRotation = b.Rotation;
			}
			(float Up, float Left) Miss( Vector3 move ) { Place( move ); var t = Tip(); return (t.Up + drop / t.Distance, t.Left); }
			for ( int step = 0; step < 4; step++ )
			{
				var m0 = Miss( eye ); var mz = Miss( eye + Vector3.Up * 0.05f ); var my = Miss( eye + Vector3.Left * 0.05f );
				var dz = (mz.Up - m0.Up) / 0.05f; var dy = (my.Left - m0.Left) / 0.05f;
				eye += new Vector3( 0f, -m0.Left / dy, -m0.Up / dz );
			}
			// Keep the eye's line along the sight within the middle 60% of the window's height: past that the dot leaves a
			// small window, as it would for a real shooter's eye, and the post shows where the hardware puts it.
			var windows = new System.Collections.Generic.Dictionary<string, Vector4> { { "att_vm_2x_west02_holo", new Vector4( 0.0000f, 1.4105f, 0.6170f, 0.5845f ) }, { "att_vm_2x_west02", new Vector4( 0.0000f, 1.3645f, 0.5090f, 0.5085f ) }, { "att_vm_reflex_02", new Vector4( 0.0205f, 1.3325f, 0.6695f, 0.3775f ) }, { "att_vm_minireddot01_tall", new Vector4( 0.0030f, 1.0790f, 0.5640f, 0.3640f ) }, { "att_vm_minireddot02_tall", new Vector4( 0.0000f, 1.0780f, 0.5810f, 0.3630f ) }, { "att_vm_minireddot03_tall", new Vector4( 0.0020f, 1.1025f, 0.4530f, 0.3695f ) }, { "att_vm_holo_west01", new Vector4( 0.0010f, 1.3175f, 0.6260f, 0.5835f ) }, { "att_vm_holo_west02", new Vector4( 0.0030f, 1.3005f, 0.6990f, 0.5055f ) }, { "att_vm_holo_east01", new Vector4( -0.0180f, 1.1018f, 0.4563f, 0.4563f ) }, { "att_vm_reflex_east01", new Vector4( 0.0000f, 1.2550f, 0.4800f, 0.4800f ) }, { "att_vm_reflex_east02_tall", new Vector4( 0.0000f, 1.4010f, 0.5950f, 0.5950f ) }, { "att_vm_reflex_west02_tall", new Vector4( 0.0000f, 1.2545f, 0.4400f, 0.4395f ) }, { "att_vm_reflex_west03", new Vector4( -0.0005f, 1.3395f, 0.4815f, 0.4515f ) }, { "att_vm_thermal_east01", new Vector4( 0.0005f, 0.9430f, 0.5435f, 0.5440f ) }, { "att_vm_thermal_west01", new Vector4( -0.0000f, 0.9245f, 0.5620f, 0.5625f ) }, { "att_vm_thermal_east01_hybrid", new Vector4( 0.0005f, 0.9430f, 0.5435f, 0.5440f ) }, { "att_vm_hybrid_west01", new Vector4( 0.0000f, 1.0785f, 0.6330f, 0.6325f ) }, { "att_vm_hybrid_west02", new Vector4( -0.0195f, 1.3465f, 0.5215f, 0.5215f ) }, { "att_vm_reflex_west04", new Vector4( 0.0000f, 1.2880f, 0.6610f, 0.5520f ) }, { "att_vm_2x_west01", new Vector4( 0.0010f, 1.2400f, 0.5780f, 0.5780f ) }, { "att_vm_hybrid_west03", new Vector4( -0.0030f, 1.0025f, 0.6000f, 0.6005f ) }, { "att_vm_4x_east01_tall", new Vector4( 0.1140f, 1.4375f, 0.7420f, 0.7495f ) }, { "att_vm_4x_west01_tall", new Vector4( -0.0040f, 1.5410f, 0.5510f, 0.5510f ) }, { "att_vm_4x_west02_tall", new Vector4( 			var window = windows[System.IO.Path.GetFileNameWithoutExtension( optic.ResourcePath )];
			Vector3 EyeIn( Vector3 move ) { Place( move ); var r = point.CurrentGameObject.Components.Get<ModelRenderer>( FindMode.EverythingInSelfAndDescendants ); return r.WorldTransform.PointToLocal( camera.WorldPosition ); }
			var full = eye; var at0 = EyeIn( Vector3.Zero ); var atFull = EyeIn( full );
			var limit = 0.3f * 2f * window.w;
			var clamped = false;
			if ( MathF.Abs( atFull.z - window.y ) > limit )
			{
				var target = window.y + MathF.Sign( atFull.z - window.y ) * limit;
				var share = (target - at0.z) / (atFull.z - at0.z);
				eye = new Vector3( 0f, full.y, full.z * share.Clamp( 0f, 1f ) );
				clamped = true;
			}
			var eyeAt = EyeIn( eye );
			var solved = Miss( eye );
			line = new Facepunch.OpticSightLine { Optic = optic, Position = eye, Angles = Angles.Zero };
			foreach ( var lift in lifts ) {
			var position = optic.OpticAimOffset + manual.OpticAimCorrection + (line?.Position ?? Vector3.Zero) + Vector3.Up * lift;
			var angles = optic.OpticAimAngles + new Angles( -0.0547f, 0.0031f, 0f );
			var turn = angles.ToRotation();
			root.WorldRotation = turn; root.WorldPosition = turn * position;
			for ( int i = 0; i < 10; i++ ) { scene.EditorTick( .02f, .02f ); renderer.SceneModel.Transform = renderer.WorldTransform; renderer.SceneModel.Update( .02f ); }

			scene.EditorTick( .01f, .01f );
			bone = renderer.SceneModel.GetBoneLocalTransform( "camera" );
			camera.WorldPosition = bone.Position; camera.WorldRotation = bone.Rotation;
			using var bmp = new Bitmap( W, H );
			camera.RenderToBitmap( bmp ); camera.RenderToBitmap( bmp );
			// The reticle: the red blob nearest the centre, re-centred on itself so a holographic ring stays whole. Red parts
			// of an optic's body sit degrees away, outside the window.
			var focal = W * 0.5 / Math.Tan( Fov * Math.PI / 360.0 );
			var reds = new System.Collections.Generic.List<(int X, int Y, float Weight)>();
			var reach = (int)(focal * Math.Tan( 3.0 * Math.PI / 180 ));
			for ( int y = Math.Max( 0, H / 2 - reach ); y < Math.Min( H, H / 2 + reach ); y++ )
				for ( int x = Math.Max( 0, W / 2 - reach ); x < Math.Min( W, W / 2 + reach ); x++ )
				{
					var p = bmp.GetPixel( x, y );
					var red = p.r - MathF.Max( p.g, p.b );
					if ( p.r > 0.35f && red > 0.25f ) reds.Add( (x, y, red) );
				}
			double right = double.NaN, down = double.NaN; int count = 0;
			if ( reds.Count > 0 )
			{
				var seed = reds.OrderBy( r => (r.X - W / 2.0) * (r.X - W / 2.0) + (r.Y - H / 2.0) * (r.Y - H / 2.0) ).First();
				double cx = seed.X, cy = seed.Y;
				var radius = focal * Math.Tan( 1.0 * Math.PI / 180 );
				for ( int pass = 0; pass < 5; pass++ )
				{
					double sx = 0, sy = 0, sw = 0; count = 0;
					foreach ( var r in reds )
						if ( (r.X - cx) * (r.X - cx) + (r.Y - cy) * (r.Y - cy) < radius * radius ) { sx += r.X * r.Weight; sy += r.Y * r.Weight; sw += r.Weight; count++; }
					cx = sx / sw; cy = sy / sw;
				}
				right = Math.Atan( (cx - (W - 1) / 2.0) / focal ) * 180 / Math.PI;
				down = Math.Atan( (cy - (H - 1) / 2.0) / focal ) * 180 / Math.PI;
			}
			// Where a collimated reticle shows: the optic model's own forward axis, seen from the camera.
			var opticRenderer = point.CurrentGameObject.Components.Get<ModelRenderer>( FindMode.EverythingInSelfAndDescendants );
			var axis = camera.WorldRotation.Inverse * opticRenderer.WorldRotation.Forward;
			if ( axis.x < 0 ) axis = -axis;
			var axisRight = -Math.Atan2( axis.y, axis.x ) * 180 / Math.PI;
			var axisDown = -Math.Atan2( axis.z, axis.x ) * 180 / Math.PI;
			var json = Json.Serialize( new { gun = name, optic = System.IO.Path.GetFileNameWithoutExtension( optic.ResourcePath ), opticName = optic.Name,
				lift, right = Math.Round( right, 4 ), down = Math.Round( down, 4 ), axisRight = Math.Round( axisRight, 4 ), axisDown = Math.Round( axisDown, 4 ),
				pixels = count, line = line is not null, linePosition = line.Position.ToString(), fullZ = full.z, clamped, eyeZ = eyeAt.z - window.y, halfH = window.w, eyeY = eyeAt.y - window.x, at0Z = at0.z - window.y, lowerThird, tipUp = solved.Up * 180 / Math.PI, tipLeft = solved.Left * 180 / Math.PI, correction = manual.OpticAimAngleCorrection.ToString(), lineAngles = line?.Angles.ToString(),
				opticAngles = optic.OpticAimAngles.ToString(), opticRoll = opticRenderer.WorldRotation.Angles().roll } );
			lines.Add( json );
			await System.IO.File.AppendAllTextAsync( outFile, json.Replace( "\n", "" ).Replace( "\r", "" ) + "\n" );
			}
		}
	}
	finally { scene.Destroy(); }
}
return lines.Count;
