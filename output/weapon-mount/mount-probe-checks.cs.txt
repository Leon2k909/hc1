// Weapon mount probe checks in a scene of its own (the editor's scenes and play state are untouched). Builds one piece of
// cover at a time in front of an eye at the origin looking along +x, and runs PlayerPawn.FindMountTop (the game's own
// probe, by reflection) and the old single-ray rule (43 ahead, copied here) standing (eye 64) and crouched (eye 48).
var type = AppDomain.CurrentDomain.GetAssemblies().Reverse().Select( a => a.GetType( "Facepunch.PlayerPawn" ) ).First( t => t is not null );
var probe = type.GetMethod( "FindMountTop", System.Reflection.BindingFlags.Static | System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Public );
if ( probe is null ) return new { Error = "FindMountTop isn't loaded yet (hotload pending)" };
var scene = new Scene();
var results = new System.Collections.Generic.List<object>();
try
{
	using ( scene.Push() )
	{
		var floor = new GameObject( true, "floor" );
		var floorBox = floor.Components.Create<BoxCollider>(); floorBox.Scale = new Vector3( 2000, 2000, 10 ); floorBox.Center = new Vector3( 0, 0, -5 );
		GameObject Box( string name, Vector3 min, Vector3 max, float pitch = 0f )
		{
			var go = new GameObject( true, name );
			go.WorldPosition = (min + max) / 2; go.WorldRotation = Rotation.FromPitch( pitch );
			var box = go.Components.Create<BoxCollider>(); box.Scale = max - min; box.Static = true;
			return go;
		}
		bool Old( Scene s, Vector3 eye, Vector3 forward, float ground )
		{
			var front = eye + forward * 43f;
			var maxTop = MathF.Min( eye.z + 1f, ground + 64f ); var minTop = MathF.Max( ground + 23f, eye.z - 28f );
			var top = s.Trace.Ray( front.WithZ( maxTop + 5f ), front.WithZ( minTop - 3f ) ).Run();
			if ( !top.Hit || top.StartedSolid || top.Normal.z < 0.85f || top.HitPosition.z < minTop || top.HitPosition.z > maxTop ) return false;
			var behind = eye + forward * 13f;
			var near = s.Trace.Ray( behind.WithZ( top.HitPosition.z + 4f ), behind.WithZ( top.HitPosition.z - 8f ) ).Run();
			if ( near.Hit && MathF.Abs( near.HitPosition.z - top.HitPosition.z ) < 5f ) return false;
			return !s.Trace.Ray( eye + forward * 10f, eye + forward * 55f ).Run().Hit;
		}
		var cases = new System.Collections.Generic.List<(string Name, bool Expect, System.Func<GameObject[]> Build)>
		{
			("waist wall 40 high, 12 thick, 18 ahead", true, () => new[] { Box( "w", new Vector3( 18, -60, 0 ), new Vector3( 30, 60, 40 ) ) }),
			("waist wall 44 high, 24 thick, 20 ahead", true, () => new[] { Box( "w", new Vector3( 20, -60, 0 ), new Vector3( 44, 60, 44 ) ) }),
			("window sill 40, 8 deep, wall above from 84", true, () => new[] { Box( "s", new Vector3( 17, -60, 0 ), new Vector3( 25, 60, 40 ) ), Box( "a", new Vector3( 17, -60, 84 ), new Vector3( 25, 60, 140 ) ) }),
			("crate 40 cube, 18 ahead", true, () => new[] { Box( "c", new Vector3( 18, -20, 0 ), new Vector3( 58, 20, 40 ) ) }),
			("thin rail 42 high (2 deep), 20 ahead", true, () => new[] { Box( "r", new Vector3( 20, -60, 40 ), new Vector3( 22, 60, 42 ) ), Box( "p", new Vector3( 20, -1, 0 ), new Vector3( 22, 1, 40 ) ) }),
			("stairs (7 up, 11 deep) from 16 ahead", false, () => Enumerable.Range( 0, 8 ).Select( k => Box( "t" + k, new Vector3( 16 + 11 * k, -40, 0 ), new Vector3( 300, 40, 7 * (k + 1) ) ) ).ToArray()),
			("ramp 30 degrees from 16 ahead", false, () => new[] { Box( "ramp", new Vector3( 16, -40, -20 ), new Vector3( 216, 40, 20 ), -30f ) }),
			("tall wall 100 high, 18 ahead", false, () => new[] { Box( "w", new Vector3( 18, -60, 0 ), new Vector3( 30, 60, 100 ) ) }),
			("low box 20 high, 18 ahead", false, () => new[] { Box( "b", new Vector3( 18, -60, 0 ), new Vector3( 30, 60, 20 ) ) }),
			("waist wall 40 high, 50 ahead (out of reach)", false, () => new[] { Box( "w", new Vector3( 50, -60, 0 ), new Vector3( 62, 60, 40 ) ) }),
		};
		foreach ( var c in cases )
		{
			var made = c.Build();
			foreach ( var (stance, eyeZ) in new[] { ("standing", 64f), ("crouched", 48f) } )
			{
				var eye = new Vector3( 0, 0, eyeZ );
				var args = new object[] { scene, null, eye, Vector3.Forward, 0f, null, null };
				var found = (bool)probe.Invoke( null, args );
				results.Add( new { Case = c.Name, Stance = stance, Expect = c.Expect, New = found, Pass = found == c.Expect, Old = Old( scene, eye, Vector3.Forward, 0f ),
					Contact = found ? ((Vector3)args[5]).ToString() : null } );
			}
			foreach ( var go in made ) go.Destroy();
			scene.ProcessDeletes();
		}
	}
}
finally { scene.Destroy(); }
return results;
